simetri.helpers.help_utils

Interactive and topic-based help for Simetri (sg.help).

sg.help(obj) accepts a string key or a callable/class/instance:

  • String keys look up defaults_help[obj] (empty string if missing), except for reserved topic names described below. Unknown topic strings return similar topic names.
  • Classes (and instances of Simetri types) return the constructor signature, class docstring, and __init__ docstring.
  • Simetri functions and methods return a signature (with resolved defaults where applicable), accepted **kwargs names when known, then the docstring.
  • Other modules and objects return inspect.getdoc(obj).

d_help_topic maps topic names to curated notes and highlight sg.* names. help_topics_generated.py (from python -m simetri.helpers.compile_help_topics) adds every matching public export per topic. help_topic_supplements.py adds semantic links (dict / many-to-many graph / aliases). sg.help('help') loads the help-utilities guide. sg.help(sg.help) summarizes how help lookup works. sg.help('topics') lists available topics.

 103def normalize(word: str) -> str:
 104    """Normalize a string for fuzzy help lookup (NFKC, casefold, strip).
 105
 106    Examples:
 107        >>> from simetri.helpers.help_utils import normalize
 108        >>> normalize('  Shape  ')
 109        'shape'
 110    """
 111    return unicodedata.normalize("NFKC", word).casefold().strip()

 114def _user_config_help_for_string(query: str) -> str | None:
 115    """Resolve config help before topic aliases when appropriate."""
 116    if query in user_config_sections():
 117        if query in USER_CONFIG_TOPIC_PREEMPT:
 118            return user_config_help(query)
 119        if query in ("tex", "viewer"):
 120            return None
 121        if query not in _TOPIC_ALIASES and query not in all_help_topic_keys():
 122            if query not in _topic_guide_aliases():
 123                return user_config_help(query)
 124    if "." in query:
 125        section, _, key = query.partition(".")
 126        if key and section in user_config_sections():
 127            return user_config_help(query)
 128    if (
 129        query in _TOPIC_ALIASES
 130        or query in all_help_topic_keys()
 131        or query in _topic_guide_aliases()
 132        or query in _topic_help_hubs()
 133    ):
 134        return None
 135    return user_config_help(query)

 145def find_similar(
 146    query: str,
 147    words: Sequence[str],
 148    threshold: float = 0.75,
 149    limit: int | None = None,
 150) -> list[tuple[str, float]]:
 151    """Return words similar to ``query`` by Damerau–Levenshtein similarity.
 152
 153    Args:
 154        query: Search string (normalized before comparison).
 155        words: Candidate names.
 156        threshold: Minimum normalized similarity in ``[0, 1]``. Defaults to 0.75.
 157        limit: Maximum matches; ``None`` uses ``defaults['help_suggestion_limit']``.
 158
 159    Returns:
 160        list[tuple[str, float]]: ``(word, score)`` pairs, highest score first.
 161
 162    Examples:
 163        >>> from simetri.helpers.help_utils import find_similar
 164        >>> find_similar('shpe', ['shape', 'group'], limit=2)[0][0]
 165        'shape'
 166    """
 167    limit = _resolve_help_suggestion_limit(limit)
 168    query_normalized = normalize(query)
 169    matches = []
 170
 171    for word in words:
 172        score = DamerauLevenshtein.normalized_similarity(
 173            query_normalized, normalize(word)
 174        )
 175
 176        if score >= threshold:
 177            matches.append((word, score))
 178
 179    return sorted(matches, key=lambda item: item[1], reverse=True)[:limit]

 182def _warning_type_path(obj: object) -> str | None:
 183    """Return ``WarningType…`` path for a subgroup class or leaf member."""
 184    if obj is WarningType:
 185        return "WarningType"
 186    if inspect.isclass(obj) and obj in _WARNING_SUBGROUPS:
 187        return f"WarningType.{_WARNING_SUBGROUPS[obj]}"
 188    if isinstance(obj, Enum) and type(obj) in _WARNING_SUBGROUPS:
 189        group_name = _WARNING_SUBGROUPS[type(obj)]
 190        return f"WarningType.{group_name}.{obj.name}"
 191    return None

 194def _warning_type_help(obj: object) -> str:
 195    """Return help text for ``WarningType``, a subgroup, or a leaf."""
 196    if obj is WarningType:
 197        group_lines = [
 198            f"  WarningType.{name}"
 199            for name in sorted(_WARNING_SUBGROUPS.values())
 200        ]
 201        base = inspect.getdoc(WarningType) or ""
 202        return f"{base}\n\nSubgroups:\n" + "\n".join(group_lines)
 203
 204    if inspect.isclass(obj) and obj in _WARNING_SUBGROUPS:
 205        base = inspect.getdoc(obj) or ""
 206        member_lines = []
 207        for member in obj:
 208            member_doc = inspect.getdoc(member) or ""
 209            member_lines.append(f"  {member.name}: {member_doc}")
 210        return f"{base}\n\nMembers:\n" + "\n".join(member_lines)
 211
 212    doc = inspect.getdoc(obj)
 213    return doc if doc is not None else ""

1225def _format_signature(
1226    name: str,
1227    signature: inspect.Signature,
1228    default_overrides: dict[str, object] | None = None,
1229) -> str:
1230    """Return a readable display form for a callable signature."""
1231    parts: list[str] = []
1232    has_var_positional = False
1233    inserted_kw_separator = False
1234    if default_overrides is None:
1235        default_overrides = {}
1236    for parameter in signature.parameters.values():
1237        if (
1238            parameter.kind == inspect.Parameter.KEYWORD_ONLY
1239            and not has_var_positional
1240            and not inserted_kw_separator
1241        ):
1242            parts.append("*")
1243            inserted_kw_separator = True
1244
1245        parts.append(
1246            _format_parameter(
1247                parameter,
1248                default_override=default_overrides.get(
1249                    parameter.name, inspect.Signature.empty
1250                ),
1251            )
1252        )
1253        if parameter.kind == inspect.Parameter.VAR_POSITIONAL:
1254            has_var_positional = True
1255
1256    text = f"{name}({', '.join(parts)})"
1257    if signature.return_annotation is not inspect.Signature.empty:
1258        text += f" -> {_format_annotation(signature.return_annotation)}"
1259
1260    if len(text) <= _MAX_INLINE_SIGNATURE_LENGTH:
1261        return text
1262
1263    lines = [f"{name}("]
1264    lines.extend(f"    {part}," for part in parts)
1265
1266    closing = ")"
1267    if signature.return_annotation is not inspect.Signature.empty:
1268        closing += f" -> {_format_annotation(signature.return_annotation)}"
1269    lines.append(closing)
1270
1271    return "\n".join(lines)

1652def _parse_qmd_help_aliases(path: Path) -> tuple[str, ...]:
1653    """Return ``help_aliases`` from YAML front matter when present."""
1654    try:
1655        text = path.read_text(encoding="utf-8")
1656    except OSError:
1657        return ()
1658    if not text.startswith("---"):
1659        return ()
1660    end = text.find("\n---", 3)
1661    if end == -1:
1662        return ()
1663    aliases: list[str] = []
1664    in_aliases = False
1665    for line in text[3:end].splitlines():
1666        if re.match(r"^help_aliases:\s*$", line):
1667            in_aliases = True
1668            continue
1669        inline = re.match(r"^help_aliases:\s*\[(.*)\]\s*$", line)
1670        if inline:
1671            inner = inline.group(1)
1672            for part in inner.split(","):
1673                part = part.strip().strip("'\"")
1674                if part:
1675                    aliases.append(part)
1676            return tuple(aliases)
1677        if in_aliases:
1678            item = re.match(r"^\s+-\s+(.+)$", line)
1679            if item:
1680                aliases.append(item.group(1).strip().strip("'\""))
1681                continue
1682            if line and not line.startswith(" "):
1683                break
1684    return tuple(aliases)
