  63    def __init__(
  64        self,
  65        pdf_path: str,
  66        pos: PointType = (0, 0),
  67        size: Sequence[int] | None = None,
  68        **kwargs: object,
  69    ) -> None:
  70        """Initialize a PDF drawable.
  71
  72        Args:
  73            pdf_path: Path to the PDF file.
  74            pos: Placement anchor point on the canvas.

  79            FileNotFoundError: If ``pdf_path`` does not exist.

  82            >>> import os
  83            >>> import tempfile

  88            ...         _ = handle.write(b"%PDF-1.0\\n%%EOF\\n")
  89            ...     pdf = PDF(path)
  90            ...     pdf.pdf_path == path
  91            True

  96        self.type = Types.PDF_SKETCH
  97        self.subtype = Types.PDF_SKETCH
  98        self.anchor = kwargs.get("anchor", Anchor.CENTER)

 155
 156    Pillow method names are available via attribute delegation. See the
 157    `Pillow docs <https://pillow.readthedocs.io/en/stable/>`_ for raster ops.
 158
 159    Examples:
 160        >>> import simetri.graphics as sg
 161        >>> isinstance(sg.Image(size=(2, 2), mode="RGB"), sg.Image)

 165    def __init__(
 166        self,
 167        img: str | PIL_Image.Image | None = None,
 168        pos: PointType = (0, 0),
 169        size: Sequence[int] | None = None,
 170        mode: ImageMode | str = ImageMode.RGB,
 171        **kwargs: object,
 172    ) -> None:
 173        """Initialize an image drawable.
 174
 175        Args:
 176            img: File path, Pillow image, or ``None`` to create a blank image.
 177            pos: Canvas point for the image ``anchor``.
 178            size: Required when ``img`` is ``None`` (new image dimensions).
 179            mode: Pillow mode when creating a blank image.
 180            **kwargs: Forwarded to :class:`~simetri.shapes.geom_items.Rectangle`
 181                and Pillow ``Image.new`` when applicable.
 182                ``anchor`` chooses which point of the image sits on ``pos``.
 183
 184        Raises:
 185            FileNotFoundError: If ``img`` is a path that does not exist.
 186            TypeError: If ``img`` is not a path, Pillow image, or ``None``.
 187
 188        Examples:
 189            >>> import simetri.graphics as sg
 190            >>> im = sg.Image(size=(10, 10), mode="RGB")
 191            >>> im.width
 192            10
 193            >>> im.mode
 194            'RGB'
 195            >>> im = sg.Image(
 196            ...     size=(40, 20),
 197            ...     mode="RGB",
 198            ...     pos=(20, 60),
 199            ...     anchor=sg.Anchor.SOUTHWEST,
 200            ... )
 201            >>> im.anchor == sg.Anchor.SOUTHWEST
 202            True
 203            >>> im.pos[0], im.pos[1]
 204            (20.0, 60.0)
 205            >>> im.southwest[0], im.southwest[1]
 206            (20.0, 60.0)
 207        """
 208        anchor = kwargs.pop("anchor", Anchor.CENTER)
 209        file_path = None
 210        if img is None:
 211            img = PIL_Image.new(mode=mode, size=size, **kwargs)
 212            width, height = size
 213        elif isinstance(img, str):
 214            if os.path.exists(img):
 215                file_path = img
 216                img = PIL_Image.open(img)
 217                width, height = img.size
 218            else:
 219                raise FileNotFoundError(f"File {img} not found.")
 220        elif isinstance(img, PIL_Image.Image):
 221            width, height = img.size
 222        elif not isinstance(img, PIL_Image.Image):
 223            raise TypeError("img must be a PIL Image object or a file path.")
 224        self.__dict__["pil_img"] = img
 225        kwargs["fill"] = False
 226        kwargs["stroke"] = False
 227        left, bottom, right, top = _anchor_bounds(
 228            anchor, pos, width, height
 229        )
 230        center = ((left + right) / 2, (bottom + top) / 2)
 231        super().__init__(width, height, center=center, **kwargs)
 232        self.file_path = file_path
 233        self.type = Types.IMAGE
 234        self.subtype = Types.IMAGE
 235        self.__dict__["anchor"] = anchor
 236        if "xform_matrix" in kwargs:
 237            self.xform_matrix = kwargs["xform_matrix"]
 238        else:
 239            self.xform_matrix = identity_matrix()

 299    def _update(
 300        self,
 301        xform_matrix: "array",
 302        reps: int = 0,
 303        take: slice | None = None,
 304        incr: float
 305        | tuple[float, float]
 306        | tuple[callable, Any]
 307        | tuple[InPlace, Any]
 308        | NDArray
 309        | Sequence[Sequence[float]]
 310        | None = None,
 311        dyn_ref: bool | None = None,
 312        merge: bool = False,
 313        xform_type: TransformationType | None = None,
 314    ) -> Group | Image:
 315        """Used internally. Update the shape with a transformation matrix.
 316
 317        Args:
 318            xform_matrix (array): The transformation matrix.
 319            reps (int, optional): The number of repetitions, defaults to 0.
 320            take: Not supported; must be ``None``.
 321            incr: Increment applied between repetitions when ``reps > 0``.
 322            merge: If True and ``reps > 0``, merge the copies.
 323            xform_type: Transform kind used with ``incr``.
 324
 325        Returns:
 326            Group: The updated shape or a group of shapes.
 327
 328        Raises:
 329            ValueError: If ``take`` is set, or if ``dyn_ref`` is used.
 330        """
 331        if dyn_ref:
 332            raise ValueError(
 333                "Image does not support dynamic references. Only Shape and "
 334                "Group resolve dyn_ref."
 335            )
 336        if take is not None:
 337            raise ValueError(
 338                "Image._update does not support take=; transform the whole image."
 339            )
 340        if reps == 0:
 341            self.xform_matrix = self.xform_matrix @ xform_matrix
 342            if "_final_coords" in self.__dict__:
 343                delattr(self, "_final_coords")
 344            if "_vertices" in self.__dict__:
 345                delattr(self, "_vertices")
 346            return self
 347        images = [self]
 348        image = self
 349        for i in range(reps):
 350            if incr is not None and i > 0:
 351                xform_matrix = _update_inplace(xform_matrix, xform_type, incr)
 352            image = image.copy()
 353            image._update(xform_matrix)
 354            images.append(image)
 355        res = Group(images)
 356        if merge:
 357            return res.merge_images()
 358        return res

 360    def _drawn_extent(self, pixels: int, axis: int) -> int | float:
 361        """Drawn length on one axis.
 362
 363        ``axis`` 0 is width and 1 is height. A scale of 1 returns the
 364        pixel count unchanged.
 365        """
 366        _translation, _rotation, scale = decompose_transformations(
 367            self.xform_matrix
 368        )
 369        factor = float(scale[axis])
 370        if factor == 1:
 371            return pixels
 372        return pixels * factor
