simetri.group.batch

Group containers for Shape, Group, and Tag objects.

A Group applies transforms to its members and supports set-like geometry operations (union, intersection, …) and edge merging.

Examples:
>>> import simetri.graphics as sg
>>> g = sg.Group(
...     [
...         sg.Shape([(0, 0), (40, 0), (40, 40)], closed=True),
...         sg.Shape([(20, 0), (30, 0)]),
...     ]
... )
>>> len(g)
2

 129    """Collection of drawable elements that transform together.

 131    Elements may be Shape, Group, or Tag objects. Methods such as
 132    ``all_vertices``, ``all_edges``, ``all_segments``, and ``all_shapes``
 133    flatten nested groups recursively.
 134    Alias: ``Batch`` on the ``simetri.graphics`` namespace.

 137        elements: Flat list of top-level members.
 138        type: Always ``Types.GROUP``.

 140        modifiers: Optional animation/constraint modifiers.
 141        visible: Whether the group is drawn.

 145        >>> import simetri.graphics as sg

 163    def __init__(
 164        self,
 165        *elements: Any,
 166        modifiers: Sequence[Modifier] | None = None,
 167        subtype: Types = Types.GROUP,
 168    ) -> None:
 169        """Initialize a Group.
 170
 171        Args:
 172            *elements: Drawables to include. Pass them individually
 173                (``Group(a, b)``) or as one sequence (``Group([a, b])``).
 174                Nested lists/tuples are flattened. Omit for an empty group.
 175            modifiers: Optional modifiers applied to the group.
 176            subtype: Group subtype enum or name. Defaults to ``Types.GROUP``.
 177
 178        Examples:
 179            >>> import simetri.graphics as sg
 180            >>> len(sg.Group([sg.Shape([(0, 0), (40, 0)])]))
 181            1
 182            >>> len(sg.Group())
 183            0
 184        """
 185
 186        if not elements:
 187            self.elements = []
 188        elif len(elements) == 1 and isinstance(elements[0], (list, tuple)):
 189            _elements = []
 190            for element in elements[0]:
 191                if isinstance(element, (list, tuple)):
 192                    _elements.extend(elem for elem in flatten2(element) if elem)
 193                else:
 194                    if element:
 195                        _elements.append(element)
 196            self.elements = _elements[:]
 197        elif len(elements) == 1:
 198            self.elements = [elements[0]]
 199        else:
 200            _elements = []
 201            for element in elements:
 202                if isinstance(element, (list, tuple)):
 203                    _elements.extend(elem for elem in flatten2(element) if elem)
 204                else:
 205                    if element:
 206                        _elements.append(element)
 207            self.elements = _elements[:]
 208
 209        self.type = Types.GROUP
 210        self.subtype = get_enum_value(Types, subtype)
 211        if modifiers is None:
 212            modifiers = []
 213        self.modifiers = modifiers
 214        self.visible = True
 215        self.id = get_unique_id(self)

 217    def set_style(self, mapping: Any = None, **kwargs: object) -> Self:
 218        """Set style fields on every member (mutated).
 219
 220        Nested groups are walked like ``set_attribs``. A missing attribute
 221        on a member raises.
 222
 223        Args:
 224            mapping: A ``Style``, a dict of draw aliases, or omitted.
 225            **kwargs: Draw-alias fields; overwrite ``mapping`` for those keys.
 226
 227        Examples:
 228            >>> import simetri.graphics as sg
 229            >>> g = sg.Group([sg.Shape([(0, 0), (40, 0)])])
 230            >>> g.set_style(line_width=3) is g
 231            True
 232            >>> g[0].line_width
 233            3
 234        """
 235        if mapping is None and not kwargs:
 236            raise TypeError(
 237                "Group.set_style() requires a Style, a dict, or keyword arguments"
 238            )
 239        overlay = coerce_style_overlay(mapping, kwargs)
 240        for element in self.elements:
 241            if element.type == Types.GROUP:
 242                element.set_style(overlay)
 243            else:
 244                for key, value in overlay.items():
 245                    setattr(element, key, value)
 246        return self

 276    def set_attribs(
 277        self, attrib: str, value: Any, key: Callable | None = None
 278    ) -> Self:
 279        """
 280        Sets the attribute to the given value for all elements in the group if it is applicable.
 281
 282        Args:
 283            attrib (str): The attribute to set.
 284            value (Any): The value to set the attribute to.
 285            key (Callable, optional): A function to filter elements by a specific key. Defaults to None.
 286
 287        Returns:
 288            Self: The group object.
 289
 290        Examples:
 291            >>> import simetri.graphics as sg
 292            >>> g = sg.Group([sg.Shape([(0, 0), (40, 0)])])
 293            >>> _ = g.set_attribs("line_width", 2)
 294            >>> g[0].line_width
 295            2
 296        """
 297        for element in self.elements:
 298            if key is not None:
 299                if key(element):
 300                    if element.type == Types.GROUP:
 301                        element.set_attribs(attrib, value, key=key)
 302                    elif hasattr(element, attrib):
 303                        setattr(element, attrib, value)
 304            else:
 305                if element.type == Types.GROUP:
 306                    element.set_attribs(attrib, value)
 307                elif hasattr(element, attrib):
 308                    setattr(element, attrib, value)
 309
 310        return self

 522    def check_dist_tol(
 523        self,
 524        n: int,
 525        n_round: int | None = None,
 526    ) -> set[float]:
 527        """Return up to ``n`` smallest positive rounded vertex distances.
 528
 529        Args:
 530            n: Number of distinct distances to return.
 531            n_round: Number of decimal places used to round distances. If
 532                ``None``, uses the configured ``n_round`` default.
 533
 534        Returns:
 535            set[float]: Up to ``n`` smallest positive rounded distances.
 536
 537        Examples:
 538            >>> import simetri.graphics as sg
 539            >>> group = sg.Group([sg.Shape([(0, 0), (5e-14, 0), (40, 0)])])
 540            >>> group.check_dist_tol(2, n_round=12) == {40.0}
 541            True
 542        """
 543        if n <= 0:
 544            raise ValueError("n must be a positive integer.")
 545        if n_round is None:
 546            n_round = runtime_defaults["n_round"]
 547        if n_round < 0:
 548            raise ValueError("n_round must be a nonnegative integer.")
 549
 550        vertices = self.all_vertices
 551        if len(vertices) < 2:
 552            return set()
 553
 554        distances = set()
 555        for first_point, second_point in combinations(vertices, 2):
 556            point_distance = round(
 557                distance(first_point[:2], second_point[:2]),
 558                n_round,
 559            )
 560            if point_distance > 0:
 561                distances.add(point_distance)
 562
 563        return set(sorted(distances)[:n])

 565    def check_angle_tol(
 566        self,
 567        n: int,
 568        n_round: int | None = None,
 569    ) -> set[float]:
 570        """Return up to ``n`` smallest positive rounded angle differences.
 571
 572        Args:
 573            n: Number of distinct angle differences to return.
 574            n_round: Number of decimal places used to round angle differences.
 575                If ``None``, uses the configured ``n_round`` default.
 576
 577        Returns:
 578            set[float]: Up to ``n`` smallest positive rounded angle differences.
 579
 580        Examples:
 581            >>> import simetri.graphics as sg
 582            >>> group = sg.Group(
 583            ...     [
 584            ...         sg.Shape([(0, 0), (40, 0)]),
 585            ...         sg.Shape([(0, 0), (0, 40)]),
 586            ...     ]
 587            ... )
 588            >>> group.check_angle_tol(1, n_round=2) == {1.57}
 589            True
 590        """
 591        if n <= 0:
 592            raise ValueError("n must be a positive integer.")
 593        if n_round is None:
 594            n_round = runtime_defaults["n_round"]
 595        if n_round < 0:
 596            raise ValueError("n_round must be a nonnegative integer.")
 597
 598        angles = _segment_angles(self.all_segments)
 599        pair_count = len(angles) * (len(angles) - 1) // 2
 600        angle_differences = _closest_angle_differences(angles, pair_count)
 601        rounded_differences = set()
 602        for angle_difference in angle_differences:
 603            rounded_difference = round(angle_difference, n_round)
 604            if rounded_difference > 0:
 605                rounded_differences.add(rounded_difference)
 606
 607        return set(sorted(rounded_differences)[:n])

1149    def _update(
1150        self,
1151        xform_matrix: NDArray,
1152        reps: int = 0,
1153        take: slice | None = None,
1154        incr: float
1155        | tuple[float, float]
1156        | tuple[callable, Any]
1157        | tuple[InPlace, Any]
1158        | None = None,
1159        dyn_ref: Callable | None = None,
1160        merge: bool = False,
1161        xform_type: TransformationType = None,
1162    ) -> Self:
1163        """Updates the group with the given transformation matrix.
1164        If reps is 0, the transformation is applied to all elements.
1165        If reps is greater than 0, the transformation creates
1166        new elements with the transformed xform_matrix.
1167
1168        Args:
1169            xform_matrix (ndarray): The transformation matrix.
1170            reps (int, optional): The number of repetitions. Defaults to 0.
1171            dyn_ref: Matrix factory built by the transform method when
1172                dynamic references are in use. Defaults to None.
1173            merge(bool, optional): If True, shapes are merged.
1174        """
1175        if take is None:
1176            elements = self.elements[:]
1177        else:
1178            elements = self.elements[take]
1179        if reps == 0:
1180            for element in elements:
1181                element._update(xform_matrix, reps=0)
1182                if self.modifiers:
1183                    for modifier in self.modifiers:
1184                        modifier.apply(element)
1185        else:
1186            if dyn_ref:
1187                # self grows in place, so it is the accumulated pattern.
1188                targets = _Targets(Group(self.elements[:]), self)
1189                targets.active = Group()
1190            else:
1191                targets = None
1192            new = []
1193            for i in range(reps):
1194                if targets is not None:
1195                    targets.active.elements = elements
1196                xform_matrix = _next_xform_matrix(
1197                    xform_matrix, xform_type, incr, dyn_ref, targets, i
1198                )
1199                for element in elements:
1200                    new_element = element.copy()
1201                    new_element._update(xform_matrix)
1202                    self.elements.append(new_element)
1203                    new.append(new_element)
1204                    if self.modifiers:
1205                        for modifier in self.modifiers:
1206                            modifier.apply(new_element)
1207                elements = new[:]
1208                new = []
1209        if merge and reps > 0:
1210            merged = combine_shapes(self)
1211            _copy_style_from_members(self, merged)
1212            self[:] = merged.elements[:]
1213        # check if the bounding-boxes of the elements have changes
1214
1215        return self

 587            ...     ]
 588            ... )
 589            >>> group.check_angle_tol(1, n_round=2) == {1.57}
 590            True
 591        """
 592        if n <= 0:
 593            raise ValueError("n must be a positive integer.")
 594        if n_round is None:
 595            n_round = runtime_defaults["n_round"]
 596        if n_round < 0:
 597            raise ValueError("n_round must be a nonnegative integer.")
 598
 599        angles = _segment_angles(self.all_segments)
 600        pair_count = len(angles) * (len(angles) - 1) // 2
 601        angle_differences = _closest_angle_differences(angles, pair_count)
 602        rounded_differences = set()
 603        for angle_difference in angle_differences:
 604            rounded_difference = round(angle_difference, n_round)
 605            if rounded_difference > 0:
 606                rounded_differences.add(rounded_difference)
 607
 608        return set(sorted(rounded_differences)[:n])

1150    def _update(
1151        self,
1152        xform_matrix: NDArray,
1153        reps: int = 0,
1154        take: slice | None = None,
1155        incr: float
1156        | tuple[float, float]
1157        | tuple[callable, Any]
1158        | tuple[InPlace, Any]
1159        | None = None,
1160        dyn_ref: Callable | None = None,
1161        merge: bool = False,
1162        xform_type: TransformationType = None,
1163    ) -> Self:
1164        """Updates the group with the given transformation matrix.
1165        If reps is 0, the transformation is applied to all elements.
1166        If reps is greater than 0, the transformation creates
1167        new elements with the transformed xform_matrix.
1168
1169        Args:
1170            xform_matrix (ndarray): The transformation matrix.
1171            reps (int, optional): The number of repetitions. Defaults to 0.
1172            dyn_ref: Matrix factory built by the transform method when
1173                dynamic references are in use. Defaults to None.
1174            merge(bool, optional): If True, shapes are merged.
1175        """
1176        if take is None:
1177            elements = self.elements[:]
1178        else:
1179            elements = self.elements[take]
1180        if reps == 0:
1181            for element in elements:
1182                element._update(xform_matrix, reps=0)
1183                if self.modifiers:
1184                    for modifier in self.modifiers:
1185                        modifier.apply(element)
1186        else:
1187            if dyn_ref:
1188                # self grows in place, so it is the accumulated pattern.
1189                targets = _Targets(Group(self.elements[:]), self)
1190                targets.active = Group()
1191            else:
1192                targets = None
1193            new = []
1194            for i in range(reps):
1195                if targets is not None:
1196                    targets.active.elements = elements
1197                xform_matrix = _next_xform_matrix(
1198                    xform_matrix, xform_type, incr, dyn_ref, targets, i
1199                )
1200                for element in elements:
1201                    new_element = element.copy()
1202                    new_element._update(xform_matrix)
1203                    self.elements.append(new_element)
1204                    new.append(new_element)
1205                    if self.modifiers:
1206                        for modifier in self.modifiers:
1207                            modifier.apply(new_element)
1208                elements = new[:]
1209                new = []
1210        if merge and reps > 0:
1211            merged = combine_shapes(self)
1212            _copy_style_from_members(self, merged)
1213            self[:] = merged.elements[:]
1214        # check if the bounding-boxes of the elements have changes
1215
1216        return self
