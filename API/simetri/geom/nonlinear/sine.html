 14class SineWave(Shape):
 15    """A sampled sine wave as a ``Shape`` polyline.
 16
 17    Attributes:
 18        period: Period of the sine wave.
 19        amplitude: Amplitude of the sine wave.
 20        duration: Horizontal length of the sampled wave.
 21        n_points: Points sampled per period.
 22        phase: Phase angle in radians.
 23        damping: Exponential damping coefficient (typical range 0.001–0.005).
 24        rot_angle: Rotation angle stored with the instance.
 25
 26    Examples:
 27
 28        >>> import simetri.graphics as sg
 29        >>> wave = sg.SineWave(period=40, amplitude=20, duration=80, n_points=4)
 30        >>> [[round(float(c), 6) or 0.0 for c in q[:2]] for q in wave.vertices]

 32    """

 34    def __init__(
 35        self,
 36        period: float = 40,
 37        amplitude: float = 20,
 38        duration: float = 40,
 39        n_points: int = 100,
 40        phase_angle: float = 0,
 41        damping: float = 0,
 42        rot_angle: float = 0,
 43        xform_matrix: NDArray | None = None,
 44        **kwargs: object,
 45    ) -> None:
 46        """Create a sine-wave shape from sampled points.
 47
 48        Args:
 49            period: Period of the sine wave. Defaults to 40.
 50            amplitude: Amplitude of the sine wave. Defaults to 20.
 51            duration: Duration (x-span) of the sine wave. Defaults to 40.
 52            n_points: Sampling rate per period. Defaults to 100.
 53            phase_angle: Phase angle in radians. Defaults to 0.
 54            damping: Damping coefficient; 0.001–0.005 is typical. Defaults to 0.
 55            rot_angle: Rotation angle stored on the instance. Defaults to 0.
 56            xform_matrix: Optional transformation matrix. Defaults to None.
 57            **kwargs: Additional keyword arguments passed to ``Shape``.
 58        """
 59        phase = phase_angle
 60        freq = 1 / period
 61        n_cycles = duration / period
 62        x = np.linspace(0, duration, int(n_points * n_cycles))
 63        y = amplitude * np.sin(2 * np.pi * freq * x + phase)
 64        if damping:
 65            y *= np.exp(-damping * x)
 66        vertices = np.column_stack((x, y)).tolist()
 67        super().__init__(vertices, xform_matrix=xform_matrix, **kwargs)
 68        self.subtype = Types.SINE_WAVE
 69        self.period = period
 70        self.amplitude = amplitude
 71        self.duration = duration
 72        self.n_points = n_points
 73        self.phase = phase
 74        self.damping = damping
 75        self.rot_angle = rot_angle

123def sine_wave(
124    amplitude: float,
125    frequency: float,
126    duration: float,
127    sample_rate: float,
128    phase: float = 0,
129) -> tuple[NDArray, NDArray]:
130    """
131    Generate a sine wave.
132
133    Args:
134        amplitude (float): Amplitude of the wave.
135        frequency (float): Frequency of the wave.
136        duration (float): Duration of the wave.
137        sample_rate (float): Sample rate.
138        phase (float, optional): Phase angle of the wave. Defaults to 0.
139
140    Returns:
141        np.ndarray: Time and signal arrays representing the sine wave.
142
143    Examples:
144        >>> import simetri.graphics as sg
145        >>> time, signal = sg.sine_wave(1.0, 1.0, 1.0, 4.0)
146        >>> [round(float(x), 6) for x in time]

148        >>> [round(float(x), 6) for x in signal]
149        [0.0, 1.0, 0.0, -1.0]
150    """
151    time = np.linspace(0, duration, int(sample_rate * duration), endpoint=False)
152    signal = amplitude * np.sin(2 * np.pi * frequency * time + phase)
153    # plt.plot(time, signal)

155    # plt.ylabel('Amplitude')
156    # plt.title('Discretized Sine Wave')
157    # plt.grid(True)
158    # plt.show()
159    return time, signal

192    duration: float = 40,

194    phase_angle: float = 0,
195    damping: float = 0,

197    """
198    Generate sine wave points.
199
200    Args:
201        period: Period of the wave. Defaults to 40.
202        amplitude: Amplitude of the wave. Defaults to 20.
203        duration: Duration of the wave. Defaults to 40.
204        n_points: Samples per period. Defaults to 100.
205        phase_angle: Phase angle in radians. Defaults to 0.
206        damping: Damping coefficient. Defaults to 0.
207
208    Returns:
209        list[list[float]]: ``(x, y)`` samples of the sine wave.
210
211    Examples:
212        >>> import simetri.graphics as sg
213        >>> pts = sg.sine_points(period=10, amplitude=5, duration=10, n_points=4)
214        >>> [[round(float(c), 6) or 0.0 for c in q[:2]] for q in pts]
215        [[0.0, 0.0], [3.333333, 4.330127], [6.666667, -4.330127], [10.0, 0.0]]
216    """
217    phase = phase_angle
218    freq = 1 / period
219    n_cycles = duration / period
220    x = np.linspace(0, duration, int(n_points * n_cycles))
221    y = amplitude * np.sin(2 * np.pi * freq * x + phase)
222    if damping:
223        y *= np.exp(-damping * x)
224    vertices = np.column_stack((x, y)).tolist()
225
226    return vertices
