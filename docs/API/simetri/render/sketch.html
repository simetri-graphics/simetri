simetri.render.sketch

Neutral sketch snapshots used by SVG/TikZ renderers.

Sketches capture drawable state at draw time and are not meant to be edited. Each sketch dataclass has type == Types.SKETCH and a specific subtype.

Note:

Prefer creating sketches through canvas draw APIs rather than constructing them by hand unless writing a backend.

Examples:
>>> sk = CircleSketch(center=(0, 0), radius=40)
>>> sk.subtype.name

  53class CircleSketch:
  54    """Immutable circle snapshot for rendering backends.
  55
  56    Attributes:
  57        center: Circle center after applying ``xform_matrix``.
  58        radius: Circle radius.
  59        xform_matrix: Affine transform at draw time.
  60        type: Always ``Types.SKETCH``.
  61        subtype: Always ``Types.CIRCLE_SKETCH``.
  62    Examples:
  63        >>> import simetri.graphics as sg
  64        >>> sk = CircleSketch(center=(0, 0), radius=40)
  65        >>> sk.subtype.name
  66        'CIRCLE_SKETCH'
  67

  69
  70    center: tuple
  71    radius: float
  72    xform_matrix: NDArray = None
  73
  74    def __post_init__(self) -> None:
  75        """Bake transform into ``center`` and set type metadata."""
  76        self.type = Types.SKETCH
  77        self.subtype = Types.CIRCLE_SKETCH
  78        self.id = get_unique_id(self)
  79        if self.xform_matrix is None:
  80            self.xform_matrix = identity_matrix()
  81            center = self.center
  82        else:
  83            center = homogenize([self.center])
  84            center = (center @ self.xform_matrix).tolist()[0][:2]
  85            _, _, scale = decompose_transformations(self.xform_matrix)
  86            scale_x = float(scale[0])
  87            if scale_x != 1.0:
  88                self.radius = float(self.radius) * scale_x
  89        self.center = center
  90        self.closed = True

  93        """Return a CircleSketch string from center and radius.

 285    @staticmethod
 286    def _line_limits(
 287        canvas: Canvas | None,
 288    ) -> tuple[float, float, float, float] | None:
 289        if canvas is None:
 290            return None
 291        limits = None
 292        if canvas.limits is not None:
 293            limits = tuple(canvas.limits)
 294        elif canvas._all_vertices:
 295            bbox = bounding_box(canvas._all_vertices)
 296            limits = (*bbox.southwest, *bbox.northeast)
 297
 298        page = getattr(canvas, "active_page", None)
 299        sketches = getattr(page, "sketches", []) if page is not None else []
 300        for sketch in sketches:
 301            if (
 302                getattr(sketch, "subtype", None) == Types.HELPLINES_SKETCH
 303                and hasattr(sketch, "pos")
 304                and hasattr(sketch, "width")
 305                and hasattr(sketch, "height")
 306            ):
 307                x, y = sketch.pos[:2]
 308                candidate = (x, y, x + sketch.width, y + sketch.height)
 309                if limits is None:
 310                    limits = candidate
 311                else:
 312                    limits = (
 313                        min(limits[0], candidate[0]),
 314                        min(limits[1], candidate[1]),
 315                        max(limits[2], candidate[2]),
 316                        max(limits[3], candidate[3]),
 317                    )
 318
 319        return limits

 321    @staticmethod
 322    def _clip_line_to_rect(
 323        start: PointType,
 324        end: PointType,
 325        rect: tuple[float, float, float, float] | None,
 326        draw_type: Extent,
 327    ) -> tuple[PointType, PointType]:
 328        if rect is None:
 329            return start, end
 330
 331        x1, y1 = start[:2]
 332        x2, y2 = end[:2]
 333        dx = x2 - x1
 334        dy = y2 - y1
 335        if abs(dx) < 1e-12 and abs(dy) < 1e-12:
 336            return start, end
 337
 338        xmin, ymin, xmax, ymax = rect
 339        t_min = float("-inf")
 340        t_max = float("inf")
 341
 342        if abs(dx) < 1e-12:
 343            if x1 < xmin or x1 > xmax:
 344                return start, end
 345        else:
 346            tx1 = (xmin - x1) / dx
 347            tx2 = (xmax - x1) / dx
 348            t_min = max(t_min, min(tx1, tx2))
 349            t_max = min(t_max, max(tx1, tx2))
 350
 351        if abs(dy) < 1e-12:
 352            if y1 < ymin or y1 > ymax:
 353                return start, end
 354        else:
 355            ty1 = (ymin - y1) / dy
 356            ty2 = (ymax - y1) / dy
 357            t_min = max(t_min, min(ty1, ty2))
 358            t_max = min(t_max, max(ty1, ty2))
 359
 360        if t_min > t_max:
 361            return start, end
 362
 363        if draw_type == Extent.INFINITE:
 364            t0, t1 = t_min, t_max
 365        elif draw_type == Extent.RAY:
 366            t0, t1 = max(0.0, t_min), t_max
 367            if t0 > t1:
 368                return start, end
 369        else:
 370            return start, end
 371
 372        p0 = (x1 + t0 * dx, y1 + t0 * dy)
 373        p1 = (x1 + t1 * dx, y1 + t1 * dy)
 374        return p0, p1
 375
 376    def populate(self, canvas: Canvas) -> None:
 377        """Populate rendered vertices for deferred draw types (RAY/INFINITE).
 378
 379        Examples:
 380            >>> import simetri.graphics as sg
 381            >>> canvas = sg.Canvas()
 382            >>> sk = LineSketch(vertices=[(0, 0), (100, 0)])  # doctest: +SKIP
 383            >>> sk.populate(canvas)  # doctest: +SKIP
 384            >>> len(sk.vertices)  # doctest: +SKIP
 385            2
 386        """
 387        extent = getattr(
 388            self, "extent", getattr(self, "draw_type", Extent.SEGMENT)
 389        )
 390        if not isinstance(extent, Extent) and extent is not None:
 391            extent = Extent(extent)
 392
 393        self.vertices = self._raw_vertices[:]
 394        if extent not in (Extent.RAY, Extent.INFINITE):
 395            return
 396
 397        limits = self._line_limits(canvas)
 398        start, end = self._clip_line_to_rect(
 399            self._raw_vertices[0], self._raw_vertices[1], limits, extent
 400        )
 401        self.vertices = [start, end]

1267    """Return grid coordinates from ``low`` through ``high`` at ``step``."""
1268    value = float(np.floor(low / step) * step)

325        end: PointType,
326        rect: tuple[float, float, float, float] | None,

328    ) -> tuple[PointType, PointType]:

332        x1, y1 = start[:2]
333        x2, y2 = end[:2]
334        dx = x2 - x1
335        dy = y2 - y1
