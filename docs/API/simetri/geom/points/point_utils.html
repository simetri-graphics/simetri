 109def offset_point_on_line(
 110    point: PointType, line: LineType, offset: float
 111) -> PointType:
 112    """Return a point on a line that is offset from the given point.
 113
 114    Args:
 115        point (PointType): Input point.
 116        line (LineType): Input line.
 117        offset (float): Offset distance.
 118
 119    Returns:
 120        PointType: Offset point on the line.
 121
 122    Examples:
 123        >>> import simetri.graphics as sg
 124        >>> sg.offset_point_on_line((0, 0), [(0, 0), (40, 0)], 2)
 125        (2.0, 0.0)
 126"""
 127    x, y = point[:2]
 128    x1, y1 = line[0][:2]
 129    x2, y2 = line[1][:2]
 130    dx = x2 - x1
 131    dy = y2 - y1
 132    # normalize the vector
 133    mag = (dx * dx + dy * dy) ** 0.5
 134    dx = dx / mag
 135    dy = dy / mag
 136    return x + dx * offset, y + dy * offset

 139def perp_offset_point(
 140    point: PointType, line: LineType, offset: float
 141) -> PointType:
 142    """Return a point that is offset from the given point in the perpendicular direction to the given line.
 143

 145        point (PointType): Input point.

 164def fix_degen_points(
 165    points: list[PointType],
 166    loop: bool = False,
 167    closed: bool = False,
 168    rel_tol: float | None = None,
 169    abs_tol: float | None = None,
 170    check_collinear: bool = True,
 171) -> list[PointType]:
 172    """Return points with duplicates and collinear middles removed.
 173
 174    Args:
 175        points: Point list (mutated in place).
 176        loop: Whether to treat the list as a loop. Defaults to False.
 177        closed: Whether the polyline is closed. Defaults to False.
 178        rel_tol (float, optional): Relative tolerance. Defaults to None.
 179        abs_tol (float, optional): Absolute tolerance. Defaults to None.
 180        check_collinear (bool, optional): Whether to check for collinear points. Defaults to True.
 181
 182    Returns:
 183        list[PointType]: List of points with duplicate and collinear points removed.
 184
 185    Examples:
 186        >>> import simetri.graphics as sg
 187        >>> sg.fix_degen_points(
 188        ...     [(0, 0), (0, 0), (40, 0), (80, 0)],
 189        ...     check_collinear=False,
 190        ... )
 191        [(0, 0), (40, 0), (80, 0)]
 192"""
 193    rel_tol, abs_tol = resolve_tol(rel_tol, abs_tol)
 194    abs_tol2 = abs_tol * abs_tol
 195    new_points = []
 196    for i, point in enumerate(points):
 197        if i == 0:
 198            new_points.append(point)
 199        else:
 200            if not close_points_square(point, new_points[-1], dist2=abs_tol2):
 201                new_points.append(point)
 202    if loop and close_points_square(
 203        new_points[0], new_points[-1], dist2=abs_tol2
 204    ):
 205        new_points.pop(-1)
 206
 207    if check_collinear:
 208        # Check for collinear points and remove the middle one.
 209        from simetri.geom.segments.line_utils import (
 210            merge_consecutive_collinear_edges,
 211        )
 212
 213        new_points = merge_consecutive_collinear_edges(
 214            new_points,
 215            closed,
 216            rel_tol,
 217            abs_tol,
 218        )
 219
 220    return new_points

 286def between3(a: PointType, b: PointType, c: PointType) -> bool:
 287    """Return True if ``c`` lies on segment ``ab`` (collinear and in range).
 288
 289    Args:
 290        a: Segment start.
 291        b: Segment end.
 292        c: Query point.
 293
 294    Returns:
 295        True if ``c`` is between ``a`` and ``b``.
 296
 297    Examples:
 298        >>> import simetri.graphics as sg
 299        >>> sg.between3((0, 0), (80, 0), (40, 0))
 300        True
 301        >>> sg.between3((0, 0), (40, 0), (80, 0))
 302        False
 303"""
 304    from simetri.geom.segments.line_utils import collinear3
 305
 306    if not collinear3(a, b, c):
 307        res = False
 308    elif a[0] != b[0]:
 309        res = ((a[0] <= c[0]) and (c[0] <= b[0])) or (
 310            (a[0] >= c[0]) and (c[0] >= b[0])
 311        )
 312    else:
 313        res = ((a[1] <= c[1]) and (c[1] <= b[1])) or (
 314            (a[1] >= c[1]) and (c[1] >= b[1])
 315        )
 316    return res

 319def check_consecutive_duplicates(
 320    points: Sequence[PointType] | NDArray[np.floating],
 321    rel_tol: float = 0,
 322    abs_tol: float | None = None,
 323) -> bool:
 324    """Return True if any consecutive vertices match within tolerance.
 325
 326    Args:
 327        points: Points to check.
 328        rel_tol: Relative tolerance. Defaults to 0.
 329        abs_tol: Absolute tolerance. Defaults to ``runtime_defaults['abs_tol']``.
 330
 331    Returns:
 332        bool: True if consecutive duplicate points are found, False otherwise.
 333
 334    Examples:
 335        >>> import simetri.graphics as sg
 336        >>> sg.check_consecutive_duplicates([(0, 0), (0, 0), (40, 0)], abs_tol=0.001)
 337        True
 338        >>> sg.check_consecutive_duplicates([(0, 0), (40, 0)], abs_tol=0.001)
 339        False
 340"""
 341    if abs_tol is None:
 342        abs_tol = runtime_defaults["abs_tol"]
 343    if isinstance(points, np.ndarray):
 344        points = points.tolist()
 345    if points and len(points) > 1:
 346        for i, pnt in enumerate(points[:-1]):
 347            next_pnt = points[i + 1]
 348            val1 = pnt[0] + pnt[1]
 349            val2 = next_pnt[0] + next_pnt[1]
 350            if isclose(
 351                val1, val2, rel_tol=rel_tol, abs_tol=abs_tol
 352            ) and np.allclose(pnt, next_pnt, rtol=0, atol=abs_tol):
 353                return True
 354
 355    return False

 358def left3(a: PointType, b: PointType, c: PointType) -> bool:
 359    """
 360    Check if point c is left of line ab.
 361    Args:
 362        a (PointType): The first point defining the line.
 363        b (PointType): The second point defining the line.
 364        c (PointType): The point to test.
 365    Returns:
 366        bool: True if point c is left of line ab, False otherwise.
 367
 368    Examples:
 369        >>> import simetri.graphics as sg
 370        >>> sg.left3((0, 0), (40, 0), (0, 40))
 371        True
 372        >>> sg.left3((0, 0), (40, 0), (0, -40))
 373        False
 374"""
 375
 376    ax, ay = a[:2]
 377    bx, by = b[:2]
 378    cx, cy = c[:2]
 379    return (bx - ax) * (cy - ay) - (cx - ax) * (by - ay) > 0

 383    points: list[PointType],
 384    rel_tol: float | None = None,
 385    abs_tol: float | None = None,
 386) -> list[PointType]:
 387    """

 415def remove_collinear_points(
 416    points: list[PointType],
 417    rel_tol: float | None = None,
 418    abs_tol: float | None = None,
 419) -> list[PointType]:
 420    """
 421    Return a list of points with collinear points removed.
 422
 423    Args:
 424        points (list[PointType]): List of points.
 425        rel_tol (float, optional): Relative tolerance. Defaults to None.
 426        abs_tol (float, optional): Absolute tolerance. Defaults to None.
 427
 428    Returns:
 429        list[PointType]: List of points with collinear points removed.
 430
 431    Examples:
 432        >>> import simetri.graphics as sg
 433        >>> sg.remove_collinear_points([(0, 0), (40, 0), (80, 0)], abs_tol=0.001)
 434        [(0, 0), (80, 0)]
 435"""
 436    from simetri.geom.segments.line_utils import collinear3
 437
 438    rel_tol, abs_tol = resolve_tol(rel_tol, abs_tol)
 439    n = len(points)
 440    closed = n > 2 and points[0] == points[-1]
 441    new_points = []
 442    for i, point in enumerate(points):
 443        if i == 0:
 444            new_points.append(point)
 445            continue
 446        if not closed and i == n - 1:
 447            new_points.append(point)
 448            continue
 449        next_index = (i + 1) % n if closed else i + 1
 450        if not collinear3(
 451            new_points[-1],
 452            point,
 453            points[next_index],
 454            rel_tol=rel_tol,
 455            abs_tol=abs_tol,
 456        ):
 457            new_points.append(point)
 458    return new_points

 461def clockwise3(p: PointType, q: PointType, r: PointType) -> bool:
 462    """Return 1 if the points p, q, and r are in clockwise order,
 463    return -1 if the points are in counter-clockwise order,
 464    return 0 if the points are collinear
 465
 466    Args:
 467        p (PointType): First point.
 468        q (PointType): Second point.
 469        r (PointType): Third point.
 470
 471    Returns:
 472        int: 1 if the points are in clockwise order, -1 if counter-clockwise, 0 if collinear.
 473
 474    Examples:
 475        >>> import simetri.graphics as sg
 476        >>> sg.clockwise3((0, 0), (40, 0), (0, 40))
 477        -1
 478        >>> sg.clockwise3((0, 0), (0, 40), (40, 0))
 479        1
 480"""
 481    px, py = p[:2]
 482    qx, qy = q[:2]
 483    rx, ry = r[:2]
 484    area_ = (qx - px) * (ry - py) - (rx - px) * (qy - py)
 485    if area_ > 0:
 486        res = -1
 487    elif area_ < 0:
 488        res = 1
 489    else:
 490        res = 0
 491
 492    return res

 495def on_segment(
 496    a: PointType, b: PointType, p: PointType, eps: float = 1e-12
 497) -> bool:
 498    """Return True if point ``p`` lies on segment ``ab`` within ``eps``.
 499
 500    Args:
 501        a: Segment start point.
 502        b: Segment end point.
 503        p: Query point.
 504        eps: Numeric tolerance. Defaults to ``1e-12``.
 505
 506    Returns:
 507        bool: True if ``p`` is collinear with ``ab`` and inside its bbox.
 508
 509    Examples:
 510        >>> import simetri.graphics as sg
 511        >>> sg.on_segment((0, 0), (80, 0), (40, 0))
 512        True
 513        >>> sg.on_segment((0, 0), (40, 0), (80, 0))
 514        False
 515"""
 516
 517    # check collinear + within bbox
 518    def cross(ax: float, ay: float, bx: float, by: float) -> float:
 519        return ax * by - ay * bx
 520
 521    def orient3(a: PointType, b: PointType, c: PointType) -> float:
 522        # cross((b-a),(c-a))
 523        return cross(b[0] - a[0], b[1] - a[1], c[0] - a[0], c[1] - a[1])
 524
 525    if abs(orient3(a, b, p)) > eps:
 526        return False
 527    return (
 528        min(a[0], b[0]) - eps <= p[0] <= max(a[0], b[0]) + eps
 529        and min(a[1], b[1]) - eps <= p[1] <= max(a[1], b[1]) + eps
 530    )

 638def point_to_line_distance(point: PointType, line: LineType) -> float:
 639    """Return the distance between a line and a point.
 640
 641    Args:
 642        point (PointType): Input point.
 643        line (LineType): Input line.
 644
 645    Returns:
 646        float: Distance from the point to the line.
 647
 648    Examples:
 649        >>> import simetri.graphics as sg
 650        >>> sg.point_to_line_distance((0, 40), [(0, 0), (40, 0)])
 651        40.0
 652"""
 653    x0, y0 = point
 654    x1, y1 = line[0][:2]
 655    x2, y2 = line[1][:2]
 656    dx = x2 - x1
 657    dy = y2 - y1
 658    return abs(dx * (y1 - y0) - (x1 - x0) * dy) / sqrt(dx**2 + dy**2)
