simetri.extensions.easing

Robert Penner easing functions for animation and interpolation.

Adapted from https://github.com/semitable/easing-functions. Each ease class maps a progress value in [0, 1] (or a custom duration) to an eased value between start and end.

Examples:
>>> e = QuadEaseInOut(start=0, end=100)
>>> round(e(0.5), 2)
50.0

 22class EasingBase:
 23    """Base class for Penner-style easing functions.
 24
 25    Attributes:
 26        limit: Normalized input range used by ``ease``, typically ``(0, 1)``.
 27        start: Output value at progress 0.
 28        end: Output value at progress 1.
 29        duration: Progress scale; ``alpha`` is divided by this before easing.
 30
 31    Examples:
 32        >>> round(LinearInOut(start=0, end=100)(0.5), 2)
 33        50.0
 34    """
 35
 36    limit = (0, 1)
 37
 38    def __init__(self, start: float = 0, end: float = 1, duration: float = 1) -> None:
 39        """Initialize the easing range.
 40
 41        Args:
 42            start: Output value when progress is 0.
 43            end: Output value when progress is 1.
 44            duration: Divisor applied to normalized progress before ``func``.
 45
 46        Examples:
 47            >>> e = LinearInOut(start=10, end=20)
 48            >>> round(e.ease(0.5), 2)
 49            15.0
 50        """
 51        self.start = start
 52        self.end = end
 53        self.duration = duration
 54
 55    def func(self, t: float) -> float:
 56        """Map normalized time ``t`` in roughly ``[0, 1]`` to eased unit progress.
 57
 58        Args:
 59            t: Normalized time.
 60
 61        Returns:
 62            Eased unit value, typically in ``[0, 1]``.
 63
 64        Raises:
 65            NotImplementedError: Subclasses must override this method.
 66
 67        Examples:
 68            >>> EasingBase.func(EasingBase(), 0.5)  # doctest: +IGNORE_EXCEPTION_DETAIL
 69            Traceback (most recent call last):
 70            NotImplementedError
 71        """
 72        raise NotImplementedError
 73
 74    def ease(self, alpha: float) -> float:
 75        """Ease progress ``alpha`` into the configured ``start``/``end`` range.
 76
 77        Args:
 78            alpha: Progress value (often in ``[0, 1]``).
 79
 80        Returns:
 81            Interpolated value between ``start`` and ``end``.
 82
 83        Examples:
 84            >>> round(LinearInOut(start=0, end=10).ease(0.5), 2)
 85            5.0
 86        """
 87        t = self.limit[0] * (1 - alpha) + self.limit[1] * alpha
 88        t /= self.duration
 89        a = self.func(t)
 90        return self.end * a + self.start * (1 - a)
 91
 92    def __call__(self, alpha: float) -> float:
 93        """Call ``ease`` so the instance is usable as a function.
 94
 95        Args:
 96            alpha: Progress value.
 97
 98        Returns:
 99            Eased value between ``start`` and ``end``.
100
101        Examples:
102            >>> round(LinearInOut(start=0, end=10)(0.5), 2)
103            5.0
104        """
105        return self.ease(alpha)
