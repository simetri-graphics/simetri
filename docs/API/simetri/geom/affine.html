simetri.geom.affine

3×3 affine transformation matrices and point-transform helpers.

Matrices are stored in row-major form suitable for points @ matrix with homogeneous coordinates. Public aliases on simetri.graphics include TM, RM, MirM, GM, SM, and SHM.

Examples

import simetri.graphics as sg
M = sg.translation_matrix(40, 20)
points = sg.homogenize([[0, 0], [40, 40]])
print(points @ M)
# [[40. 20.  1.]
# [80. 60.  1.]]

112def rot_about_origin_matrix(angle: float) -> NDArray:
113    """
114    Return a rotation matrix in row-major form
115    [[cos(angle), sin(angle), 0], [-sin(angle), cos(angle), 0], [0, 0, 1.0]].

117    Args:
118        angle (float): The rotation angle in radians.

127        >>> points = sg.homogenize([[40, 0], [0, 40]])
128        >>> print(np.round(points @ M, 10))

133    s = sin(angle)
134    return np.array([[c, s, 0], [-s, c, 0], [0, 0, 1.0]])

137def rotation_matrix(angle: float, about: PointType = (0, 0)) -> NDArray:
138    """Return a rotation matrix about a point.
139
140    Composes translate-to-origin, rotate by ``angle`` (radians), then
141    translate back. row-major form for ``points @ matrix``.
142
143    Args:
144        angle: Rotation angle in radians (counterclockwise).
145        about: Point to rotate about. Defaults to ``(0, 0)``.
146
147    Returns:
148        np.ndarray: The rotation matrix.
149
150    Examples:
151        >>> import simetri.graphics as sg
152        >>> import numpy as np
153        >>> M = sg.rotation_matrix(sg.pi / 2, about=(40, 40))
154        >>> points = sg.homogenize([[80, 40], [40, 80]])
155        >>> print(np.round(points @ M, 10))
156        [[40. 80.  1.]
157        [ 0. 40.  1.]]
158"""
159    dx, dy = about[:2]
160    # translate 'about' to the origin
161    trans_mat = translation_matrix(-dx, -dy)
162    # rotate around the origin
163    rot_mat = rot_about_origin_matrix(angle)
164    # translate it back to initial pos
165    inv_trans_mat = translation_matrix(dx, dy)
166    # compose the transformation matrix
167    return trans_mat @ rot_mat @ inv_trans_mat

170def inv_rotation_matrix(angle: float, about: PointType = (0, 0)) -> NDArray:
171    """
172    Construct the inverse of a rotation matrix that can be used to rotate a point
173    about another point by angle float.
174    Return a rotation matrix in row-major form
175    dx, dy = about
176    [[cos(angle), -sin(angle), 0],
177    [sin(angle), cos(angle), 0],
178    -cos(angle)dx-sin(angle)dy+x, -sin(angle)dx+cos(angle)dy+y, 1]].
179
180    Args:
181        angle (float): The rotation angle in radians.
182        about (tuple, optional): The point to rotate about, defaults to (0, 0).
183
184    Returns:
185        np.ndarray: The inverse rotation matrix.
186
187    Examples:
188        >>> import simetri.graphics as sg
189        >>> import numpy as np
190        >>> M = sg.inv_rotation_matrix(sg.pi / 2)
191        >>> points = sg.homogenize([[0, 40], [-40, 0]])
192        >>> print(np.round(points @ M, 10))
193        [[40.  0.  1.]
194        [-0. 40.  1.]]
195"""
196    dx, dy = about[:2]
197    # translate 'about' to the origin
198    trans_mat = translation_matrix(-dx, -dy)
199    # rotate around the origin
200    rot_mat = rot_about_origin_matrix(angle)
201    # translate it back to initial pos
202    inv_trans_mat = translation_matrix(dx, dy)
203    # compose the transformation matrix
204    return trans_mat @ rot_mat.T @ inv_trans_mat

207def glide_matrix(mirror_line: LineType, distance: float) -> NDArray:
208    """
209    Return a glide-reflection matrix in row-major form.
210    Reflect about the given vector then translate by dx
211    along the same vector.
212
213    Args:
214        mirror_line (LineType): The line to mirror about.
215        distance (float): The distance to translate along the line.
216
217    Returns:
218        np.ndarray: The glide-reflection matrix.
219
220    Examples:
221        >>> import simetri.graphics as sg
222        >>> M = sg.glide_matrix([(0, 0), (40, 0)], 40)
223        >>> points = sg.homogenize([[0, 40], [80, 120]])
224        >>> print(points @ M)
225        [[  40.  -40.    1.]
226        [ 120. -120.    1.]]
227"""
228    mirror_mat = mirror_about_line_matrix(mirror_line)
229    x, y = vec_along_line(mirror_line, distance)[:2]
230    trans_mat = translation_matrix(x, y)
231
232    return mirror_mat @ trans_mat

236    """
237    Return the inverse of a glide-reflection matrix in row-major form.
238    Translate back along the given line then reflect about that line.
239
240    Args:
241        mirror_line (LineType): The line to mirror about.
242        distance (float): The distance to translate along the line.
243
244    Returns:
245        np.ndarray: The inverse glide-reflection matrix.

259    return trans_matrix @ mirror_mat

315def scale_in_place_matrix(
316    scale_x: float, scale_y: float, about: PointType = (0, 0)
317) -> NDArray:
318    """
319    Return a scale matrix in row-major form that scales about a point.
320
321    Args:
322        scale_x (float): Scale factor in x direction.
323        scale_y (float): Scale factor in y direction.
324        about (PointType): PointType about which the scaling is performed.
325
326    Returns:
327        np.ndarray: A scale matrix in row-major form that scales about a point.

368    """
369    Return the inverse of a shear matrix in row-major form.
370
371    Args:
372        angle_x (float): Angle of shear in x direction.
373        angle_y (float, optional): Angle of shear in y direction, defaults to 0.
374
375    Returns:
376        np.ndarray: The inverse of a shear matrix in row-major form.

378    Examples:

380        >>> import numpy as np
381        >>> M = sg.inv_shear_matrix(sg.pi / 4)

384        [[ 0. 40.  1.]
385        [40.  0.  1.]]

387    tan_x = tan(angle_x)
388    tan_y = tan(angle_y)
389    determinant = 1 - tan_x * tan_y
390    return np.array(
391        [
392            [1 / determinant, -tan_y / determinant, 0],
393            [-tan_x / determinant, 1 / determinant, 0],
394            [0, 0, 1.0],
395        ]
396    )

399def mirror_matrix(about: LineType | PointType) -> NDArray:
400    """
401    Return a matrix to perform reflection about a line or a point.
402
403    Args:
404        about (LineType | PointType): A line or point about which the reflection is performed.
405
406    Returns:
407        np.ndarray: A matrix to perform reflection about a line or a point.
408
409    Raises:
410        RuntimeError: If about is not a line or a point.
411
412    Examples:
413        >>> import simetri.graphics as sg
414        >>> M = sg.mirror_matrix((0, 0))
415        >>> points = sg.homogenize([[20, 40], [60, 80]])
416        >>> print(points @ M)
417        [[-20. -40.   1.]
418        [-60. -80.   1.]]
419        >>> M = sg.mirror_matrix([(0, 0), (40, 0)])
420        >>> points = sg.homogenize([[60, 20], [80, -20]])
421        >>> print(points @ M)
422        [[ 60. -20.   1.]
423        [ 80.  20.   1.]]
424"""
425    if is_line(about):
426        res = mirror_about_line_matrix(about)
427    elif is_point(about):
428        res = mirror_about_point_matrix(about)
429    else:
430        raise RuntimeError(f"{about} is invalid!")
431    return res

470def mirror_about_line_matrix(line: LineType) -> NDArray:
471    """
472    Return a matrix to perform reflection about a line.
473
474    Args:
475        line (LineType): The line about which the reflection is performed.
476
477    Returns:
478        np.ndarray: A matrix to perform reflection about a line.
479
480    Examples:
481        >>> import simetri.graphics as sg
482        >>> M = sg.mirror_about_line_matrix([(0, 0), (40, 0)])
483        >>> points = sg.homogenize([[60, 20], [80, -20]])
484        >>> mirrored_points = points @ M
485        >>> print(mirrored_points)
486        [[ 60. -20.   1.]
487        [ 80.  20.   1.]]
488"""
489    p1, p2 = line
490    x1, y1 = p1[:2]
491    theta = atan2(p2[1] - p1[1], p2[0] - p1[0])
492    two_theta = 2 * theta
493
494    # translate the line to the origin
495    # T = translation_matrix(-x1, -y1)
496    # rotate about the origin by 2*theta
497    # R = rot_about_origin_matrix(2*theta)
498    # translate back
499    # inv_t = translation_matrix(x1, y1)
500    # return T @ R @ inv_t
501
502    # We precompute the matrix
503    c2 = cos(two_theta)
504    s2 = sin(two_theta)
505    return np.array(
506        [
507            [c2, s2, 0],
508            [s2, -c2, 0],
509            [-x1 * c2 + x1 - y1 * s2, -x1 * s2 + y1 * c2 + y1, 1.0],
510        ]
511    )

737

739        point: Point to rotate.

764    inv_xform = inv_rotation @ inv_translation
765    x, y, _ = [x, y, 1] @ inv_xform

767    return (x, y)
