122def edges_to_nodes(edges: Sequence[Sequence[Any]]) -> list[Any]:
123    """Return the node sequence of a connected edge list.
124
125    Args:
126        edges: List of two-endpoint edges.
127
128    Returns:
129        list: Nodes in chain order. A closed chain repeats the start node
130        at the end.
131
132    Examples:
133        >>> from simetri.helpers.graph import edges_to_nodes
134        >>> edges_to_nodes([(1, 2), (0, 1), (2, 3)])
135        [0, 1, 2, 3]
136        >>> edges_to_nodes([(0, 1), (1, 2), (2, 0)])
137        [0, 1, 2, 0]
138    """
139    chain = longest_chain(edges)
140    closed = chain[0][0] == chain[-1][-1]
141    if closed:
142        nodes = [x[0] for x in chain[:-1]]
143    else:
144        nodes = [x[0] for x in chain] + [chain[-1][1]]
145    if closed:
146        last_edge = chain[-1]
147        if last_edge[1] == nodes[-1]:
148            nodes.extend(reversed(last_edge))
149        elif last_edge[0] == nodes[-1]:
150            nodes.extend(last_edge)
151        elif last_edge[0] == nodes[0]:
152            nodes.extend(reversed(last_edge))
153        elif last_edge[1] == nodes[0]:
154            nodes.extend(last_edge)
155
156    return nodes

159def get_cycles(edges: Sequence[Sequence[Any]]) -> list[list[Any]] | None:
160    """Return every cycle in a graph given as an edge list.
161
162    Each cycle is a node list with the first node repeated at the end.
163    Returns ``None`` when the graph has no cycle.
164
165    Args:
166        edges: Two-endpoint edges (hashable nodes).
167
168    Returns:
169        list[list] | None: Closed cycles, or ``None`` if there is none.
170
171    Examples:
172        >>> from simetri.helpers.graph import get_cycles
173        >>> get_cycles([(0, 1), (1, 2), (2, 0)])
174        [[1, 0, 2, 1]]
175        >>> get_cycles([(0, 1), (1, 2)])
176        >>> get_cycles([])
177    """
178    nx_graph = nx.Graph()
179    nx_graph.add_edges_from(edges)
180    cycles = nx.cycle_basis(nx_graph)
181    res = None
182    if cycles:
183        for cycle in cycles:
184            cycle.append(cycle[0])
185
186        res = cycles
187    return res

191def find_all_paths(graph: nx.Graph, node: Any) -> list[list[Any]]:
192    """Find all simple paths of length at least 1 starting from ``node``.
193
194    Args:
195        graph: The graph.
196        node: The starting node.

198    Returns:

220def is_open_walk2(graph: nx.Graph, island: Sequence[Any]) -> bool:
221    """Return True when ``island`` is an open walk in ``graph``.
222
223    Unlike ``is_open_walk``, a two-node island is not treated as an
224    open walk unless the degree condition also holds (two degree-1
225    nodes and the rest degree 2).
226
227    Args:
228        graph: The graph.
229        island: Nodes of one connected component.
230
231    Returns:
232        bool: True if the island is an open walk, False otherwise.

236        >>> from simetri.helpers.graph import is_open_walk2

239        True
240        >>> is_open_walk2(path, [0, 1])

242    """
243    degrees = [graph.degree(node) for node in island]
244    return set(degrees) == {1, 2} and degrees.count(1) == 2

247def longest_chain(edges: Sequence[Sequence[Any]]) -> list[tuple[Any, Any]]:
248    """Order edges into one connected chain.
249
250    Starts at ``edges[0]`` and grows from either end. Edges in the
251    result are oriented so consecutive edges share the connecting node.
252
253    Args:
254        edges: Two-endpoint edges.
255
256    Returns:
257        list[tuple]: Oriented edges in chain order, or ``[]`` if
258        ``edges`` is empty.
259
260    Examples:
261        >>> from simetri.helpers.graph import longest_chain
262        >>> longest_chain([(40, 80), (0, 40), (80, 60)])
263        [(0, 40), (40, 80), (80, 60)]
264        >>> longest_chain([(0, 40), (40, 80), (80, 0)])
265        [(0, 40), (40, 80), (80, 0)]
266        >>> longest_chain([])
267        []
268    """
269    if not edges:
270        return []
271
272    endpoint_edges = {}
273    for edge_index, edge in enumerate(edges):
274        start, end = edge
275        endpoint_edges.setdefault(start, []).append(edge_index)
276        endpoint_edges.setdefault(end, []).append(edge_index)
277
278    chain = [tuple(edges[0])]
279    processed_indices = {0}
280
281    while True:
282        extended = False
283        for edge_index in endpoint_edges[chain[-1][1]]:
284            if edge_index not in processed_indices:
285                edge = edges[edge_index]
286                if edge[0] == chain[-1][1]:
287                    chain.append(tuple(edge))
288                else:
289                    chain.append((edge[1], edge[0]))
290                processed_indices.add(edge_index)
291                extended = True
292                break
293        if extended:
294            continue
295
296        for edge_index in endpoint_edges[chain[0][0]]:
297            if edge_index not in processed_indices:
298                edge = edges[edge_index]
299                if edge[1] == chain[0][0]:
300                    chain.insert(0, tuple(edge))
301                else:
302                    chain.insert(0, (edge[1], edge[0]))
303                processed_indices.add(edge_index)
304                extended = True
305                break
306        if not extended:
307            break
308
309    return chain

334def is_open_walk(graph: nx.Graph, island: Sequence[Any]) -> bool:
335    """Return True when ``island`` is an open walk in ``graph``.
336
337    A two-node island is always True. Longer islands need exactly two
338    degree-1 nodes and the rest degree 2.
339
340    Args:
341        graph: The graph.
342        island: Nodes of one connected component.
343
344    Returns:
345        bool: True if the island is an open walk, False otherwise.
346
347    Examples:
348        >>> import networkx as nx
349        >>> from simetri.helpers.graph import is_open_walk
350        >>> path = nx.path_graph(3)
351        >>> is_open_walk(path, [0, 1, 2])
352        True
353        >>> is_open_walk(path, [0, 40])
354        True
355        >>> is_open_walk(nx.cycle_graph(3), [0, 1, 2])
356        False
357    """
358    if len(island) == 2:
359        return True
360    degrees = [graph.degree(node) for node in island]
361    return set(degrees) == {1, 2} and degrees.count(1) == 2

364def graph_summary(graph: nx.Graph) -> str:
365    """Return a summary of cycles, open walks, and degenerate nodes.
366
367    Args:
368        graph: The graph.
369
370    Returns:
371        str: One block per connected component.
372
373    Examples:
374        >>> import networkx as nx
375        >>> from simetri.helpers.graph import graph_summary
376        >>> print(graph_summary(nx.path_graph(3)))
377        Island: {0, 1, 2}
378        Open Walk: 3 nodes
379        ----------------------------------------
380        >>> print(graph_summary(nx.cycle_graph(3)))
381        Island: {0, 1, 2}
382        Cycle: 3 nodes
383        ----------------------------------------
384        >>> print(graph_summary(nx.star_graph(3)))
385        Island: {0, 1, 2, 3}
386        Degenerate: 4 nodes
387        (Node, Degree): [(0, 3)]
388        ----------------------------------------
389    """
390    lines = []
391    for island in nx.connected_components(graph):
392        if len(island) > 8:
393            island = list(island)
394            lines.append(f"Island: {island[:4]}, ... , {island[-4:]}")
395        else:
396            lines.append(f"Island: {island}")
397        if is_cycle(graph, island):
398            lines.append(f"Cycle: {len(island)} nodes")
399        elif is_open_walk(graph, island):
400            lines.append(f"Open Walk: {len(island)} nodes")
401        else:
402            degenerates = [node for node in island if graph.degree(node) > 2]
403            degrees = f"{[(node, graph.degree(node)) for node in degenerates]}"
404            lines.append(f"Degenerate: {len(island)} nodes")
405            lines.append(f"(Node, Degree): {degrees}")
406        lines.append("-" * 40)
407    return "\n".join(lines)

510def sanitize_weighted_graph_edges(
511    edges: Sequence[tuple[Any, Any, Any]],
512) -> list[tuple[Any, Any, Any]]:
513    """Drop duplicate undirected pairs, keeping the first weight.
514
515    Args:
516        edges: Weighted edges ``(u, v, weight)``.
517
518    Returns:
519        list[tuple]: Deduplicated weighted edges, sorted.
520
521    Examples:
522        >>> from simetri.helpers.graph import sanitize_weighted_graph_edges
523        >>> sanitize_weighted_graph_edges([(40, 0, 20), (0, 40, 9), (80, 60, 1)])
524        [(40, 0, 20), (80, 60, 1)]
525    """
526    clean_edges = []
527    s_seen = set()
528    for edge in edges:
529        e1, e2, _ = edge
530        frozen_edge = frozenset((e1, e2))
531        if frozen_edge in s_seen:
532            continue
533        s_seen.add(frozen_edge)
534        clean_edges.append(edge)
535    clean_edges.sort()
536    return clean_edges

539def sanitize_graph_edges(
540    edges: Sequence[tuple[Any, Any]],
541) -> list[tuple[Any, Any]]:
542    """Drop duplicate undirected pairs and order each pair ``(min, max)``.
543
544    Args:
545        edges: Two-endpoint edges.
546
547    Returns:
548        list[tuple]: Deduplicated edges with ordered endpoints, sorted.
549
550    Examples:
551        >>> from simetri.helpers.graph import sanitize_graph_edges
552        >>> sanitize_graph_edges([(40, 0), (0, 40), (80, 60), (60, 80)])
553        [(0, 40), (60, 80)]
554    """
555    s_edge_set = set()
556    for edge in edges:
557        s_edge_set.add(frozenset(edge))
558    edges = [tuple(x) for x in s_edge_set]
559    edges = [(min(x), max(x)) for x in edges]
560    edges.sort()
561    return edges

def longest_chain( edges: Sequence[Sequence[typing.Any]]) -> list[tuple[typing.Any, typing.Any]]: 
248def longest_chain(edges: Sequence[Sequence[Any]]) -> list[tuple[Any, Any]]:
249    """Order edges into one connected chain.
250
251    Starts at ``edges[0]`` and grows from either end. Edges in the
252    result are oriented so consecutive edges share the connecting node.
253
254    Args:
255        edges: Two-endpoint edges.
256
257    Returns:
258        list[tuple]: Oriented edges in chain order, or ``[]`` if
259        ``edges`` is empty.
260
261    Examples:
262        >>> from simetri.helpers.graph import longest_chain
263        >>> longest_chain([(40, 80), (0, 40), (80, 60)])
264        [(0, 40), (40, 80), (80, 60)]
265        >>> longest_chain([(0, 40), (40, 80), (80, 0)])
266        [(0, 40), (40, 80), (80, 0)]
267        >>> longest_chain([])
268        []
269    """
270    if not edges:
271        return []
272
273    endpoint_edges = {}
274    for edge_index, edge in enumerate(edges):
275        start, end = edge
276        endpoint_edges.setdefault(start, []).append(edge_index)
277        endpoint_edges.setdefault(end, []).append(edge_index)
278
279    chain = [tuple(edges[0])]
280    processed_indices = {0}
281
282    while True:
283        extended = False
284        for edge_index in endpoint_edges[chain[-1][1]]:
285            if edge_index not in processed_indices:
286                edge = edges[edge_index]
287                if edge[0] == chain[-1][1]:
288                    chain.append(tuple(edge))
289                else:
290                    chain.append((edge[1], edge[0]))
291                processed_indices.add(edge_index)
292                extended = True
293                break
294        if extended:
295            continue
296
297        for edge_index in endpoint_edges[chain[0][0]]:
298            if edge_index not in processed_indices:
299                edge = edges[edge_index]
300                if edge[1] == chain[0][0]:
301                    chain.insert(0, tuple(edge))
302                else:
303                    chain.insert(0, (edge[1], edge[0]))
304                processed_indices.add(edge_index)
305                extended = True
306                break
307        if not extended:
308            break
309
310    return chain
