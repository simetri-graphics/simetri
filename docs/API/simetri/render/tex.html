131def run_job(
132    parent_dir: str,
133    file_name: str,
134    extension: str,
135    tex_path: str,
136) -> None:
137    """Compile a TeX file and write the requested output format.
138
139    Args:
140        parent_dir: Directory containing the TeX file and outputs.
141        file_name: Base name without extension.
142        extension: Desired output extension (e.g. ``.pdf``, ``.svg``).
143        tex_path: Full path to the ``.tex`` source file.
144
145    Examples:
146        >>> from simetri.render.tex import run_job
147        >>> run_job('.', 'test', '.pdf', 'test.tex')  # doctest: +SKIP
148    """
149    output_path = os.path.join(parent_dir, file_name + extension)
150    pdf_path = os.path.join(parent_dir, file_name + ".pdf")
151    tex_settings = get_tex_compiler()
152    if tex_settings["command"] is not None:
153        run_tex_compiler(input_path=tex_path, output_path=pdf_path)
154    else:
155        compiler = _default_latex_compiler_name()
156        if shutil.which(compiler) is None:
157            _raise_latex_compiler_not_found(compiler)
158        cmd = f'{compiler} "{tex_path}" --output-directory "{parent_dir}"'
159        result = compile_tex(cmd, parent_dir, print_output=False)
160        shell_output = f"{result.stdout}\n{result.stderr}".strip()
161        if _shell_output_suggests_missing_compiler(shell_output):
162            _raise_latex_compiler_not_found(compiler)
163        if "No pages of output" in result.stdout:
164            raise RuntimeError("Failed to compile the tex file.")
165        if not os.path.exists(pdf_path):
166            if result.returncode == 127:
167                _raise_latex_compiler_not_found(compiler)
168            raise RuntimeError("Failed to compile the tex file.")
169
170    if extension in (".eps", ".ps"):
171        ps_path = os.path.join(parent_dir, file_name + extension)
172        os.chdir(parent_dir)
173        cmd = f"pdf2ps {pdf_path} {ps_path}"
174        res = subprocess.run(cmd, shell=True, check=False)
175        if res.returncode != 0:
176            raise RuntimeError("Failed to convert pdf to ps.")
177    elif extension == ".svg":
178        doc = fitz.open(pdf_path)
179        page = doc.load_page(0)
180        svg = page.get_svg_image()
181        with open(output_path, "w", encoding="utf-8") as f:
182            f.write(svg)

185def compile_tex(
186    cmd: str, parent_dir: str, print_output: bool
187) -> _CompileTexResult:
188    """Run a shell LaTeX compile command and capture output.
189
190    Args:
191        cmd: Shell command (typically ``pdflatex`` with paths quoted).
192        parent_dir: Working directory for the subprocess.
193        print_output: When True, print a short tail of stdout.
194
195    Returns:
196        _CompileTexResult: Captured stdout, stderr, and process exit code.
197
198    Examples:
199        >>> from simetri.render.tex import compile_tex
200        >>> compile_tex('echo ok', '.', False).returncode  # doctest: +SKIP
201        0
202    """
203    os.chdir(parent_dir)
204    with subprocess.Popen(
205        cmd,
206        stdin=subprocess.PIPE,
207        stdout=subprocess.PIPE,
208        stderr=subprocess.PIPE,
209        shell=True,
210        text=True,
211    ) as process:
212        stdout, stderr = process.communicate("_s\n_l\n")
213        returncode = process.returncode
214    if print_output:
215        print(stdout.split("\n")[-3:])
216    exit_code = returncode if returncode is not None else 0
217    return _CompileTexResult(stdout, stderr, exit_code)

221class Tex:
222    """Tex class for generating tex code.
223
224    Attributes:
225        begin_document (str): The beginning of the document.
226        end_document (str): The end of the document.
227        begin_tikz (str): The beginning of the TikZ environment.

236        >>> 'document' in Tex().begin_document
237        True
238    """

241    end_document: str = runtime_defaults["end_doc"]

243    end_tikz: str = runtime_defaults["end_tikz"]
244    packages: list[str] = None
245    tikz_libraries: list[str] = None
246    tikz_code: str = ""  # Generated by the canvas by using sketches

255    def tex_code(self, canvas: Canvas, aux_code: str) -> str:
256        """Generate the final TeX code.
257
258        Args:
259            canvas (Canvas): The canvas object.
260            aux_code (str): Auxiliary code to include.
261
262        Returns:
263            str: The final TeX code.
264
265        Examples:
266            >>> import simetri.graphics as sg
267            >>> from simetri.render.tex import Tex
268            >>> Tex().tex_code(sg.Canvas(), '')  # doctest: +SKIP
269        """
270        doc_code = "\n".join(
271            sketch.code
272            for page in canvas.pages
273            for sketch in page.sketches
274            if sketch.subtype == Types.TEX_SKETCH
275            and sketch.location == TexLoc.DOCUMENT
276        )
277        if canvas.back_color is None:
278            back_color = ""
279        else:
280            back_color = f"\\pagecolor{color_to_tikz(canvas.back_color)}"
281        self.begin_document = self.begin_document + back_color + "\n"
282        if canvas.overlay:
283            begin_t = self.begin_tikz
284            i = begin_t.index("]\n")
285            overlay = ", remember picture, overlay"
286            self.begin_tikz = begin_t[:i] + overlay + begin_t[i:]
287        if canvas.limits is not None or canvas.inset != 0:
288            begin_tikz = self.begin_tikz + get_limits_code(canvas) + "\n"
289        else:
290            begin_tikz = self.begin_tikz + "\n"
291        if scope_code_required(canvas):
292            scope = get_canvas_scope(canvas)
293            code = (
294                self.get_preamble(canvas)
295                + self.begin_document
296                + doc_code
297                + begin_tikz
298                + scope
299                + self.get_tikz_code()
300                + aux_code
301                + "\\end{scope}\n"
302                + self.end_tikz
303                + self.end_document
304            )
305        else:
306            code = (
307                self.get_preamble(canvas)
308                + self.begin_document
309                + doc_code
310                + begin_tikz
311                + self.get_tikz_code()
312                + aux_code
313                + self.end_tikz
314                + self.end_document
315            )
316
317        return code

319    def get_doc_class(self, border: float, font_size: int) -> str:

372    def get_packages(
373        self, canvas: Canvas

395    def get_preamble(self, canvas: Canvas) -> str:
396        """Returns the TeX preamble.
397
398        Args:
399            canvas: The canvas object.
400
401        Returns:
402            str: The TeX preamble.
403
404        Examples:
405            >>> import simetri.graphics as sg
406            >>> from simetri.render.tex import Tex
407            >>> len(Tex().get_preamble(sg.Canvas())) > 0
408            True
409        """
410        libraries, packages = self.get_packages(canvas)
411
412        if packages:
413            packages = f"\\usepackage{{{','.join(packages)}}}\n"
414            if canvas_uses_label_halos(canvas):
415                packages += label_halo_preamble_line()
416            if "fontspec" in packages:
417                fonts_section = f"""\\setmainfont{{{runtime_defaults["main_font"]}}}
418\\setsansfont{{{runtime_defaults["sans_font"]}}}
419\\setmonofont{{{runtime_defaults["mono_font"]}}}\n"""
420        else:
421            packages = ""
422
423        if libraries:
424            libraries = f"\\usetikzlibrary{{{','.join(libraries)}}}\n"
425        else:
426            libraries = ""
427        if canvas.page_size is not None:
428            border = 0
429        else:
430            if canvas.border is None:
431                border = runtime_defaults["border"]
432            elif isinstance(canvas.border, (int, float)):
433                border = canvas.border
434            elif (
435                isinstance(canvas.border, (list, tuple))
436                and len(canvas.border) == 4
437            ):
438                left, bottom, right, top = canvas.border
439                border = f"{{{left}pt {bottom}pt {right}pt {top}pt}}"
440            else:
441                raise ValueError(
442                    "Canvas.border must be a numeric value or a tuple of 4 numeric values."
443                )
444        doc_class = self.get_doc_class(border, runtime_defaults["font_size"])
445        # Check if different fonts are used
446        fonts_section = ""
447        fonts = canvas.get_fonts_list()
448        for font in fonts:
449            if font is None:
450                continue
451            font_family = font.replace(" ", "")
452            fonts_section += (
453                f"\\newfontfamily\\{font_family}[Scale=1.0]{{{font}}}\n"
454            )
455        preamble = f"{doc_class}{packages}{libraries}{fonts_section}"
456
457        indices = False
458        for sketch in canvas.active_page.sketches:
459            if (
460                hasattr(sketch, "marker_type")
461                and sketch.marker_type == "indices"
462            ):
463                indices = True
464                break
465        if indices:
466            font_family = runtime_defaults["indices_font_family"]
467            font_size = runtime_defaults["index_font_size"]
468            if isinstance(font_size, (int, float)):
469                baseline = ceil(float(font_size) * 1.2)
470                font_spec = f"\\{font_family}\\fontsize{{{font_size}}}{{{baseline}}}\\selectfont"
471            else:
472                font_spec = f"\\{font_family}\\{font_size}"
473            count = 0
474            for sketch in canvas.active_page.sketches:
475                if (
476                    hasattr(sketch, "marker_type")
477                    and sketch.marker_type == "indices"
478                ):
479                    preamble += "\\tikzset{\n"
480                    node_style = (
481                        f"nodestyle{count}/.style={{draw, circle, gray, "
482                        f"text=black, fill=white, line width = .5, inner sep=.5, "
483                        f"font={font_spec}}}\n}}\n"
484                    )
485                    preamble += node_style
486                    count += 1
487        for sketch in canvas.active_page.sketches:
488            if sketch.subtype == Types.TEX_SKETCH:  # noqa: SIM102
489                if sketch.location == TexLoc.PREAMBLE:
490                    preamble += sketch.code + "\n"
491        return preamble
