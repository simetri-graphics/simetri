 136def _apply_prefixed_line(
 137    shape: object, style: dict[str, object], prefix: str
 138) -> None:
 139    """Set line attributes from ``prefix`` keys, then ``color`` and ``alpha``."""
 140    if shape is None:
 141        return
 142    for attr in ("line_alpha", "line_color", "line_dash_array", "line_width"):
 143        key = prefix + attr
 144        if key in style and style[key] is not None:
 145            setattr(shape, attr, style[key])
 146        elif attr == "line_color" and "color" in style and style["color"] is not None:
 147            setattr(shape, attr, style["color"])
 148        elif attr == "line_alpha" and "alpha" in style and style["alpha"] is not None:
 149            setattr(shape, attr, style["alpha"])

 177class RadialDimension(Group):
 178    """A RadialDimension object is a dimension that represents a radius.
 179
 180    Args:
 181        center (PointType): The center of the circle.
 182        radius (float): The radius of the circle.
 183        angle (float): The angle of the dimension line.
 184        text_offset (float, optional): Offset for the dimension text.
 185            ``None`` uses ``runtime_defaults["text_offset"]``.
 186        gap (float, optional): The gap between the dimension line and the text.
 187            ``None`` uses ``runtime_defaults["gap"]``.
 188        text (str, optional): Label text. Empty string uses the radius
 189            rounded to ``runtime_defaults["n_dim_digits"]``.
 190        ext_length (float, optional): Extension length when ``reverse_arrow``
 191            is True. ``None`` uses ``runtime_defaults["rev_arrow_length"]``.
 192        reverse_arrow (bool, optional): If True, an extension line is created.
 193            Defaults to False.
 194        keep_inside (bool, optional): Defaults to True.
 195        **kwargs: Label and line styles. Prefixed names match
 196            ``canvas.draw_dimension`` (``ext_``, ``shaft_``, ``head_``,
 197            ``tag_``). Unprefixed ``font_color``, ``font_family``,
 198            ``font_alpha``, ``bold``, ``italic``, ``anchor``, ``align``,
 199            ``fill``, ``fill_color``, ``color``, and ``alpha`` are also
 200            accepted.
 201
 202    Examples:
 203        >>> import simetri.graphics as sg
 204        >>> dim = sg.RadialDimension((0, 0), 10)
 205        >>> dim.text
 206        '10.0'
 207        >>> dim.center
 208        (0, 0)
 209        >>> dim.radius
 210        10
 211        >>> tuple(dim.tag.pos)
 212        (5.0, 0.0)
 213        >>> len(dim)
 214        2
 215    """
 216
 217    def __init__(
 218        self,
 219        center: PointType,
 220        radius: float | None = None,
 221        angle: float = 0,
 222        text: str = "",
 223        text_offset: float | None = None,
 224        ext_length: float | None = None,
 225        reverse_arrow: bool = False,
 226        keep_inside: bool = True,
 227        gap: float | None = None,
 228        **kwargs: object,
 229    ) -> None:
 230        """Create a radial dimension annotation.
 231
 232        See the class docstring for argument details.
 233        """
 234        text_offset, gap, ext_length = get_defaults(
 235            ["text_offset", "gap", "rev_arrow_length"],
 236            [text_offset, gap, ext_length],
 237        )
 238        self.center = center
 239        self.radius = radius
 240        self.angle = angle
 241        self.text = text
 242        self.text_offset = text_offset
 243        self.ext_length = ext_length
 244        self.reverse_arrow = reverse_arrow
 245        self.keep_inside = keep_inside
 246        self.gap = gap
 247        _check_dimension_kwargs(kwargs)
 248        self.kwargs = kwargs
 249        super().__init__(subtype=Types.RADIAL_DIMENSION)
 250
 251        p2 = polar_to_cartesian(self.radius, self.angle, self.center)
 252
 253        self.arrow = Arrow(center, p2)
 254        self.extension = None
 255        if self.reverse_arrow:
 256            self.extension = extended_line(ext_length, [center, p2])
 257
 258        if self.text == "":
 259            if self.radius is not None:
 260                self.text = _format_dim_value(self.radius)
 261            else:
 262                self.text = f"r = {_format_dim_value(distance(center, p2))}"
 263
 264        self.tag = Tag(
 265            self.text,
 266            midpoint(center, p2),
 267            **_dimension_tag_kwargs(
 268                kwargs, runtime_defaults["font_size"], Anchor.CENTER, Align.CENTER
 269            ),
 270        )
 271        _apply_prefixed_line(self.arrow.line, kwargs, "shaft_")
 272        for head in self.arrow.heads:
 273            _apply_prefixed_head(head, kwargs)
 274        _apply_prefixed_line(self.extension, kwargs, "ext_")
 275        self._items = [self.arrow, self.tag]
 276
 277        super().__init__(self._items, subtype=Types.RADIAL_DIMENSION)

 297class AngularDimension(Group):
 298    """An angle dimension: two extension lines, an arc, and a label.
 299
 300    The arc runs counter-clockwise from ``start_angle`` to
 301    ``end_angle`` at ``radius``. ``gap_angle`` shortens the arc at
 302    each end. The extension lines lie on the two sides, from ``gap``
 303    out to ``radius * (1 + ext_angle)``. The label is the full sweep
 304    in radians, outside the arc by ``text_offset``.
 305
 306    Args:
 307        center (PointType): The vertex of the angle.
 308        radius (float): Radius of the dimension arc.
 309        start_angle (float): First side, in radians.
 310        end_angle (float): Second side, in radians. The sweep is
 311            counter-clockwise from ``start_angle`` to ``end_angle``.
 312        ext_angle (float, optional): Extra radius of each extension past
 313            the arc, as a fraction of ``radius`` via ``1 + ext_angle``.
 314            ``None`` uses ``runtime_defaults["ext_angle"]``.
 315        gap_angle (float, optional): Angle cut from each end of the
 316            dimension arc. ``None`` uses ``runtime_defaults["gap_angle"]``.
 317        text_offset (float, optional): Distance from the arc to the
 318            label. ``None`` uses ``runtime_defaults["text_offset"]``.
 319        gap (float, optional): Distance from the vertex to the start
 320            of each extension. ``None`` uses ``runtime_defaults["gap"]``.
 321        text (str, optional): Label text. ``None`` uses the sweep in
 322            radians, rounded to ``runtime_defaults["n_dim_digits"]``.
 323            Defaults to None.
 324        font_size (float, optional): Label size. ``None`` uses
 325            ``runtime_defaults["font_size"]``.
 326        **kwargs: Label and line styles. Prefixed names match
 327            ``canvas.draw_dimension`` (``ext_``, ``shaft_``, ``head_``,
 328            ``tag_``). Unprefixed ``font_color``, ``font_family``,
 329            ``font_alpha``, ``bold``, ``italic``, ``anchor``, ``align``,
 330            ``fill``, ``fill_color``, ``color``, and ``alpha`` are also
 331            accepted.
 332
 333    Examples:
 334        >>> import simetri.graphics as sg
 335        >>> dim = sg.AngularDimension((0, 0), 20, 0, sg.pi / 2, 0.1, 0.1)
 336        >>> dim.center
 337        (0, 0)
 338        >>> dim.radius
 339        20
 340        >>> dim.start_angle
 341        0
 342        >>> dim.end_angle
 343        1.5707963267948966
 344        >>> dim.text
 345        '1.57'
 346        >>> len(dim)
 347        3
 348        >>> bare = sg.AngularDimension((0, 0), 20, 0, sg.pi / 2)
 349        >>> bare.ext_angle
 350        0
 351        >>> bare.gap_angle
 352        0
 353        >>> bare.text_offset
 354        5
 355    """
 356
 357    def __init__(
 358        self,
 359        center: PointType,
 360        radius: float,
 361        start_angle: float,
 362        end_angle: float,
 363        ext_angle: float | None = None,
 364        gap_angle: float | None = None,
 365        text_offset: float | None = None,
 366        gap: float | None = None,
 367        text: str | None = None,
 368        font_size: float | None = None,
 369        **kwargs: object,
 370    ) -> None:
 371        """Create an angular dimension with extension lines and an arc.
 372
 373        See the class docstring for argument details.
 374        """
 375        text_offset, gap, ext_angle, gap_angle, font_size = get_defaults(
 376            ["text_offset", "gap", "ext_angle", "gap_angle", "font_size"],
 377            [text_offset, gap, ext_angle, gap_angle, font_size],
 378        )
 379        if radius <= 0:
 380            raise ValueError("AngularDimension radius must be positive.")
 381        if ext_angle < 0:
 382            raise ValueError("AngularDimension ext_angle must be non-negative.")
 383        if gap_angle < 0:
 384            raise ValueError("AngularDimension gap_angle must be non-negative.")
 385        if gap < 0:
 386            raise ValueError("AngularDimension gap must be non-negative.")
 387        if gap >= radius:
 388            raise ValueError("AngularDimension gap must be less than radius.")
 389        abs_tol = runtime_defaults["abs_tol"]
 390        sweep = (end_angle - start_angle) % (2 * pi)
 391        if sweep < abs_tol:
 392            raise ValueError("AngularDimension angle must be nonzero.")
 393        if sweep - 2 * gap_angle <= abs_tol:
 394            raise ValueError(
 395                "AngularDimension gap_angle removes the dimension arc."
 396            )
 397        if text is None:
 398            text = _format_dim_value(sweep)
 399        center_x, center_y = center[:2]
 400        arc_start = start_angle + gap_angle
 401        arc_end = start_angle + sweep - gap_angle
 402        mid_angle = start_angle + sweep / 2
 403        self.text_pos = polar_to_cartesian(
 404            radius + text_offset, mid_angle, (center_x, center_y)
 405        )
 406
 407        self.center = center
 408        self.radius = radius
 409        self.start_angle = start_angle
 410        self.end_angle = end_angle
 411        self.ext_angle = ext_angle
 412        self.gap_angle = gap_angle
 413        self.text_offset = text_offset
 414        self.gap = gap
 415        self.text = text
 416        self.font_size = font_size
 417        _check_dimension_kwargs(kwargs)
 418        self.kwargs = kwargs
 419        self.text_anchor = Anchor.CENTER
 420        self.text_align = Align.CENTER
 421        self.ext3 = None
 422        self.arrow1 = None
 423        self.arrow2 = None
 424        self.midline = None
 425
 426        outer = radius * (1.0 + ext_angle)
 427        self.ext1 = Line(
 428            polar_to_cartesian(gap, start_angle, (center_x, center_y)),
 429            polar_to_cartesian(outer, start_angle, (center_x, center_y)),
 430        )
 431        self.ext2 = Line(
 432            polar_to_cartesian(gap, end_angle, (center_x, center_y)),
 433            polar_to_cartesian(outer, end_angle, (center_x, center_y)),
 434        )
 435        self.dim_line = ArcArrow(
 436            (center_x, center_y),
 437            radius,
 438            start_angle=arc_start,
 439            end_angle=arc_end,
 440        )
 441        self.dim_line.line = self.dim_line.arc
 442        self.dim_line.heads = [
 443            self.dim_line.arrow_head1,
 444            self.dim_line.arrow_head2,
 445        ]
 446        self.tag = Tag(
 447            text,
 448            pos=self.text_pos,
 449            **_dimension_tag_kwargs(
 450                kwargs, self.font_size, self.text_anchor, self.text_align
 451            ),
 452        )
 453        super().__init__(
 454            [self.ext1, self.ext2, self.dim_line],
 455            subtype=Types.ANGULAR_DIMENSION,
 456        )

 476class Dimension(Group):
 477    """A linear dimension: extension lines, dimension line, and a label.
 478
 479    ``orientation`` is ``"horizontal"`` or ``"vertical"``. ``None``
 480    compares the two points: the larger change in ``x`` is horizontal,
 481    and the larger change in ``y`` is vertical. An equal change is
 482    horizontal. The label position is stored on ``text_pos``.
 483
 484    Args:
 485        p1 (PointType): First feature point.
 486        p2 (PointType): Second feature point.
 487        side (str): Which side of the features the dimension line is
 488            on. Horizontal: ``"up"`` or ``"down"``. Vertical:
 489            ``"left"`` or ``"right"``.
 490        text_offset (float, optional): Distance from the gap to the
 491            dimension line. ``None`` uses ``runtime_defaults["text_offset"]``.
 492        text (str, optional): Label text. ``None`` uses the measured
 493            length, rounded to ``runtime_defaults["n_dim_digits"]``.
 494            Defaults to None.
 495        ext_line_extension (float, optional): How far each extension
 496            continues past the dimension line. ``None`` uses
 497            ``runtime_defaults["overshoot"]``.
 498        ext_line_offset (float, optional): Gap from the feature to the
 499            start of the extension. ``None`` uses ``runtime_defaults["gap"]``.
 500        text_horiz_offset (float, optional): Offset of the label along
 501            the dimension line when ``text_loc`` is not ``"middle"``.
 502            ``None`` uses ``runtime_defaults["ext_length2"]``.
 503        text_loc (str, optional): ``"middle"``, or a side.
 504            Horizontal: ``"left"`` or ``"right"``. Vertical: ``"up"``
 505            or ``"down"``. Defaults to ``"middle"``.
 506        stub_length (float, optional): Outward shaft length when the
 507            label is not in the middle. ``None`` uses
 508            ``runtime_defaults["stub_length"]``.
 509        show_midline (bool, optional): Draw the line between the side
 510            arrows. Defaults to True.
 511        orientation (str, optional): ``"horizontal"`` or ``"vertical"``.
 512            ``None`` uses the larger change in ``x`` or ``y``. An equal
 513            change is horizontal. Defaults to None.
 514        font_size (float, optional): Label size. ``None`` uses
 515            ``runtime_defaults["font_size"]``.
 516        **kwargs: Label and line styles. Prefixed names match
 517            ``canvas.draw_dimension`` (``ext_``, ``shaft_``, ``head_``,
 518            ``tag_``). Unprefixed ``font_color``, ``font_family``,
 519            ``font_alpha``, ``bold``, ``italic``, ``anchor``, ``align``,
 520            ``fill``, ``fill_color``, ``color``, and ``alpha`` are also
 521            accepted.
 522
 523    Examples:
 524        >>> import simetri.graphics as sg
 525        >>> dim = sg.Dimension((0, 0), (40, 0), 'up', 8)
 526        >>> dim.text
 527        '40.0'
 528        >>> dim.orientation
 529        'horizontal'
 530        >>> sg.Dimension((0, 0), (0, 40), "right").orientation
 531        'vertical'
 532        >>> sg.Dimension((0, 0), (40, 10), "up").orientation
 533        'horizontal'
 534        >>> sg.Dimension((0, 0), (10, 40), "right").orientation
 535        'vertical'
 536        >>> sg.Dimension((0, 0), (10, 10), "up").orientation
 537        'horizontal'
 538        >>> dim.text_pos
 539        (20.0, 13)
 540        >>> dim.p1
 541        (0, 0)
 542        >>> dim.p2
 543        (40, 0)
 544        >>> len(dim)
 545        3
 546        >>> hidden = sg.Dimension(
 547        ...     (0, 0), (40, 0), "up", 8, text_loc="left", show_midline=False
 548        ... )
 549        >>> hidden.midline is not None
 550        True
 551        >>> hidden.midline in hidden.elements
 552        False
 553        >>> outer = sg.Dimension((0, 40), (0, 0), "right", text_loc="up")
 554        >>> outer.text_pos
 555        (10, 65)
 556        >>> outer.arrow1.p1[1]
 557        55
 558        >>> outer.arrow2.p2[1]
 559        -15
 560        >>> plain = sg.Dimension((0, 0), (40, 0), "up")
 561        >>> plain.text_offset
 562        5
 563        >>> plain.stub_length
 564        15
 565        >>> plain.text_pos
 566        (20.0, 10)
 567    """

 570        self,

 572        p2: PointType,
 573        side: str,
 574        text_offset: float | None = None,

 814def _aligned_side_normal(angle: float, side: str) -> tuple[float, float]:
 815    """Return the unit normal of ``angle`` that faces ``side``."""

 819            "lower_left, or lower_right."

 823    left_y = cos(angle)
