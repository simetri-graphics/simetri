simetri.render.style_map

This module contains the Style classes used to set the style of shapes, lines, text, and tags. Shape and Tag objects use the maps to create aliases for style attributes.

  40def _set_style_args(
  41    obj: Any,
  42    attribs: list[str],
  43    exact: list[str] | None = None,
  44    prefix: str | None = None,
  45    values: dict[str, Any] | None = None,
  46) -> None:
  47    """Apply default style values to ``obj`` for the given attribute names.
  48
  49    Args:
  50        obj: Style instance to mutate.
  51        attribs: Attribute names to initialize.
  52        exact: Names read from ``defaults`` without a prefix.
  53        prefix: Prefix for ``runtime_defaults[f"{prefix}_{attrib}"]`` lookups.
  54        values: Explicit overrides keyed by attribute name.
  55    """
  56    for attrib in attribs:
  57        if values and values.get(attrib, None) is not None:
  58            setattr(obj, attrib, values[attrib])
  59        else:
  60            if exact and attrib in exact:
  61                default = runtime_defaults.get(attrib, VOID)
  62                if default != VOID:
  63                    setattr(obj, attrib, default)
  64            else:
  65                if prefix:
  66                    setattr(obj, attrib, runtime_defaults[f"{prefix}_{attrib}"])
  67                else:
  68                    default = runtime_defaults.get(attrib, VOID)
  69                    if default != VOID:
  70                        setattr(obj, attrib, default)

  73def _get_style_attribs(
  74    style: Types.STYLE,
  75    prefix: str | None = None,
  76    exact: list | None = None,
  77    exclude: list | None = None,
  78) -> list[str]:
  79    """Get the list of attributes from the given Style object.
  80
  81    Args:
  82        style (Types.STYLE): The style object to get attributes from.
  83        prefix (str, optional): The prefix to use for the attributes. Defaults to None.
  84        exact (list, optional): List of exact attributes to include. Defaults to None.
  85        exclude (list, optional): List of attributes to exclude. Defaults to None.
  86
  87    Returns:
  88        list[str]: List of attributes.
  89    """
  90    attribs = style.__dict__.keys()
  91    res = []
  92    for attrib in attribs:
  93        if attrib in exclude:
  94            continue
  95        if attrib in exact:
  96            res.append(attrib)
  97        else:
  98            res.append(f"{prefix}_{attrib}")
  99    return res

 344def _style_init(
 345    style: Any,
 346    exact: list[str] | None = None,
 347    exclude: list[str] | None = None,
 348    prefix: str = "",
 349    subtype: Types | None = None,
 350) -> None:
 351    """Initialize the style object.
 352
 353    Args:
 354        style: The style object to initialize (mutated).
 355        exact: List of exact attributes to include.
 356        exclude: List of attributes to exclude.
 357        prefix: Prefix to use for the attributes.
 358        subtype: The subtype of the style.
 359    """
 360    if exclude is None:
 361        exclude = []
 362    if exact is None:
 363        exact = []
 364    attribs = [x for x in style.__dict__ if x not in exclude]
 365    _set_style_args(style, attribs, exact, prefix=prefix)
 366
 367    style.attribs = _get_style_attribs(
 368        style, prefix=prefix, exact=exact, exclude=exclude
 369    )
 370    style.type = Types.STYLE
 371    style.subtype = subtype

1720    map_dict: dict[str, tuple[str, str]],
1721    styles: list[Any],
1722    paths: list[str],
1723    prefixes: list[str],
1724    debug: bool = False,
1725) -> dict[str, tuple[str, str]]:

1908class StyleObj:
1909    """
1910    An object-based style class that only allows updates to predefined attributes.
1911    Setting new attributes will raise an AttributeError.
1912    Provides attribute-style access (obj.attr) instead of dictionary access.
1913    Includes type validation based on settings.default_types.
1914
1915    Examples:
1916        >>> from simetri.render.style_map import line_style_obj
1917        >>> obj = line_style_obj(validate_types=False, line_width=2)
1918        >>> obj.line_width
1919        2
1920    """
1921
1922    def __init__(
1923        self,
1924        style_map: dict[str, tuple[str, str]],
1925        validate_types: bool = True,
1926        **kwargs: object,
1927    ) -> None:
1928        """Initialize StyleObj with predefined attributes from style_map.
1929
1930        Args:
1931            style_map: Dictionary mapping attribute names to their paths
1932            validate_types: Whether to validate types against default_types
1933            **kwargs: Initial attribute values
1934        """
1935        # Store the allowed attributes and validation setting
1936        object.__setattr__(self, "_allowed_attrs", set(style_map.keys()))
1937        object.__setattr__(self, "_validate_types", validate_types)
1938
1939        # Initialize all attributes to None
1940        for attr in style_map:
1941            object.__setattr__(self, attr, None)
1942
1943        # Set any provided values (with validation)
1944        for key, value in kwargs.items():
1945            setattr(self, key, value)
1946
1947    def _validate_value(self, name: str, value: object) -> None:
1948        """Validate a value against its expected type from default_types.
1949
1950        Args:
1951            name: The attribute name
1952            value: The value to validate
1953
1954        Raises:
1955            TypeError: If value doesn't match expected type
1956            ValueError: If value is invalid for the type
1957        """
1958        if not self._validate_types or value is None:
1959            return  # Skip validation if disabled or value is None
1960
1961        # Get expected type from default_types
1962        expected_type = default_types.get(name)
1963        if expected_type is None:
1964            return  # No type information available, skip validation
1965
1966        # Handle special cases
1967        if expected_type == Sequence:
1968            # Check if it's sequence-like (list, tuple, etc.)
1969            if not hasattr(value, "__iter__") or isinstance(
1970                value, (str, bytes)
1971            ):
1972                raise TypeError(
1973                    f"Attribute '{name}' must be a sequence (list, tuple, etc.), got {type(value).__name__}"
1974                )
1975            return
1976
1977        # Handle enum types
1978        if (
1979            hasattr(expected_type, "__bases__")
1980            and enum.Enum in expected_type.__bases__
1981        ):
1982            if not isinstance(value, expected_type):
1983                # Allow string values for enums if they match enum names
1984                if isinstance(value, str):
1985                    try:
1986                        # Try to convert string to enum - just check validity
1987                        _ = expected_type[value.upper()]
1988                        return  # String matches enum name, validation passes
1989                    except (KeyError, AttributeError):
1990                        pass
1991                raise TypeError(
1992                    f"Attribute '{name}' must be of type {expected_type.__name__} or a valid enum name, got {type(value).__name__}"
1993                )
1994            return
1995
1996        # Handle Union types (e.g., FontFamily | str)
1997        if (
1998            hasattr(expected_type, "__origin__")
1999            and expected_type.__origin__ is Union
2000        ):
2001            # Check if value matches any of the union types
2002            union_args = expected_type.__args__
2003            for union_type in union_args:
2004                try:
2005                    if isinstance(value, union_type):
2006                        return  # Matches one of the union types
2007                except TypeError:
2008                    continue  # Some types can't be used with isinstance
2009            raise TypeError(
2010                f"Attribute '{name}' must be one of {union_args}, got {type(value).__name__}"
2011            )
2012
2013        # Handle basic types
2014        if not isinstance(value, expected_type):
2015            # Special case for numeric types - allow some flexibility
2016            if expected_type is float and isinstance(value, (int, float)):
2017                return  # Allow int for float attributes
2018
2019            raise TypeError(
2020                f"Attribute '{name}' must be of type {expected_type.__name__}, got {type(value).__name__}"
2021            )
2022
2023        # Additional validation for specific types
2024        if expected_type is float:
2025            # Check for special float constraints (e.g., alpha should be 0-1)
2026            if (name.endswith("_alpha") or name == "alpha") and not (
2027                0 <= value <= 1
2028            ):
2029                raise ValueError(
2030                    f"Alpha attribute '{name}' must be between 0 and 1, got {value}"
2031                )
2032
2033            # Check for positive values where appropriate
2034            if any(
2035                keyword in name
2036                for keyword in [
2037                    "width",
2038                    "height",
2039                    "size",
2040                    "radius",
2041                    "distance",
2042                    "gap",
2043                    "sep",
2044                    "length",
2045                ]
2046            ) and value < 0:
2047                raise ValueError(
2048                    f"Attribute '{name}' must be non-negative, got {value}"
2049                )
2050
2051    def __setattr__(self, name: str, value: object) -> None:
2052        """Override attribute setting to prevent adding new attributes and validate values."""
2053        if hasattr(self, "_allowed_attrs") and name not in self._allowed_attrs:
2054            raise AttributeError(
2055                f"Cannot add new attribute '{name}' to StyleObj. "
2056                f"Allowed attributes are: {sorted(self._allowed_attrs)}"
2057            )
2058
2059        # Validate the value if not a private attribute
2060        if hasattr(self, "_validate_types") and not name.startswith("_"):
2061            self._validate_value(name, value)
2062
2063        object.__setattr__(self, name, value)
2064
2065    def __getattr__(self, name: str) -> Any:
2066        """Provide helpful error message for non-existent attributes."""
2067        if hasattr(self, "_allowed_attrs"):
2068            raise AttributeError(
2069                f"'{self.__class__.__name__}' object has no attribute '{name}'. "
2070                f"Allowed attributes are: {sorted(self._allowed_attrs)}"
2071            )
2072        raise AttributeError(
2073            f"'{self.__class__.__name__}' object has no attribute '{name}'"
2074        )
2075
2076    def update(self, **kwargs: object) -> None:
2077        """Update multiple attributes at once.
2078
2079        Examples:
2080            >>> from simetri.render.style_map import line_style_obj
2081            >>> obj = line_style_obj(validate_types=False)
2082            >>> obj.update(line_width=3)
2083            >>> obj.line_width
2084            3
2085        """
2086        for key, value in kwargs.items():
2087            setattr(self, key, value)
2088
2089    def to_dict(self) -> dict[str, Any]:
2090        """Convert StyleObj to dictionary, excluding None values.
2091
2092        Examples:
2093            >>> from simetri.render.style_map import line_style_obj
2094            >>> obj = line_style_obj(validate_types=False, line_width=2)
2095            >>> obj.to_dict()['line_width']
2096            2
2097        """
2098        return {
2099            attr: getattr(self, attr)
2100            for attr in self._allowed_attrs
2101            if getattr(self, attr) is not None
2102        }
2103
2104    def __repr__(self) -> str:
2105        """Provide a readable representation showing non-None attributes."""
2106        non_none_attrs = {
2107            attr: getattr(self, attr)
2108            for attr in self._allowed_attrs
2109            if getattr(self, attr) is not None
2110        }
2111        return f"{self.__class__.__name__}({non_none_attrs})"
