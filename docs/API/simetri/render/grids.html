 40def _adjacent_index_pairs(
 41    cluster: Sequence[tuple[int, PointType]],
 42) -> list[tuple[int, int]]:
 43    """Return consecutive undirected index pairs along a sorted collinear cluster."""
 44    if len(cluster) < 2:
 45        return []
 46    pairs: list[tuple[int, int]] = []
 47    for first_item, second_item in zip(cluster, cluster[1:]):
 48        first = first_item[0]
 49        second = second_item[0]
 50        if first > second:
 51            first, second = second, first
 52        pairs.append((first, second))
 53    return pairs

 56def _consecutive_pairs_sharing_axis(
 57    indexed_points: Sequence[tuple[int, PointType]],
 58    axis: int,
 59) -> list[tuple[int, int]]:
 60    """Connect consecutive vertices that share coordinate ``axis`` (0 is x, 1 is y)."""
 61    if not indexed_points:
 62        return []
 63    other = 1 - axis
 64    ordered = sorted(
 65        indexed_points,
 66        key=lambda item: (item[1][axis], item[1][other]),
 67    )
 68    pairs: list[tuple[int, int]] = []
 69    cluster: list[tuple[int, PointType]] = [ordered[0]]
 70    cluster_key = ordered[0][1][axis]
 71    for item in ordered[1:]:
 72        if _coords_close(item[1][axis], cluster_key):
 73            cluster.append(item)
 74            continue
 75        pairs.extend(_adjacent_index_pairs(cluster))
 76        cluster = [item]
 77        cluster_key = item[1][axis]
 78    pairs.extend(_adjacent_index_pairs(cluster))
 79    return pairs

 82def _extreme_vertex_index(
 83    points: Sequence[PointType],
 84    plus: bool,
 85    maximize: bool,
 86) -> int:
 87    """Return the index of the vertex with extreme ``x+y`` or ``x-y``.
 88
 89    ``plus`` True uses ``x + y`` (NE / SW). False uses ``x - y``
 90    (SE / NW).
 91    """
 92    first_x, first_y = points[0][:2]
 93    best_value = (first_x + first_y) if plus else (first_x - first_y)
 94    best_index = 0
 95    for index, point in enumerate(points[1:], start=1):
 96        point_x, point_y = point[:2]
 97        value = (point_x + point_y) if plus else (point_x - point_y)
 98        if maximize:
 99            if value > best_value:
100                best_value = value
101                best_index = index
102        elif value < best_value:
103            best_value = value
104            best_index = index
105    return best_index

122def _rectangular_corner_indices(points: Sequence[PointType]) -> list[int]:
123    """Return the four geometric corner indices of a rectangular point set."""
124    if len(points) < 4:
125        raise ValueError(
126            "corners_only requires a rectangular grid with four distinct corners"
127        )
128    corners = [
129        _extreme_vertex_index(points, True, True),
130        _extreme_vertex_index(points, False, True),
131        _extreme_vertex_index(points, True, False),
132        _extreme_vertex_index(points, False, False),
133    ]
134    unique = sorted(set(corners))
135    if len(unique) != 4:
136        raise ValueError(
137            "corners_only requires four distinct rectangular corners"
138        )
139    return unique

142def _rect_grid_points(
143    n_x: int,
144    n_y: int,
145    cell_size_x: float,
146    cell_size_y: float,
147) -> list[PointType]:
148    """Return rectangular perimeter grid points ordered by polar angle."""
149    x_values = [cell_size_x * (index - n_x / 2) for index in range(n_x + 1)]
150    y_values = [cell_size_y * (index - n_y / 2) for index in range(n_y + 1)]
151    points: list[PointType] = []
152    for x_index, x_value in enumerate(x_values):
153        for y_index, y_value in enumerate(y_values):
154            if x_index in (0, n_x) or y_index in (0, n_y):
155                points.append((x_value, y_value))
156    points.sort(key=_polar_angle)
157    return points

160def _square_grid_points(
161    center: PointType,
162    n: int,
163    cell_size: float,
164) -> list[PointType]:
165    """Return square-perimeter points counterclockwise from the right midpoint."""
166    if type(n) is not int or n <= 0 or n % 8 != 0:
167        raise ValueError("n must be a positive multiple of 8")
168
169    center_x, center_y = center[:2]
170    side_cells = n // 4
171    half_cells = side_cells // 2
172    half_extent = half_cells * cell_size
173
174    points: list[PointType] = [(center_x + half_extent, center_y)]
175    for y_index in range(1, half_cells + 1):
176        points.append((center_x + half_extent, center_y + y_index * cell_size))
177    for x_index in range(half_cells - 1, -half_cells - 1, -1):
178        points.append((center_x + x_index * cell_size, center_y + half_extent))
179    for y_index in range(half_cells - 1, -half_cells - 1, -1):
180        points.append((center_x - half_extent, center_y + y_index * cell_size))
181    for x_index in range(-half_cells + 1, half_cells + 1):
182        points.append((center_x + x_index * cell_size, center_y - half_extent))
183    for y_index in range(-half_cells + 1, 0):
184        points.append((center_x + half_extent, center_y + y_index * cell_size))
185
186    return points

189class Grid(Group):
190    """A base-class for all grids.
191
192    ``connections``, ``skip``, ``corners_only``, ``border``,
193    ``centerlines``, ``orthogonals``, and ``diagonals`` control what
194    ``canvas.draw(grid)`` strokes. ``connections = [3, 5]`` draws chords ``(i, i+3)`` and
195    ``(i, i+5)`` for every start index ``i`` (modulo the number of
196    vertices). ``skip = 2`` draws ``(i, i+2)``, ``(i, i+4)``, … until
197    the index wraps to ``i``. If ``corners_only`` is True, those start
198    indices are only the geometric corner vertices (on a ``SquareGrid``
199    with index ``0`` on +x, ``n=16``: ``2``, ``6``, ``10``, ``14``).
200    ``RectGrid`` uses its four rectangular corners the same way. On a
201    circular or hexagonal grid every vertex is a corner, so the chords
202    are unchanged. ``border``, ``centerlines``, ``orthogonals``,
203    and ``diagonals`` are ``False``, ``True`` (default line color and
204    width), or a style dict, like ``canvas.draw_bbox``. ``orthogonals``
205    defaults to True and strokes adjacent horizontal and vertical
206    segments between vertices that share an x- or y-coordinate.
207    ``centerlines`` defaults to False and strokes the horizontal and
208    vertical midlines of the grid's axis-aligned bounding box.
209    ``diagonals`` defaults to False and, on a ``SquareGrid`` or
210    ``RectGrid``, strokes the two main diagonals between opposite
211    geometric corners.
212
213    Examples:
214        >>> from simetri.base.all_enums import GridType
215        >>> from simetri.render.grids import Grid
216        >>> pts = [(0, 0), (40, 0), (40, 40), (0, 40)]
217        >>> grid = Grid(GridType.SQUARE, points=pts, n=4)
218        >>> len(grid.points)
219        4
220    """
221
222    def __init__(
223        self,
224        grid_type: GridType,
225        center: PointType = (0, 0),
226        n: int = 9,
227        radius: float = 100,
228        points: Sequence[PointType] | None = None,
229        n_circles: int = 1,
230    ) -> None:
231        """Initialize a geometric grid of points and connecting lines.
232
233        Args:
234            grid_type (GridType): Kind of grid to build.
235            center (PointType, optional): Grid center. Defaults to ``(0, 0)``.
236            n (int, optional): Number of points. Defaults to 9.
237            radius (float, optional): Circumradius. Defaults to 100.
238            points (Sequence[PointType], optional): Explicit grid points.
239            n_circles (optional): Number of concentric circles for circular grids.
240
241        Raises:
242            ValueError: If ``grid_type`` is not recognized.
243        """
244        if grid_type not in d_grid_types:
245            raise ValueError(f"Invalid grid type: {grid_type}.")
246        super().__init__(subtype=d_grid_types[grid_type])
247        self.center = center
248        self.radius = radius
249        self.n = n
250        self.n_circles = n_circles
251        self.connections: Sequence[int] | None = None
252        self.skip: int | None = None
253        self.border: bool | dict[str, object] = False
254        self.centerlines: bool | dict[str, object] = False
255        self.orthogonals: bool | dict[str, object] = True
256        self.diagonals: bool | dict[str, object] = False
257        self.corners_only: bool = False
258
259        self._points = Shape(points)
260        self.append(self._points)
261
262    @property
263    def points(self) -> list[PointType]:
264        """Return the grid vertex points.
265
266        Examples:
267            >>> from simetri.base.all_enums import GridType
268            >>> from simetri.render.grids import Grid
269            >>> pts = [(0, 0), (40, 0), (40, 40), (0, 40)]
270            >>> Grid(GridType.SQUARE, points=pts, n=4).points[0]
271            (0.0, 0.0)
272        """
273        return self._points.vertices

275    def _connection_start_indices(self) -> list[int]:
276        """Return vertex indices that start ``connections`` and ``skip`` chords.
277
278        If ``corners_only`` is False, every vertex is a start. If True, a
279        square grid uses the four geometric corners; circular and hexagonal
280        grids still use every vertex.
281        """
282        n_vertices = len(self.points)
283        if n_vertices == 0:
284            return []
285        if not self.corners_only:
286            return list(range(n_vertices))
287        if self.subtype == Types.SQUARE_GRID:
288            return _rectangular_corner_indices(self.points)
289        return list(range(n_vertices))

291    def line_index_pairs(self) -> list[tuple[int, int]]:
292        """Return undirected vertex-index pairs for ``connections`` and ``skip``.
293
294        Indices wrap with ``mod n``. A chord is listed once.
295
296        Returns:
297            list[tuple[int, int]]: Pairs ``(i, j)`` with ``i < j``.
298
299        Raises:
300            TypeError: If ``connections`` is not a sequence of ints.
301            ValueError: If ``skip`` is not a positive integer, or if
302                ``corners_only`` is True on a square grid without four
303                distinct corners.
304
305        Examples:
306            >>> import simetri.graphics as sg
307            >>> grid = sg.CircularGrid(n=6, radius=40)
308            >>> grid.connections = [3]
309            >>> grid.line_index_pairs()
310            [(0, 3), (1, 4), (2, 5)]
311            >>> grid.connections = None
312            >>> grid.skip = 2
313            >>> grid.line_index_pairs()
314            [(0, 2), (0, 4), (1, 3), (1, 5), (2, 4), (3, 5)]
315            >>> sq = sg.SquareGrid(n=16, cell_size=25)
316            >>> sq.skip = 2
317            >>> sq.corners_only = True
318            >>> pairs = sq.line_index_pairs()
319            >>> (2, 4) in pairs and (2, 6) in pairs and (2, 8) in pairs
320            True
321            >>> (6, 8) in pairs and (6, 10) in pairs
322            True
323            >>> (1, 3) in pairs
324            False
325            >>> (0, 4) in pairs
326            False
327        """
328        points = self.points
329        n_vertices = len(points)
330        if n_vertices == 0:
331            return []
332        starts = self._connection_start_indices()
333        pairs: list[tuple[int, int]] = []
334        seen: set[tuple[int, int]] = set()
335
336        def add_pair(start: int, end: int) -> None:
337            first = start % n_vertices
338            second = end % n_vertices
339            if first == second:
340                return
341            if first > second:
342                first, second = second, first
343            key = (first, second)
344            if key in seen:
345                return
346            seen.add(key)
347            pairs.append(key)
348
349        connections = self.connections
350        if connections is not None:
351            if isinstance(connections, (str, bytes)):
352                raise TypeError("connections must be a sequence of integers")
353            for offset in connections:
354                if type(offset) is not int:
355                    raise TypeError(
356                        "connections must be a sequence of integers"
357                    )
358                for index in starts:
359                    add_pair(index, index + offset)
360
361        skip = self.skip
362        if skip is not None:
363            if type(skip) is not int or skip <= 0:
364                raise ValueError("skip must be a positive integer")
365            for index in starts:
366                offset = skip
367                while offset % n_vertices != 0:
368                    add_pair(index, index + offset)
369                    offset += skip
370
371        return pairs

373    def orthogonal_index_pairs(self) -> list[tuple[int, int]]:
374        """Return undirected pairs of vertices that share an x or y coordinate.
375
376        Along each vertical line (shared x) and each horizontal line
377        (shared y), consecutive vertices are connected. A pair is listed
378        once with ``i < j``.
379
380        Returns:
381            list[tuple[int, int]]: Adjacent axis-aligned index pairs.
382
383        Examples:
384            >>> import simetri.graphics as sg
385            >>> circ = sg.CircularGrid(n=6, radius=40)
386            >>> circ.orthogonal_index_pairs()
387            [(2, 4), (1, 5), (4, 5), (0, 3), (1, 2)]
388            >>> sq = sg.SquareGrid(n=16, cell_size=25)
389            >>> len(sq.orthogonal_index_pairs())
390            22
391        """
392        indexed_points = list(enumerate(self.points))
393        vertical = _consecutive_pairs_sharing_axis(indexed_points, 0)
394        horizontal = _consecutive_pairs_sharing_axis(indexed_points, 1)
395        seen: set[tuple[int, int]] = set()
396        pairs: list[tuple[int, int]] = []
397        for pair in vertical + horizontal:
398            if pair in seen:
399                continue
400            seen.add(pair)
401            pairs.append(pair)
402        return pairs

404    def diagonal_index_pairs(self) -> list[tuple[int, int]]:
405        """Return the two main-diagonal vertex pairs of a rectangular grid.
406
407        ``SquareGrid`` and ``RectGrid`` have these pairs. Other grid
408        types return an empty list. The pairs are opposite geometric corners:
409        northeast–southwest (max ``x+y`` to min ``x+y``) and
410        southeast–northwest (max ``x-y`` to min ``x-y``).
411
412        Returns:
413            list[tuple[int, int]]: At most two pairs ``(i, j)`` with
414            ``i < j``.
415
416        Examples:
417            >>> import simetri.graphics as sg
418            >>> sg.CircularGrid(n=6, radius=40).diagonal_index_pairs()
419            []
420            >>> sq = sg.SquareGrid(n=16, cell_size=25)
421            >>> sq.diagonal_index_pairs()
422            [(2, 10), (6, 14)]
423        """
424        if self.subtype != Types.SQUARE_GRID:
425            return []
426        points = self.points
427        if len(points) < 2:
428            return []
429        northeast = _extreme_vertex_index(points, True, True)
430        southwest = _extreme_vertex_index(points, True, False)
431        southeast = _extreme_vertex_index(points, False, True)
432        northwest = _extreme_vertex_index(points, False, False)
433        pairs: list[tuple[int, int]] = []
434        seen: set[tuple[int, int]] = set()
435        for first, second in (
436            (northeast, southwest),
437            (southeast, northwest),
438        ):
439            if first == second:
440                continue
441            if first > second:
442                first, second = second, first
443            key = (first, second)
444            if key in seen:
445                continue
446            seen.add(key)
447            pairs.append(key)
448        return pairs

450    def intersect(
451        self, line1: Sequence[int], line2: Sequence[int]
452    ) -> PointType:
453        """Return the intersection of two grid chords given by vertex indices.
454
455        Args:
456            line1: Two vertex indices ``(ind1, ind2)``.
457            line2: Two vertex indices ``(ind3, ind4)``.
458

471        ind3, ind4 = line2
472        points = self.points

476        return intersect(line1, line2)

517    def between(self, ind1: int, ind2: int, t: float = 0.5) -> PointType:
518        """
519        Returns the point on the line connecting the given indices interpolated
520        by using the given t parameter.
521
522        Args:
523            ind1 (int): The first index.
524            ind2 (int): The second index.
525            t (float): The parameter used for interpolation. Default is 0.5.
526
527        Returns:
528            PointType: The point on the line connecting the two points.
529
530        Examples:
531            >>> from simetri.base.all_enums import GridType
532            >>> from simetri.render.grids import Grid
533            >>> pts = [(0, 0), (40, 0), (40, 40), (0, 40)]
534            >>> g = Grid(GridType.SQUARE, points=pts, n=4)
535            >>> g.between(0, 1, 0.5)
536            (20.0, 0.0)
537        """
538        if t < 0 or t > 1:
539            raise ValueError("t must be between 0 and 1.")
540        if ind1 < 0 or ind1 >= len(self.points):
541            raise ValueError(
542                f"ind1 must be between 0 and {len(self.points) - 1}."
543            )
544        if ind1 == ind2:
545            raise ValueError("ind1 and ind2 must be different.")
546
547        return lerp_point(self.points[ind1], self.points[ind2], t)
