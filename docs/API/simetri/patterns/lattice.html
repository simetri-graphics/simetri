2D lattices and wallpaper-group isometry helpers.

Provides Lattice / Isometry plus factory functions for the seventeen wallpaper groups (lattice_p1, lattice_p6m, …). Motifs are typically Shape / Group instances from simetri.graphics.

>>> import simetri.graphics as sg

>>> _ = lat.populate_unit(sg.Circle(40, (40, 40)))

  47def basis_to_cart(
  48    a_: float,
  49    b_: float,
  50    u: PointType,
  51    v: PointType,
  52) -> tuple[float, float]:
  53    """Convert lattice basis coordinates ``(a_, b_)`` to Cartesian ``(x, y)``.
  54
  55    Args:
  56        a_: Coefficient along basis vector ``u``.
  57        b_: Coefficient along basis vector ``v``.
  58        u: First basis vector ``(ux, uy)``.
  59        v: Second basis vector ``(vx, vy)``.
  60
  61    Returns:
  62        tuple[float, float]: Cartesian point ``(x, y)``.
  63
  64    Examples:
  65        >>> basis_to_cart(1, 0, (40, 0), (0, 40))
  66        (40, 0)
  67    """
  68    ux, uy = u
  69    vx, vy = v
  70    x = a_ * ux + b_ * vx
  71    y = a_ * uy + b_ * vy
  72    return x, y

  75def cart_to_basis(
  76    x: float,
  77    y: float,
  78    u: PointType,
  79    v: PointType,
  80) -> tuple[float, float]:
  81    """Convert Cartesian ``(x, y)`` to lattice basis coordinates.
  82
  83    Args:
  84        x: Cartesian x-coordinate.
  85        y: Cartesian y-coordinate.
  86        u: First basis vector ``(ux, uy)``.
  87        v: Second basis vector ``(vx, vy)``.
  88
  89    Returns:
  90        tuple[float, float]: Basis coordinates ``(a, b)``.
  91
  92    Raises:
  93        ValueError: If ``u`` and ``v`` are linearly dependent.
  94
  95    Examples:
  96        >>> cart_to_basis(10, 20, (40, 0), (0, 40))
  97        (0.25, 0.5)
  98        >>> cart_to_basis(1, 1, (40, 0), (80, 0))  # doctest: +IGNORE_EXCEPTION_DETAIL
  99        Traceback (most recent call last):
 100        ValueError: Basis vectors are linearly dependent.
 101    """
 102    ux, uy = u
 103    vx, vy = v
 104    det = ux * vy - uy * vx
 105    if abs(det) < 1e-12:
 106        raise ValueError("Basis vectors are linearly dependent.")
 107
 108    a = (x * vy - y * vx) / det
 109    b = (ux * y - uy * x) / det
 110    return a, b

 113def all_axes() -> list[tuple[int, int]]:
 114    """Return index pairs of unique lattice axes.
 115
 116    A segment is kept if it crosses a corner or both endpoints are corners.
 117
 118    Returns:
 119        list[tuple[int, int]]: Pairs of axis endpoint indices in ``0..15``.
 120
 121    Examples:
 122        >>> len(all_axes())
 123        84
 124    """
 125    corners = {0, 4, 8, 12}
 126
 127    return [
 128        (i, j)
 129        for i in range(16)
 130        for j in range(i + 1, 16)
 131        if (
 132            # 1. Is there a corner between them?
 133            # (Excluding the wrap-around side where i=0 and j > 12)
 134            (any(i < c < j for c in [4, 8, 12]) and not (i == 0 and j > 12))
 135            # 2. OR are both endpoints corners? (Covers the 4 main sides)
 136            or ({i, j} <= corners)
 137        )
 138    ]

 171class Lattice:
 172    """A 2D lattice of type PAR, RECT, RHOMB, SQR, or HEX.
 173
 174    Attributes:
 175        subtype: Lattice type (``LatType``).
 176        a: First lattice spacing.
 177        b: Second lattice spacing (equals ``a`` for HEX/SQR/RHOMB).
 178        theta: Angle between basis vectors.
 179        origin: Lattice origin in Cartesian coordinates.
 180        u: First basis vector.
 181        v: Second basis vector.
 182        unit: Unit-cell parallelogram as a ``Shape``.
 183        isometries: List of ``Isometry`` objects for the wallpaper group.
 184        pattern: Motif ``Group`` after ``populate_unit``.
 185
 186    Examples:
 187        >>> lat = Lattice(LatType.SQR, a=10)
 188        >>> lat.a
 189        10
 190    """
 191
 192    def __init__(
 193        self,
 194        subtype: LatType = LatType.HEX,
 195        a: float = 40,
 196        b: float | None = None,
 197        theta: float | None = None,
 198        origin: PointType = (0, 0),
 199        wp_group: str | None = None,
 200    ) -> None:
 201        """Create a 2D lattice.
 202
 203        Args:
 204            subtype: Lattice type. Defaults to ``LatType.HEX``.
 205            a: Primary spacing. Defaults to 40.
 206            b: Secondary spacing (RECT/PAR). Defaults to None.
 207            theta: Angle between basis vectors when not implied by type.
 208            origin: Cartesian origin. Defaults to ``(0, 0)``.
 209            wp_group: Wallpaper group. Optional. (p1, p2, pg, cmm, etc.)
 210
 211        Raises:
 212            ValueError: If the basis vectors are linearly dependent.
 213
 214        Examples:
 215            >>> lat = Lattice(LatType.RECT, a=30, b=20)
 216            >>> lat.b
 217            20
 218        """
 219        angles = {
 220            LatType.HEX: pi / 3,
 221            LatType.RECT: pi / 2,
 222            LatType.SQR: pi / 2,
 223        }
 224        self.type = Types.LATTICE
 225        self.subtype = subtype
 226        self.wp_group = wp_group
 227        self.a = a
 228        self.b = b
 229
 230        if subtype in (LatType.RECT, LatType.SQR, LatType.HEX):
 231            self.theta = angles[self.subtype]
 232        else:
 233            self.theta = theta
 234
 235        if subtype in (LatType.HEX, LatType.SQR, LatType.RHOMB):
 236            self.b = self.a
 237
 238        self.origin = (0, 0)
 239
 240        if subtype == LatType.RHOMB:
 241            angle1 = theta / 2
 242            angle2 = angle1
 243        else:
 244            angle1 = 0
 245            angle2 = self.theta
 246        self.ax = self.a * cos(angle1)
 247        self.ay = -self.a * sin(angle1)
 248        self.bx = self.b * cos(angle2)
 249        self.by = self.b * sin(angle2)
 250        self.u = Vector(self.ax, self.ay)
 251        self.v = Vector(self.bx, self.by)
 252        self.origin = origin[:2]
 253
 254        det = self.ax * self.by - self.ay * self.bx
 255
 256        if abs(det) < 1e-6:
 257            raise ValueError(
 258                "Lattice vectors are linearly dependent (not a 2D lattice)."
 259            )
 260
 261        # unit cell
 262        p1 = (0, 0)
 263        p2 = self.u[:2]
 264        p3 = (self.u + self.v)[:2]
 265        p4 = self.v[:2]
 266
 267        self.unit = Shape(
 268            [p1, p2, p3, p4],
 269            closed=True,
 270            fill=False,
 271        )
 272
 273        self.unit_cell = None
 274
 275        self.isometries = []
 276        self.pattern = None
 277        self.kernel = None

 279    def _resolve(self, reference: Any) -> Any:
 280        """Resolve a lattice reference to a numeric or geometric value.
 281
 282        Args:
 283            reference: ``(LatRef, quantifier)`` pair describing the reference.
 284
 285        Returns:
 286            Any: Resolved coordinate, axis pair, distance, or numeric value.
 287        """
 288        ref, quantifier = reference
 289        if is_number(ref):
 290            res = ref
 291        elif ref == LatRef.COORD:
 292            # t1, t2 = quantifier
 293            res = self.basis_to_cartesian(quantifier)
 294        elif ref == LatRef.LERP:
 295            i, t = quantifier
 296            res = lerp_point(*self.unit.edges[i], t)
 297        elif ref == LatRef.VERTEX:
 298            res = self.unit[quantifier]
 299        elif ref == LatRef.EDGE:
 300            res = self.edges[quantifier]
 301        elif ref == LatRef.AXIS:
 302            p1, p2 = quantifier[:2]
 303            p1 = self._resolve(p1)
 304            p2 = self._resolve(p2)
 305            res = (p1, p2)
 306        elif ref == LatRef.DISTANCE:
 307            p1, p2 = quantifier
 308            p1 = self._resolve(p1)
 309            p2 = self._resolve(p2)
 310            res = distance(p1, p2)
 311
 312        return res

 330    def apply(self, isometry: Isometry) -> Group | Shape | None:
 331        """Apply an isometry to ``self.pattern`` using lattice references.
 332
 333        Args:
 334            isometry: ``Isometry`` describing translation, rotation, mirror,
 335                glide, or identity.
 336
 337        Returns:
 338            Group | Shape | None: Transformed pattern (or ``self.pattern`` for
 339            identity).
 340
 341        Examples:
 342            >>> import simetri.graphics as sg
 343            >>> lat = lattice_p4(a=20)
 344            >>> _ = lat.populate_unit(sg.Circle(20, (20, 20)))
 345            >>> isom = Isometry(IsometryType.IDENTITY, reps=0)
 346            >>> lat.apply(isom) is lat.pattern
 347            True
 348        """
 349        reference = isometry.reference
 350        quantifier = isometry.quantifier
 351        reps = isometry.reps
 352        take = isometry.take
 353        if isometry.subtype == IsometryType.TRANSLATION:
 354            if reference is None:
 355                dx, dy = quantifier
 356                dx = self._resolve(dx)
 357                dy = self._resolve(dy)
 358
 359            return self.pattern.translate(dx, dy, reps=reps, take=take)
 360        elif isometry.subtype == IsometryType.ROTATION:
 361            about = self._resolve(reference)
 362            return self.pattern.rotate(
 363                angle=quantifier,
 364                about=about,
 365                reps=reps,
 366                take=take,
 367            )
 368        elif isometry.subtype == IsometryType.MIRROR:
 369            about = self._resolve(reference)
 370
 371            return self.pattern.mirror(about=about, reps=reps, take=take)
 372        elif isometry.subtype == IsometryType.GLIDE_REFLECTION:
 373            about = self._resolve(reference)
 374            distance = self._resolve(quantifier)
 375            return self.pattern.glide(
 376                glide_line=about,
 377                glide_dist=distance,
 378                reps=reps,
 379                take=take,
 380            )
 381        elif isometry.subtype == IsometryType.IDENTITY:
 382            return self.pattern

 384    def populate_unit(self, kernel: Shape | Group) -> Lattice:
 385        """Set ``pattern`` from ``kernel`` and apply configured isometries.
 386
 387        Args:
 388            kernel: Motif ``Shape`` or ``Group`` to place in the unit cell.
 389
 390        Returns:
 391            Lattice: ``self``, for chaining.
 392
 393        Raises:
 394            ValueError: If ``kernel`` is not a Shape or Group.
 395
 396        Examples:
 397            >>> import simetri.graphics as sg
 398            >>> lat = lattice_p4(a=20)
 399            >>> _ = lat.populate_unit(sg.Circle(20, (20, 20)))
 400            >>> len(lat.pattern)
 401            4
 402        """
 403        if kernel.subtype == "PATH2D" or kernel.type == "SHAPE":
 404            self.pattern = Group(kernel)
 405        elif kernel.type == "BATCH" or kernel.type == "GROUP":
 406            self.pattern = kernel
 407
 408        else:
 409            raise ValueError("kernel needs to be a Shape or Group object!")
 410
 411        for isom in self.isometries:
 412            self.apply(isom)
 413
 414        return self

 416    def span(
 417        self,
 418        kernel: Shape | Group,
 419        horizontal: bool = True,
 420        reps: int = 1,
 421    ) -> Group:
 422        """Populate the unit and translate the pattern along one axis.
 423
 424        Args:
 425            kernel: Motif ``Shape`` or ``Group``.
 426            horizontal: If True, span along ``a``; otherwise along ``b``.
 427            reps: Number of translations. Defaults to 1.
 428
 429        Returns:
 430            Group: Updated pattern (also stored on ``self.pattern``).
 431
 432        Examples:
 433            >>> import simetri.graphics as sg
 434            >>> lat = Lattice(LatType.SQR, a=40)
 435            >>> row = lat.span(sg.Shape([(0, 0), (40, 0)]), horizontal=True, reps=2)
 436            >>> len(row)
 437            3
 438        """
 439        self.populate_unit(kernel)
 440        pattern = self.pattern
 441        if horizontal:
 442            dx = self.a
 443            self.pattern = pattern.translate(dx, 0, reps=reps)
 444        else:
 445            dy = self.b
 446            self.pattern = pattern.translate(0, dy, reps=reps)
 447        return self.pattern

 449    def expand(
 450        self,
 451        kernel: Shape | Group,
 452        reps1: int = 1,
 453        reps2: int | None = None,
 454    ) -> "Lattice":
 455        """Replace ``pattern`` by expanding a fresh copy of ``kernel``.
 456
 457        Stores a clean copy on ``self.kernel`` and builds ``self.pattern`` from
 458        another copy, so the caller's motif is not shared with the pattern.
 459        A later ``expand`` replaces both ``kernel`` and ``pattern``.
 460
 461        Args:
 462            kernel: Motif ``Shape`` or ``Group``.
 463            reps1: Extra copies along the first lattice axis (``a`` / ``u``).
 464                Must be ``> 0``. Defaults to 1.
 465            reps2: Extra copies along the second lattice axis (``b`` / ``v``).
 466                If None, uses ``reps1``. Must be ``> 0``. Defaults to None.
 467
 468        Returns:
 469            Lattice: ``self``, for chaining.
 470
 471        Raises:
 472            ValueError: If ``reps1`` or ``reps2`` is not greater than 0.
 473
 474        Examples:
 475            >>> import simetri.graphics as sg
 476            >>> lat = Lattice(LatType.SQR, a=10)
 477            >>> _ = lat.expand(sg.Shape([(0, 0), (40, 0)]), reps1=1, reps2=1)
 478            >>> lat.pattern is not None
 479            True
 480        """
 481        if reps2 is None:
 482            reps2 = reps1
 483        if reps1 <= 0:
 484            raise ValueError("expand requires reps1 > 0")
 485        if reps2 <= 0:
 486            raise ValueError("expand requires reps2 > 0")
 487
 488        self.kernel = kernel.copy()
 489        self.populate_unit(self.kernel.copy())
 490        pattern = self.pattern
 491        subtype = self.subtype
 492        if subtype in (LatType.HEX, LatType.PAR):
 493            pattern.translate(dx=self.bx, dy=self.by, reps=1)
 494            pattern.translate(dx=0, dy=2 * self.by, reps=reps2).translate(
 495                dx=self.a, reps=reps1
 496            )
 497
 498        elif subtype in [LatType.SQR, LatType.RECT]:
 499            pattern.translate(self.a, reps=reps1).translate(
 500                0, self.b, reps=reps2
 501            )
 502
 503        elif subtype == LatType.RHOMB:
 504            width = self.unit.width
 505            height = self.unit.height
 506            pattern.translate(dx=self.bx, dy=self.by, reps=1).translate(
 507                dx=0, dy=height, reps=reps2
 508            ).translate(dx=width, reps=reps1)
 509
 510        return self

 561    def clipped_points(
 562        self, lower_left: PointType, upper_right: PointType
 563    ) -> list:
 564        """Return lattice points that lie inside the given axis-aligned rectangle.
 565
 566        Args:
 567            lower_left: Lower-left corner of the clip rectangle.
 568            upper_right: Upper-right corner of the clip rectangle.
 569
 570        Returns:
 571            list: Lattice points inside the rectangle.
 572
 573        Examples:
 574            >>> lat = Lattice(LatType.SQR, a=10)
 575            >>> len(lat.clipped_points((0, 0), (25, 25)))
 576            9
 577        """
 578        # Define the 4 corners of the rectangle
 579        x_min, y_min = lower_left[:2]
 580        x_max, y_max = upper_right[:2]
 581        ox, oy = self.origin[:2]
 582
 583        corners = [
 584            (x_min, y_min),
 585            (x_max, y_min),
 586            (x_min, y_max),
 587            (x_max, y_max),
 588        ]
 589
 590        # Transform corners to (i, j) space
 591        # Formula: [i, j]^T = M^-1 * ([x, y]^T - [ox, oy]^T)
 592        i_vals = []
 593        j_vals = []
 594        ux, uy = self.u
 595        vx, vy = self.v
 596        det = ux * vy - uy * vx
 597        for px, py in corners:
 598            dx, dy = px - ox, py - oy
 599            # Inverse matrix multiplication
 600            i = (dx * vy - dy * vx) / det
 601            j = (-dx * uy + dy * ux) / det
 602            i_vals.append(i)
 603            j_vals.append(j)
 604
 605        # Determine integer bounds for i and j
 606        i_start = floor(min(i_vals))
 607        i_end = ceil(max(i_vals))
 608        j_start = floor(min(j_vals))
 609        j_end = ceil(max(j_vals))
 610
 611        # 5. Collect points and perform a final check
 612        # (The i,j range covers the parallelogram bounding the rectangle)
 613        points_inside = []
 614        for i in range(i_start, i_end + 1):
 615            for j in range(j_start, j_end + 1):
 616                px = ox + i * ux + j * vx
 617                py = oy + i * uy + j * vy
 618
 619                # Final check against the axis-aligned rectangle
 620                if x_min <= px <= x_max and y_min <= py <= y_max:
 621                    points_inside.append((px, py))
 622
 623        return points_inside

 647        vx, vy = self.v
 648        corners = np.array(
 649            [[x_min, y_min], [x_max, y_min], [x_min, y_max], [x_max, y_max]]
 650        )
 651
 652        # Solve for lattice coordinates (c_u, c_v) for each corner
 653        # Corner = Origin + c_u*u + c_v*v
 654        # [ux vx] [c_u] = [Corner_x - Origin_x]
 655        # [uy vy] [c_v]   [Corner_y - Origin_y]
 656        matrix = np.array([[ux, vx], [uy, vy]])
 657        inv_matrix = np.linalg.inv(matrix)
 658
 659        diffs = corners - np.array(origin)
 660        coords = diffs @ inv_matrix.T
 661
 662        # Get range of indices
 663        min_u, max_u = np.floor(coords[:, 0].min()), np.ceil(coords[:, 0].max())
 664        min_v, max_v = np.floor(coords[:, 1].min()), np.ceil(coords[:, 1].max())
 665
 666        clipped_lines = []
 667
 668        # Lines in direction v (indexed by u)
 669        u = self.u
 670        v = self.v
 671        for i in range(int(min_u), int(max_u) + 1):
 672            point = np.array(origin) + i * np.array(u)
 673            line = clip_line_to_rect(point, v, lower_left, upper_right)
 674            if line:
 675                clipped_lines.append(line)
 676
 677        # Lines in direction u (indexed by v)
 678        for j in range(int(min_v), int(max_v) + 1):
 679            point = np.array(origin) + j * np.array(v)
 680            line = clip_line_to_rect(point, u, lower_left, upper_right)
 681            if line:
 682                clipped_lines.append(line)
 683
 684        return clipped_lines

280    def _resolve(self, reference: Any) -> Any:
281        """Resolve a lattice reference to a numeric or geometric value.
282
283        Args:
284            reference: ``(LatRef, quantifier)`` pair describing the reference.
285
286        Returns:
287            Any: Resolved coordinate, axis pair, distance, or numeric value.
288        """
289        ref, quantifier = reference
290        if is_number(ref):
291            res = ref
292        elif ref == LatRef.COORD:
293            # t1, t2 = quantifier
294            res = self.basis_to_cartesian(quantifier)
295        elif ref == LatRef.LERP:
296            i, t = quantifier
297            res = lerp_point(*self.unit.edges[i], t)
298        elif ref == LatRef.VERTEX:
299            res = self.unit[quantifier]
300        elif ref == LatRef.EDGE:
301            res = self.edges[quantifier]
302        elif ref == LatRef.AXIS:
303            p1, p2 = quantifier[:2]
304            p1 = self._resolve(p1)
305            p2 = self._resolve(p2)
306            res = (p1, p2)
307        elif ref == LatRef.DISTANCE:
308            p1, p2 = quantifier
309            p1 = self._resolve(p1)
310            p2 = self._resolve(p2)
311            res = distance(p1, p2)
312

562    def clipped_points(
563        self, lower_left: PointType, upper_right: PointType
564    ) -> list:
565        """Return lattice points that lie inside the given axis-aligned rectangle.
566
567        Args:
568            lower_left: Lower-left corner of the clip rectangle.
569            upper_right: Upper-right corner of the clip rectangle.
570
571        Returns:
572            list: Lattice points inside the rectangle.
573
574        Examples:
575            >>> lat = Lattice(LatType.SQR, a=10)
576            >>> len(lat.clipped_points((0, 0), (25, 25)))
577            9
578        """
579        # Define the 4 corners of the rectangle
580        x_min, y_min = lower_left[:2]
581        x_max, y_max = upper_right[:2]
582        ox, oy = self.origin[:2]
583
584        corners = [
585            (x_min, y_min),
586            (x_max, y_min),
587            (x_min, y_max),
588            (x_max, y_max),
589        ]
590
591        # Transform corners to (i, j) space
592        # Formula: [i, j]^T = M^-1 * ([x, y]^T - [ox, oy]^T)
593        i_vals = []
594        j_vals = []
595        ux, uy = self.u
596        vx, vy = self.v
597        det = ux * vy - uy * vx
598        for px, py in corners:
599            dx, dy = px - ox, py - oy
600            # Inverse matrix multiplication
601            i = (dx * vy - dy * vx) / det
602            j = (-dx * uy + dy * ux) / det
603            i_vals.append(i)
604            j_vals.append(j)
605
606        # Determine integer bounds for i and j
607        i_start = floor(min(i_vals))
608        i_end = ceil(max(i_vals))
609        j_start = floor(min(j_vals))
610        j_end = ceil(max(j_vals))
611
612        # 5. Collect points and perform a final check
613        # (The i,j range covers the parallelogram bounding the rectangle)
614        points_inside = []
615        for i in range(i_start, i_end + 1):
616            for j in range(j_start, j_end + 1):
617                px = ox + i * ux + j * vx
618                py = oy + i * uy + j * vy
619
620                # Final check against the axis-aligned rectangle
621                if x_min <= px <= x_max and y_min <= py <= y_max:
622                    points_inside.append((px, py))
623
624        return points_inside

627        self, lower_left: PointType, upper_right: PointType
628    ) -> list:
629        """Return lattice lines clipped to the given axis-aligned rectangle.
630
631        Args:
632            lower_left: Lower-left corner of the clip rectangle.
633            upper_right: Upper-right corner of the clip rectangle.
634
635        Returns:
636            list: Line segments inside the rectangle.
637
638        Examples:
639            >>> lat = Lattice(LatType.SQR, a=10)
640            >>> len(lat.clipped_lines((0, 0), (25, 25))) > 0
641            True
642        """
643        # Define the 4 corners of the rectangle
644        x_min, y_min = lower_left[:2]
645        x_max, y_max = upper_right[:2]
646        origin = self.origin
647        ux, uy = self.u
648        vx, vy = self.v
649        corners = np.array(
650            [[x_min, y_min], [x_max, y_min], [x_min, y_max], [x_max, y_max]]
651        )
652
653        # Solve for lattice coordinates (c_u, c_v) for each corner
654        # Corner = Origin + c_u*u + c_v*v
655        # [ux vx] [c_u] = [Corner_x - Origin_x]
656        # [uy vy] [c_v]   [Corner_y - Origin_y]
657        matrix = np.array([[ux, vx], [uy, vy]])
658        inv_matrix = np.linalg.inv(matrix)
659
660        diffs = corners - np.array(origin)
661        coords = diffs @ inv_matrix.T
662
663        # Get range of indices
664        min_u, max_u = np.floor(coords[:, 0].min()), np.ceil(coords[:, 0].max())
665        min_v, max_v = np.floor(coords[:, 1].min()), np.ceil(coords[:, 1].max())
666
667        clipped_lines = []
668
669        # Lines in direction v (indexed by u)
670        u = self.u
671        v = self.v
672        for i in range(int(min_u), int(max_u) + 1):
673            point = np.array(origin) + i * np.array(u)
674            line = clip_line_to_rect(point, v, lower_left, upper_right)
675            if line:
676                clipped_lines.append(line)
677
678        # Lines in direction u (indexed by v)
679        for j in range(int(min_v), int(max_v) + 1):
680            point = np.array(origin) + j * np.array(v)
681            line = clip_line_to_rect(point, u, lower_left, upper_right)
682            if line:
683                clipped_lines.append(line)
684
685        return clipped_lines

581        x_max, y_max = upper_right[:2]

583
