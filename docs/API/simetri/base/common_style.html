simetri.base.common_style

Common owned-style attributes for drawable objects.

CommonStyle is a mixin for classes that store stroke/fill/marker style on self. Used by Shape, Path2D, and Pattern; other drawables can opt in the same way.

152def coerce_style_overlay(
153    mapping: Any = None, kwargs: dict | None = None
154) -> dict[str, Any]:
155    """Return a dict of draw-alias keys from a Style, a dict, and/or kwargs.
156
157    Keyword arguments overwrite keys from ``mapping``. Unknown keys raise.
158
159    Examples:
160        >>> from simetri.base.common_style import Style, coerce_style_overlay
161        >>> coerce_style_overlay({"line_width": 2.0})
162        {'line_width': 2.0}
163        >>> coerce_style_overlay(Style(stroke=True), {"line_width": 1.0})
164        {'stroke': True, 'line_width': 1.0}
165    """
166    if kwargs is None:
167        kwargs = {}
168    overlay: dict[str, Any] = {}
169    if mapping is not None:
170        if isinstance(mapping, Style):
171            source_items = mapping.items()
172        elif isinstance(mapping, dict):
173            source_items = mapping.items()
174        else:
175            raise TypeError(
176                "Style overlay must be a Style, a dict, or keyword arguments, "
177                f"got {type(mapping).__name__}"
178            )
179        overlay = dict(source_items)
180    overlay.update(kwargs)
181    for key in overlay:
182        if key not in STYLE_ALIAS_KEYS:
183            raise ValueError(f"Unknown style key {key!r}")
184    return overlay

187class CommonStyle:
188    """Mixin: color/alpha properties and ``copy_style``.
189
190    Subclasses must:
191
192    - Include the six private color/alpha names if they use ``__slots__``.
193    - Call ``_init_from_style_kwargs(kwargs)`` from ``__init__`` (or the
194      lower-level helpers) after handling non-style constructor args.
195    - Set ``_style_warning_stacklevel`` if ``__setattr__`` adds a frame
196      (e.g. Path2D under ``Group`` uses ``4``; default is ``3``).
197
198    ``color`` / ``alpha`` fan out to line and fill when set to a non-``None``
199    value. Omitted line/fill color and alpha, and omitted
200    ``STYLE_COPY_ATTRS`` fields, are stored from ``user_defaults`` at init.
201
202    Examples:
203        >>> import simetri.graphics as sg
204        >>> mark = sg.Shape([(0, 0), (100, 0)], closed=True)
205        >>> mark.style["line_width"] == sg.user_defaults["line_width"]
206        True
207        >>> mark.reset_style() is mark
208        True
209    """
210
211    _style_warning_stacklevel: int = 3
212
213    def _init_color_alpha_state(self) -> None:
214        """Set raw color/alpha fields to unset (``None``)."""
215        self._alpha = None
216        self._color = None
217        self._line_alpha = None
218        self._fill_alpha = None
219        self._line_color = None
220        self._fill_color = None
221
222    def _apply_color_alpha(
223        self,
224        *,
225        color: Color | None = None,
226        alpha: float | None = None,
227        line_color: Color | None = None,
228        fill_color: Color | None = None,
229        line_alpha: float | None = None,
230        fill_alpha: float | None = None,
231    ) -> None:
232        """Apply constructor color/alpha args in Shape order.
233
234        ``color`` / ``alpha`` fan out to line and fill without warning;
235        the property setters still warn on later assignment.
236        Explicit line/fill args override after fan-out.
237        """
238        if color is not None:
239            self._color = color
240            self._line_color = color
241            self._fill_color = color
242        if alpha is not None:
243            self._alpha = alpha
244            self._line_alpha = alpha
245            self._fill_alpha = alpha
246        if line_color is not None:
247            self._line_color = line_color
248        if fill_color is not None:
249            self._fill_color = fill_color
250        if line_alpha is not None:
251            self._line_alpha = line_alpha
252        if fill_alpha is not None:
253            self._fill_alpha = fill_alpha
254
255    def _init_style_copy_attrs(self) -> None:
256        """Set every ``STYLE_COPY_ATTRS`` field to ``None``."""
257        for name in STYLE_COPY_ATTRS:
258            setattr(self, name, None)
259
260    def _init_from_style_kwargs(self, kwargs: dict[str, Any]) -> None:
261        """Initialize owned style from ``kwargs`` (mutated).
262
263        Pops known color/alpha and ``STYLE_COPY_ATTRS`` keys from ``kwargs``.
264        A missing key or a ``None`` value is stored from ``user_defaults``.
265        ``color`` and ``alpha`` stay ``None`` unless passed.
266        Leaves any non-style keys for the caller. Typical subclass ``__init__``:
267
268            def __init__(self, points=None, **kwargs):
269                ...  # geometry / identity setup ...
270                self._init_from_style_kwargs(kwargs)
271                if kwargs:
272                    raise TypeError(
273                        f"Unexpected keyword arguments: {sorted(kwargs)}"
274                    )
275        """
276        self._init_style_copy_attrs()
277        self._init_color_alpha_state()
278
279        color = kwargs.pop("color") if "color" in kwargs else None
280        alpha = kwargs.pop("alpha") if "alpha" in kwargs else None
281        line_color = kwargs.pop("line_color") if "line_color" in kwargs else None
282        fill_color = kwargs.pop("fill_color") if "fill_color" in kwargs else None
283        line_alpha = kwargs.pop("line_alpha") if "line_alpha" in kwargs else None
284        fill_alpha = kwargs.pop("fill_alpha") if "fill_alpha" in kwargs else None
285        self._apply_color_alpha(
286            color=color,
287            alpha=alpha,
288            line_color=line_color,
289            fill_color=fill_color,
290            line_alpha=line_alpha,
291            fill_alpha=fill_alpha,
292        )
293        if self._line_color is None:
294            self._line_color = user_defaults["line_color"]
295        if self._fill_color is None:
296            self._fill_color = user_defaults["fill_color"]
297        if self._line_alpha is None:
298            self._line_alpha = user_defaults["line_alpha"]
299        if self._fill_alpha is None:
300            self._fill_alpha = user_defaults["fill_alpha"]
301
302        for name in STYLE_COPY_ATTRS:
303            if name in kwargs and kwargs[name] is not None:
304                setattr(self, name, kwargs.pop(name))
305            else:
306                if name in kwargs:
307                    kwargs.pop(name)
308                setattr(self, name, user_defaults[name])
309

388    def copy_style(self, other: Any) -> Self:
389        """Copy style from ``other`` onto this object (mutated).
390
391        Copies raw color/alpha privates (no setter fan-out), then each
392        name in ``STYLE_COPY_ATTRS``. ``other`` must own the same fields.
393        """
394        self._alpha = other._alpha
395        self._color = other._color
396        self._line_alpha = other._line_alpha
397        self._fill_alpha = other._fill_alpha
398        self._line_color = other._line_color
399        self._fill_color = other._fill_color
400
401        for name in STYLE_COPY_ATTRS:
402            setattr(self, name, getattr(other, name))
403
404        return self

407        """Return the stored style value (``None`` if unset), not the default."""

261    def _init_from_style_kwargs(self, kwargs: dict[str, Any]) -> None:
262        """Initialize owned style from ``kwargs`` (mutated).
263
264        Pops known color/alpha and ``STYLE_COPY_ATTRS`` keys from ``kwargs``.
265        A missing key or a ``None`` value is stored from ``user_defaults``.
266        ``color`` and ``alpha`` stay ``None`` unless passed.
267        Leaves any non-style keys for the caller. Typical subclass ``__init__``:
268
269            def __init__(self, points=None, **kwargs):
270                ...  # geometry / identity setup ...
271                self._init_from_style_kwargs(kwargs)
272                if kwargs:
273                    raise TypeError(
274                        f"Unexpected keyword arguments: {sorted(kwargs)}"
275                    )
276        """
277        self._init_style_copy_attrs()
278        self._init_color_alpha_state()
279
280        color = kwargs.pop("color") if "color" in kwargs else None
281        alpha = kwargs.pop("alpha") if "alpha" in kwargs else None
282        line_color = kwargs.pop("line_color") if "line_color" in kwargs else None
283        fill_color = kwargs.pop("fill_color") if "fill_color" in kwargs else None
284        line_alpha = kwargs.pop("line_alpha") if "line_alpha" in kwargs else None
285        fill_alpha = kwargs.pop("fill_alpha") if "fill_alpha" in kwargs else None
286        self._apply_color_alpha(
287            color=color,
288            alpha=alpha,
289            line_color=line_color,
290            fill_color=fill_color,
291            line_alpha=line_alpha,
292            fill_alpha=fill_alpha,
293        )
294        if self._line_color is None:
295            self._line_color = user_defaults["line_color"]
296        if self._fill_color is None:
297            self._fill_color = user_defaults["fill_color"]
298        if self._line_alpha is None:
299            self._line_alpha = user_defaults["line_alpha"]
300        if self._fill_alpha is None:
301            self._fill_alpha = user_defaults["fill_alpha"]
302
303        for name in STYLE_COPY_ATTRS:
304            if name in kwargs and kwargs[name] is not None:
305                setattr(self, name, kwargs.pop(name))
306            else:
307                if name in kwargs:
308                    kwargs.pop(name)
309                setattr(self, name, user_defaults[name])

Subclasses must:
