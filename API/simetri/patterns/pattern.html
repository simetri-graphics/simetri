simetri.patterns.pattern

Repeated geometric patterns built from a kernel and transforms.

A Pattern stores a kernel Shape/Group plus a PatternTransformation (list of TransformMat matrices with repetitions). Calling transform helpers such as translate / rotate appends transforms rather than baking them into the kernel.

Examples:
>>> kernel = sg.Shape([(0, 0), (40, 0), (40, 40), (0, 40)], closed=True)
>>> pattern = Pattern(kernel)
>>> _ = pattern.rotate(1.0471975511965976, about=(0, 0), reps=2)
>>> pattern.count
3

 110    def _changed(self) -> bool:
 111        """
 112        Checks if the transformation matrix or reps value has changed.
 113

 115            bool: True if the transformation state has changed, False otherwise.
 116
 117        Examples:
 118            >>> step = TransformMat(identity_matrix(), reps=0)
 119            >>> step._changed()
 120            False
 121        """
 122        return not (
 123            (self.hash == md5(self.xform_matrix.tobytes()).hexdigest())
 124            and (self.reps == self._reps)
 125        )

 127    def _set_partitions(self) -> None:
 128        if self.reps == 0:
 129            partition_list = [identity_matrix()]
 130        elif self.reps == 1:
 131            partition_list = [identity_matrix(), self.xform_matrix]
 132        else:
 133            xform_mat = self.xform_matrix
 134            partition_list = [identity_matrix(), xform_mat]
 135            last = xform_mat
 136            for _ in range(self.reps - 1):
 137                last = xform_mat @ last
 138                partition_list.append(last)
 139
 140        self._partitions = partition_list

 245class PatternTransformation:
 246    """Ordered list of ``TransformMat`` components forming a pattern.
 247
 248    Attributes:
 249        components: List of ``TransformMat`` instances applied in order.
 250        type: Always ``Types.TRANSFORMATION``.
 251
 252    Examples:
 253        >>> stack = PatternTransformation(
 254        ...     [TransformMat(translation_matrix(10, 0), reps=1)]
 255        ... )
 256        >>> stack.count
 257        2
 258    """
 259
 260    components: list[TransformMat] = None
 261
 262    def __post_init__(self) -> None:
 263        self.type = Types.TRANSFORMATION
 264        self.subtype = Types.TRANSFORMATION
 265        if self.components is None:
 266            self.components = []
 267
 268    def __repr__(self) -> str:
 269        return f"PatternTransformation(components={self.components})"
 270
 271    def __str__(self) -> str:
 272        return f"PatternTransformation(components={self.components})"

 274    def apply(self, kernel: Shape) -> Group:
 275        """Apply the composite transform to ``kernel`` and return copies.
 276
 277        Args:
 278            kernel: Source shape whose ``final_coords`` are transformed.
 279
 280        Returns:
 281            Group: One shape per transform partition, with kernel style copied.
 282
 283        Examples:
 284            >>> kernel = sg.Shape([(0, 0), (20, 0), (20, 20)], closed=True)
 285            >>> stack = PatternTransformation(
 286            ...     [TransformMat(translation_matrix(10, 0), reps=1)]
 287            ... )
 288            >>> expanded = stack.apply(kernel)
 289            >>> len(expanded)
 290            2
 291        """
 292        all_vertices = kernel.final_coords @ self.composite
 293        vertices_list = np.hsplit(all_vertices, self.count)
 294        res = Group()
 295        for vertices in vertices_list:
 296            shape = Shape(vertices)
 297            shape.copy_style(kernel)
 298            res.append(shape)
 299
 300        return res
 301
 302    @property
 303    def count(self) -> int:
 304        """Return the number of shapes produced by this transformation stack.
 305
 306        Returns:
 307            int: Product of ``(reps + 1)`` over all components.
 308
 309        Examples:
 310            >>> stack = PatternTransformation(
 311            ...     [
 312            ...         TransformMat(translation_matrix(1, 0), reps=1),
 313            ...         TransformMat(translation_matrix(0, 1), reps=1),
 314            ...     ]
 315            ... )
 316            >>> stack.count
 317            4
 318        """
 319        return prod([comp.reps + 1 for comp in self.components])
 320
 321    @property
 322    def partitions(self) -> list:
 323        """
 324        Returns the submatrices in the transformation.
 325
 326        Returns:
 327            list of ndarrays.
 328
 329        Examples:
 330            >>> stack = PatternTransformation()
 331            >>> len(stack.partitions)
 332            1
 333        """
 334        if len(self.components) == 0:
 335            return [identity_matrix()]
 336        elif len(self.components) == 1:
 337            partitions = [identity_matrix(), self.components[0].xform_matrix]
 338        else:
 339            partitions = []
 340            for component in self.components:
 341                partitions.extend(component.partitions)
 342
 343        return partitions
 344
 345    @property
 346    def composite(self) -> NDArray:
 347        """
 348        Returns the compound transformation matrix.
 349
 350        Returns:
 351            ndarray: The compound transformation matrix.
 352
 353        Examples:
 354            >>> stack = PatternTransformation(
 355            ...     [TransformMat(translation_matrix(40, 0), reps=0)]
 356            ... )
 357            >>> stack.composite.shape
 358            (3, 3)
 359        """
 360        if len(self.components) == 0:
 361            return identity_matrix()
 362        matrices = [component.partitions for component in self.components]
 363        res = []
 364        if len(matrices) == 1:
 365            if len(matrices[0]) == 1:
 366                return matrices[0][0]
 367            else:
 368                return np.concatenate(matrices[0], axis=1)
 369        else:
 370            res.extend(np.linalg.multi_dot(mats) for mats in product(*matrices))
 371
 372        return np.concatenate(res, axis=1)
 373
 374    def copy(self) -> "PatternTransformation":
 375        """
 376        Creates a copy of the PatternTransformation instance.
 377
 378        Returns:
 379            PatternTransformation: A new PatternTransformation with the same
 380            components.
 381
 382        Examples:
 383            >>> stack = PatternTransformation(
 384            ...     [TransformMat(translation_matrix(1, 0), reps=1)]
 385            ... )
 386            >>> copy = stack.copy()
 387            >>> copy.count
 388            2
 389            >>> copy is stack
 390            False
 391        """
 392        return PatternTransformation(
 393            [component.copy() for component in self.components]
 394        )

 964            take = t_def.take
 965            reps = t_def.reps

 968                dx, dy = self.resolve_tuple(t_def.args, kernel, pattern)

 971                pivot = self.resolve_reference(t_def.ref, kernel, pattern)
 972                angle = t_def.args
 973                pattern.rotate(angle, pivot, take=take, reps=reps, incr=incr)

 975                about = self.resolve_reference(t_def.ref, kernel, pattern)
 976                pattern.mirror(about, take=take, reps=reps, incr=incr)
 977            elif t_def.type == TransformationType.GLIDE:
 978                about = self.resolve_reference(t_def.ref, kernel, pattern)
 979                dist = self.resolve_value(t_def.args, kernel, pattern)
 980                pattern.glide(about, dist, take=take, reps=reps, incr=incr)
