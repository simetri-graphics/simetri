simetri.geom.bbox

Axis-aligned bounding boxes for Shape and Group objects.

Provides reference anchors (corners, mid-sides, diagonals) and placement helpers such as left_of, above, and polar_pos.

Examples

import simetri.graphics as sg
box = sg.BoundingBox((0, 0), (100, 50))
box.width, box.height
# (100, 50)
box.midpoint
# (50.0, 25.0)

 166    def angle_point(self, angle: float) -> PointType:
 167        """Return where a ray from the midpoint hits the box boundary.
 168
 169        Args:
 170            angle: Ray direction in radians (from positive x-axis).
 171
 172        Returns:
 173            PointType: Intersection of the ray with the bounding-box edge.
 174
 175        Examples:
 176            >>> from simetri.geom.bbox import BoundingBox
 177            >>> bb = BoundingBox((0, 0), (40, 80))
 178            >>> tuple(round(v, 10) for v in bb.angle_point(0))
 179            (40.0, 40.0)
 180        """
 181        angle = positive_angle(angle)
 182        direction_x = np.cos(angle)
 183        direction_y = np.sin(angle)
 184        midpoint_x, midpoint_y = self.midpoint[:2]
 185        southwest_x, southwest_y = self.southwest[:2]
 186        northeast_x, northeast_y = self.northeast[:2]
 187
 188        distances = []
 189        if direction_x > 0:
 190            distances.append((northeast_x - midpoint_x) / direction_x)
 191        elif direction_x < 0:
 192            distances.append((southwest_x - midpoint_x) / direction_x)
 193        if direction_y > 0:
 194            distances.append((northeast_y - midpoint_y) / direction_y)
 195        elif direction_y < 0:
 196            distances.append((southwest_y - midpoint_y) / direction_y)
 197
 198        distance_to_edge = min(distances)
 199        return (
 200            midpoint_x + distance_to_edge * direction_x,
 201            midpoint_y + distance_to_edge * direction_y,
 202        )

 621    def get_inflated_b_box(
 622        self,

 628        """
 629        Return a bounding box with offset edges.
 630
 631        Args:
 632            left_margin (float, optional): The left margin.
 633            bottom_margin (float, optional): The bottom margin.
 634            right_margin (float, optional): The right margin.
 635            top_margin (float, optional): The top margin.

 638            BoundingBox: The inflated bounding box.
 639
 640        Examples:
 641            >>> import simetri.graphics as sg

 644            >>> inflated.southwest, inflated.northeast

 648        if bottom_margin is None:
 649            bottom_margin = left_margin
 650        if right_margin is None:
 651            right_margin = left_margin
 652        if top_margin is None:
 653            top_margin = bottom_margin
 654
 655        x, y = self.southwest[:2]
 656        southwest = (x - left_margin, y - bottom_margin)
 657
 658        x, y = self.northeast[:2]
 659        northeast = (x + right_margin, y + top_margin)
 660
 661        return BoundingBox(southwest, northeast)

 663    def offset_line(
 664        self, side: Side | str, offset: float
 665    ) -> tuple[PointType, PointType]:
 666        """
 667        Offset is applied outwards. Use negative values for inward offset.
 668
 669        Args:
 670            side (Side): The side to offset.
 671            offset (float): The offset distance.
 672
 673        Returns:
 674            tuple: The offset line.
 675
 676        Examples:
 677            >>> from simetri.base.all_enums import Side
 678            >>> import simetri.graphics as sg
 679            >>> bb = sg.BoundingBox((0, 0), (40, 80))
 680            >>> bb.offset_line(Side.LEFT, 40)
 681            ((-40, 0), (-40, 80))
 682        """
 683        if isinstance(side, str):
 684            side = Side[side.upper()]
 685
 686        if side == Side.RIGHT:
 687            x1, y1 = self.southeast
 688            x2, y2 = self.northeast
 689            res = ((x1 + offset, y1), (x2 + offset, y2))
 690        elif side == Side.LEFT:
 691            x1, y1 = self.southwest
 692            x2, y2 = self.northwest
 693            res = ((x1 - offset, y1), (x2 - offset, y2))
 694        elif side == Side.TOP:
 695            x1, y1 = self.northwest
 696            x2, y2 = self.northeast
 697            res = ((x1, y1 + offset), (x2, y2 + offset))
 698        elif side == Side.BOTTOM:
 699            x1, y1 = self.southwest
 700            x2, y2 = self.southeast
 701            res = ((x1, y1 - offset), (x2, y2 - offset))
 702        elif side == Side.DIAGONAL1:
 703            res = offset_line(self.diagonal1, offset)
 704        elif side == Side.DIAGONAL2:
 705            res = offset_line(self.diagonal2, offset)
 706        elif side == Side.H_CENTERLINE:
 707            res = offset_line(self.horiz_centerline, offset)
 708        elif side == Side.V_CENTERLINE:
 709            res = offset_line(self.vert_centerline, offset)
 710        else:
 711            raise ValueError(f"Unknown side: {side}")
 712
 713        return res

1028def bounding_box(points: Sequence[PointType]) -> BoundingBox:
1029    """Build a ``BoundingBox`` from a sequence of points.
1030
1031    Args:
1032        points: Sequence or ndarray of ``(x, y)`` points.
1033
1034    Returns:
1035        BoundingBox: Axis-aligned box enclosing the points.
1036
1037    Raises:
1038        ValueError: If ``points`` is empty.
1039
1040    Examples:
1041        >>> import simetri.graphics as sg
1042        >>> bb = sg.bounding_box([(0, 0), (80, 40), (24, 64)])
1043        >>> bb.southwest, bb.northeast
1044        ((0, 0), (80, 64))
1045    """
1046    if isinstance(points, np.ndarray):
1047        points = points[:, :2]
1048    else:
1049        points = np.array(points)  # numpy array of points
1050    n_points = len(points)
1051    BB_EPSILON = runtime_defaults["BB_EPSILON"]
1052    if n_points == 0:  # empty list of points
1053        raise ValueError("Empty list of points")
1054
1055    if len(points.shape) == 1:
1056        # single point
1057        min_x, min_y = points
1058        max_x = min_x + BB_EPSILON
1059        max_y = min_y + BB_EPSILON
1060    else:
1061        # find minimum and maximum coordinates
1062        min_x, min_y = points.min(axis=0)
1063        max_x, max_y = points.max(axis=0)
1064        if min_x == max_x:  # this could be a vertical line or degenerate points
1065            max_x += BB_EPSILON
1066        if (
1067            min_y == max_y
1068        ):  # this could be a horizontal line or degenerate points
1069            max_y += BB_EPSILON
1070    # bounding box corners
1071    bottom_left = (min_x, min_y)
1072    top_right = (max_x, max_y)
1073    return BoundingBox(southwest=bottom_left, northeast=top_right)

 167    def angle_point(self, angle: float) -> PointType:
 168        """Return where a ray from the midpoint hits the box boundary.
 169
 170        Args:
 171            angle: Ray direction in radians (from positive x-axis).
 172
 173        Returns:
 174            PointType: Intersection of the ray with the bounding-box edge.
 175
 176        Examples:
 177            >>> from simetri.geom.bbox import BoundingBox
 178            >>> bb = BoundingBox((0, 0), (40, 80))
 179            >>> tuple(round(v, 10) for v in bb.angle_point(0))
 180            (40.0, 40.0)
 181        """
 182        angle = positive_angle(angle)
 183        direction_x = np.cos(angle)
 184        direction_y = np.sin(angle)
 185        midpoint_x, midpoint_y = self.midpoint[:2]
 186        southwest_x, southwest_y = self.southwest[:2]
 187        northeast_x, northeast_y = self.northeast[:2]
 188
 189        distances = []
 190        if direction_x > 0:
 191            distances.append((northeast_x - midpoint_x) / direction_x)
 192        elif direction_x < 0:
 193            distances.append((southwest_x - midpoint_x) / direction_x)
 194        if direction_y > 0:
 195            distances.append((northeast_y - midpoint_y) / direction_y)
 196        elif direction_y < 0:
 197            distances.append((southwest_y - midpoint_y) / direction_y)
 198
 199        distance_to_edge = min(distances)
 200        return (
 201            midpoint_x + distance_to_edge * direction_x,
 202            midpoint_y + distance_to_edge * direction_y,
 203        )
