 317def get_pdf_dimensions(pdf_path: str) -> tuple[float, float] | None:
 318    """Return width and height in points for the first PDF page.
 319
 320    Args:
 321        pdf_path: Path to the PDF file.
 322
 323    Returns:
 324        ``(width, height)`` in points, or ``None`` on error.
 325
 326    Examples:
 327        >>> import contextlib
 328        >>> import io
 329        >>> import simetri.graphics as sg
 330        >>> buf = io.StringIO()
 331        >>> with contextlib.redirect_stdout(buf):
 332        ...     result = sg.get_pdf_dimensions('__missing__.pdf')
 333        >>> result is None
 334        True
 335        >>> 'not found' in buf.getvalue().lower()
 336        True
 337    """
 338    try:
 339        doc = fitz.open(pdf_path)
 340        if not doc.page_count:
 341            print("PDF document contains no pages.")
 342            return None
 343
 344        page = doc.load_page(0)  # Load the first page (index 0)
 345        width = page.rect.width
 346        height = page.rect.height
 347        doc.close()
 348        return width, height
 349    except fitz.FileNotFoundError:
 350        print(f"Error: PDF file not found at {pdf_path}")
 351        return None
 352    except (fitz.FileDataError, RuntimeError, ValueError) as e:
 353        print(f"An error occurred: {e}")
 354        return None

 357def get_image_dimensions_from_pdf_pages(
 358    pdf_path: str,
 359) -> list | None:
 360    """Extract image dimensions found in a PDF.
 361
 362    Args:
 363        pdf_path: Path to the PDF file.
 364
 365    Returns:
 366        list | None: Collected page image dimension lists, or ``None`` on error.
 367
 368    Note:
 369        Current implementation initializes ``pages`` but only appends to
 370        per-page ``images`` lists; callers should treat this as incomplete.
 371    """
 372    try:
 373        doc = fitz.open(pdf_path)
 374        pages = []
 375        for page_num in range(len(doc)):
 376            images = []
 377            page = doc.load_page(page_num)
 378            image_list = page.get_images(full=True)
 379
 380            for _, img_info in enumerate(image_list):
 381                xref = img_info[0]
 382                base_image = doc.extract_image(xref)
 383
 384                # Extract image dimensions from the extracted image data
 385                width = base_image["width"]
 386                height = base_image["height"]
 387                images.append((width, height))
 388        doc.close()
 389        return pages
 390    except (fitz.FileDataError, RuntimeError, ValueError) as e:
 391        print(f"An error occurred: {e}")

 793        self,
 794        xform_matrix: NDArray[np.float64],
 795        reps: int = 0,
 796        take: slice | None = None,
 797        incr: float
 798        | tuple[float, float]

 801        | NDArray

 803        | None = None,

 811            )

 831            tags.append(tag)

 833        if merge:

 955            BoundingBox: Axis-aligned box including ``frame.inner_sep``.
 956        """
 957        xmin, ymin, xmax, ymax = self.text_bounds()
 958        text_width = xmax - xmin
 959        text_height = ymax - ymin
 960        if self.text_width is not None:
 961            text_width = self.text_width
 962        if self.minimum_width is not None:
 963            text_width = max(text_width, self.minimum_width)

 966        h2 = text_height / 2
 967        x, y = self.pos[:2]

 969        effective_anchor = (
 970            self.anchor
 971            if self.anchor is not None
 972            else runtime_defaults["anchor"]
 973        )
 974        effective_align = (
 975            self.align
 976            if self.align is not None
 977            else runtime_defaults["tag_align"]
 978        )
 979
 980        if effective_anchor in (
 981            Anchor.WEST,
 982            Anchor.SOUTHWEST,
 983            Anchor.NORTHWEST,
 984        ):
 985            xmin = x - inner_sep
 986            xmax = x + text_width + inner_sep
 987        elif effective_anchor in (
 988            Anchor.EAST,
 989            Anchor.SOUTHEAST,
 990            Anchor.NORTHEAST,
 991        ):
 992            xmin = x - text_width - inner_sep
 993            xmax = x + inner_sep
 994        elif effective_align in (Align.LEFT, Align.FLUSH_LEFT):
 995            xmin = x - inner_sep
 996            xmax = x + text_width + inner_sep
 997        elif effective_align in (Align.RIGHT, Align.FLUSH_RIGHT):
 998            xmin = x - text_width - inner_sep
 999            xmax = x + inner_sep
1000        else:
1001            xmin = x - w2 - inner_sep
1002            xmax = x + w2 + inner_sep
1003
1004        ymin = y - h2 - inner_sep
1005        ymax = y + h2 + inner_sep

1007            (xmin, ymin),
1008            (xmax, ymin),
1009            (xmax, ymax),

 794        self,
