 30class TreeNode:
 31    """Node in a tree diagram with optional styling metadata.
 32
 33    Attributes:
 34        tag: Display label for the node.
 35        id: Unique integer id assigned at construction.
 36        children: Child ``TreeNode`` instances.
 37        extra: Optional role marker (e.g. ``"class"``, ``"method"``).
 38        font_size: Label font size.
 39        font_color: Label color.
 40        bold: Whether the label is drawn bold.
 41
 42    Examples:
 43        >>> node = TreeNode('root')
 44        >>> node.tag
 45        'root'
 46        >>> isinstance(node.id, int)
 47        True
 48    """
 49
 50    def __init__(
 51        self,
 52        tag: str = "",
 53        children: Sequence["TreeNode"] | None = None,
 54        extra: Any = None,
 55        font_size: float = 12,
 56        font_color: Any = sg.black,
 57        bold: bool = False,
 58    ) -> None:
 59        """Create a tree node.
 60
 61        Args:
 62            tag: Display label for the node.
 63            children: Child nodes. Defaults to an empty list.
 64            extra: Optional role marker used when choosing icons.
 65            font_size: Label font size.
 66            font_color: Label color.
 67            bold: Whether the label is drawn bold.
 68
 69        Examples:
 70            >>> node = TreeNode('leaf', bold=True)
 71            >>> node.bold
 72            True
 73        """
 74        self.tag = tag
 75        self.id = next_id()
 76        self.children = children if children is not None else []
 77        self.extra = extra
 78        self.font_size = font_size
 79        self.font_color = font_color
 80        self.bold = bold
 81
 82    def add_child(self, child: "TreeNode") -> None:
 83        """Add a child node if it is not already present.
 84
 85        Args:
 86            child: Node to append when not already present.
 87
 88        Examples:
 89            >>> parent = TreeNode('a')
 90            >>> child = TreeNode('b')
 91            >>> parent.add_child(child)
 92            >>> parent.num_all_children()
 93            1
 94        """
 95        if child.id not in (c.id for c in self.children):
 96            self.children.append(child)
 97
 98    def num_all_children(self) -> int:
 99        """Count descendants of this node (children and deeper).
100
101        Examples:
102            >>> root = TreeNode('r')
103            >>> root.add_child(TreeNode('a'))
104            >>> root.add_child(TreeNode('b'))
105            >>> root.num_all_children()
106            2
107        """
108        return len(self.children) + sum(
109            child.num_all_children() for child in self.children
110        )
111
112    def depth(self) -> int:
113        """Return a layout depth metric based on descendant counts.
114
115        Examples:
116            >>> leaf = TreeNode('leaf')
117            >>> leaf.depth()
118            0
119            >>> root = TreeNode('r')
120            >>> root.add_child(TreeNode('c'))
121            >>> root.depth()
122            1
123        """
124        if not self.children:
125            return 0
126        n = self.num_all_children()
127        m = self.children[-1].num_all_children()
128        return n - m

131def make_tree(
132    node: TreeNode,
133    canvas: Any = None,
134    file_path: str | None = None,
135    overwrite: bool = False,
136    dx: float = 10,
137    dy: float = 18,
138    icons: Sequence[Any] | None = None,
139    line1_color: Any = sg.gray,
140    line1_width: float = 0.5,
141    line1_cap: Any = sg.LineCap.ROUND,
142    line2_color: Any = sg.gray,
143    line2_width: float = 0.75,
144    line2_cap: Any = sg.LineCap.ROUND,
145    scale: float = 1,
146) -> None:
147    """Draw a tree diagram on a canvas and optionally save it.
148
149    Args:
150        node: The root ``TreeNode``.
151        canvas: Canvas to draw on (mutated). Created if omitted.
152        file_path: Optional path to save the drawing.
153        overwrite: Whether to overwrite an existing file.
154        dx: Horizontal spacing between indentation levels.
155        dy: Vertical spacing between successive nodes.
156        icons: Optional list of four icons for node kinds.
157        line1_color: Color of connector stubs.
158        line1_width: Width of connector stubs.
159        line1_cap: Cap style of connector stubs.
160        line2_color: Color of longer connectors.
161        line2_width: Width of longer connectors.
162        line2_cap: Cap style of longer connectors.
163        scale: Overall scale factor for the drawing.
164
165    Examples:
166        >>> root = TreeNode('demo')
167        >>> root.add_child(TreeNode('child'))
168        >>> canvas = sg.Canvas()
169        >>> make_tree(root, canvas=canvas)  # doctest: +SKIP
170    """
171    diamond = sg.Shape([(0, 5), (3, 0), (0, -5), (-3, 0)], closed=True)
172    diamond.fill_color = sg.black
173    diamond.stroke = False
174
175    star_polygon = sg.reg_star_polygon
176
177    star2 = star_polygon(5, 2, 5, fill_color=sg.blue, stroke=False)
178    star2.rotate(sg.pi / 10)
179    d = 6
180    square = sg.Shape([(0, d), (d, d), (d, 0), (0, 0)], closed=True)
181    square.fill_color = sg.blue
182    square.stroke = False
183    star = star_polygon(8, 3, 7, fill_color=sg.red, stroke=False)
184    circle = sg.Circle(1.5, fill_color=sg.white, stroke=False)
185    hexagon = sg.Group(
186        [
187            sg.reg_poly_shape(6, 4, (0, 0), fill_color=sg.teal, stroke=False),
188            circle,
189        ]
190    )
191
192    # Single red diamond for enum values
193    red_diamond = sg.Shape([(0, 4), (3, 0), (0, -4), (-3, 0)], closed=True)
194    red_diamond.fill_color = sg.red
195    red_diamond.stroke = False
196    # Two red diamonds side-by-side for enum classes
197    _rd1 = sg.Shape([(-3, 4), (0, 0), (-3, -4), (-6, 0)], closed=True)
198    _rd2 = sg.Shape([(3, 4), (6, 0), (3, -4), (0, 0)], closed=True)
199    _rd1.fill_color = _rd2.fill_color = sg.red
200    _rd1.stroke = _rd2.stroke = False
201
202    count = 0
203    if icons is None:
204        icons = [star, diamond, star2, hexagon]
205
206    icon1, icon2, icon3, icon4 = icons
207    icon5 = red_diamond  # enum value / enum class
208    icon7 = sg.Group(
209        [
210            sg.reg_poly_shape(6, 4, (0, 0), fill_color=sg.teal, stroke=False),
211            sg.Circle(1.5, fill_color=sg.white, stroke=False),
212        ]
213    )  # method
214
215    def node_icon(node: TreeNode) -> Any:
216        if node.extra == "enum_value":
217            return icon5
218        if node.extra == "enum_class":
219            return icon5
220        if node.extra == "method":
221            return icon7
222        if node.extra == "property":
223            return icon4
224        if node.extra == "class":
225            return icon3
226        return icon2

228    def draw_tree(
229        node: TreeNode, indent: int = 0, canvas: Any = None
230    ) -> None:
231        """Draw ``node`` and recursively draw its children.
232
233        Args:
234            node: Current tree node.
235            indent: Indentation level (horizontal depth).
236            canvas: Canvas receiving the drawing commands.
237        """
238        nonlocal count
239        count += 1
240        x = indent * dx
241        y = -count * dy
242        x1 = x
243        y1 = y - node.depth() * dy
244        cx, cy = x, y
245        if node.depth() > 0:
246            canvas.line(
247                (x, y),
248                (x1, y1),
249                line_color=line1_color,
250                line_width=line1_width,
251                line_cap=line1_cap,
252            )
253
254        # Shorter horizontal connectors from level 1 and deeper
255        horiz_span = 1 if indent >= 1 else 0
256        x2 = x1 + horiz_span * dx
257        y2 = y
258        if indent > 0:
259            x -= dx
260        p1 = (x, y)
261        p2 = (x2, y2)
262        if not simetri.geom.points.point_utils.close_points_square(p1, p2):
263            canvas.line(
264                p1,
265                p2,
266                line_color=line2_color,
267                line_width=line2_width,
268                line_cap=line2_cap,
269            )
270        if node.depth() > 0 and indent > 0:
271            icon = node_icon(node)
272            icon.move_to((cx, cy))
273            canvas.draw(icon)
274        font_size = node.font_size
275        font_color = node.font_color
276        bold = node.bold
277
278        if indent > 0:
279            icon = node_icon(node)
280            icon.move_to((x, y))
281            canvas.draw(icon)
282        else:
283            icon1.move_to((x, y))
284            canvas.draw(icon1)
285
286        title_x = x2 + 8 if indent == 0 else x2
287        label = str(node.tag)
288        canvas.text(
289            label,
290            (title_x, y2),
291            font_family=sg.FontFamily.MONOSPACE,
292            font_size=font_size,
293            color=sg.red,
294            anchor=sg.Anchor.WEST,
295            fill=False,
296            bold=bold,
297            font_color=font_color,
298        )
299        for child in node.children:
300            draw_tree(child, indent + 1, canvas=canvas)
301
302    draw_tree(node, canvas=canvas)
303    if scale != 1:
304        canvas.scale = (scale, scale)
305    if file_path:
306        canvas.save(file_path, overwrite=overwrite)

313def _print_file_tree(root_dir: str | Path, prefix: str = "") -> None:
314    """Recursively print a directory tree using Unicode branch characters."""
315    entries = sorted(os.listdir(root_dir))
316    entries = [e for e in entries if not e.startswith(".")]  # Hide hidden files
317
318    for i, entry in enumerate(entries):
319        path = os.path.join(root_dir, entry)
320        is_last = i == len(entries) - 1
321
322        # Unicode branch characters
323        if not os.path.isdir(path):
324            print(f"{prefix}{'└──' if is_last else '├──'}🗎 {entry}")
325        else:
326            print(f"{prefix}{'└──' if is_last else '├──'}📁 {entry}")
327
328        if os.path.isdir(path):
329            new_prefix = f"{prefix}{'    ' if is_last else '│   '}"
330            _print_file_tree(path, new_prefix)

333def tree_from_class(
334    canvas: Any | None,
335    class_obj: type,
336    exclude: Iterable[str] | None = None,
337    stubs: Iterable[str] | None = None,
338    draw: bool = True,
339    file_path: str | None = None,
340    overwrite: bool = False,
341    show_enum_values: bool = False,
342    **kwargs: object,
343) -> TreeNode:
344    """Create a TreeNode hierarchy from a class and optionally draw it on a canvas.
345
346    Args:
347        canvas: Target canvas used for drawing when draw=True.
348        class_obj: Root class to introspect.
349        exclude: Iterable of attribute/class names to skip.
350        stubs: Iterable of class names that should appear as single nodes (no subnodes).
351        draw: Whether to draw to canvas.
352        file_path: Optional output file path.
353        overwrite: Overwrite output file when saving.
354        show_enum_values: If True, enum members are listed as child nodes.
355        **kwargs: Extra args forwarded to make_tree.
356
357    Examples:
358        >>> root = tree_from_class(None, TreeNode, draw=False)
359        >>> root.tag.startswith('TreeNode')
360        True
361        >>> root.num_all_children() > 0
362        True
363    """
364    from typing import Union, get_args, get_origin
365
366    def _type_to_str(tp: Any) -> str:
367        if tp is inspect._empty:
368            return "Any"
369        if isinstance(tp, str):
370            return tp
371
372        origin = get_origin(tp)
373        args = get_args(tp)
374        if origin is not None:
375            if origin is Union:
376                inner = " | ".join(_type_to_str(arg) for arg in args)
377                return inner
378
379            origin_name = getattr(
380                origin, "__name__", str(origin).replace("typing.", "")
381            )
382            if args:
383                inner = ", ".join(_type_to_str(arg) for arg in args)
384                return f"{origin_name}[{inner}]"
385            return origin_name
386
387        if hasattr(tp, "__name__"):
388            return tp.__name__
389        return str(tp).replace("typing.", "")
390
391    def _iter_referenced_types(tp: Any) -> Iterator[type]:
392        origin = get_origin(tp)
393        args = get_args(tp)
394        if origin is None:
395            if tp in (list, tuple, set, Sequence):
396                return
397            if inspect.isclass(tp):
398                yield tp
399            return
400        for arg in args:
401            if inspect.isclass(arg):
402                yield arg
403            else:
404                yield from _iter_referenced_types(arg)
405
406    def _resolve_type_name_in_context(type_name: str, cls: type) -> type | None:
407        module = inspect.getmodule(cls)
408        if module is not None:
409            candidate = getattr(module, type_name, None)
410            if inspect.isclass(candidate):
411                return candidate
412
413        candidate = getattr(sg, type_name, None)
414        if inspect.isclass(candidate):
415            return candidate
416        return None
417
418    def _infer_sequence_item_types(
419        cls: type, field_name: str, ann: Any
420    ) -> list[type]:
421        inferred = []
422        origin = get_origin(ann)
423        args = get_args(ann)
424
425        is_sequence = ann is list or origin in (list, tuple, set, Sequence)
426        if not is_sequence or args:
427            return inferred
428
429        doc = inspect.getdoc(cls) or ""
430        if doc:
431            field_pattern = re.compile(
432                rf"^\s*{re.escape(field_name)}\s*\([^)]*\)\s*:\s*(.+)$",
433                flags=re.IGNORECASE | re.MULTILINE,
434            )
435            field_match = field_pattern.search(doc)
436            if field_match:
437                desc = field_match.group(1)
438                names = [
439                    match.group(1)
440                    for match in re.finditer(
441                        r"\bList\[([A-Za-z_][A-Za-z0-9_]*)\]", desc
442                    )
443                ]
444
445                names.extend(
446                    match.group(1)
447                    for match in re.finditer(
448                        r"\b(?:list|sequence|tuple|set)\s+of\s+([A-Za-z_][A-Za-z0-9_]*)",
449                        desc,
450                        flags=re.IGNORECASE,
451                    )
452                )
453
454                for type_name in names:
455                    ref_type = _resolve_type_name_in_context(type_name, cls)
456                    if ref_type is not None:
457                        inferred.append(ref_type)
458
459        return inferred
460
461    def _method_signature(member: Callable[..., Any]) -> str:
462        sig = inspect.signature(member)
463
464        params = []
465        for i, (name, param) in enumerate(sig.parameters.items()):
466            if i == 0 and name in ("self", "cls"):
467                continue
468            params.append(str(param))
469
470        ret_text = _type_to_str(sig.return_annotation)
471        return f"({', '.join(params)}) -> {ret_text}"
472
473    def _add_class_tree(
474        node: TreeNode,
475        cls: type,
476        visited: set[type],
477        depth: int = 0,
478        max_depth: int = 2,
479    ) -> None:
480        if cls in visited or depth > max_depth:
481            return
482        if cls.__name__ in stubs:
483            return
484        visited.add(cls)
485
486        # If the class is an Enum and enum values are requested, list members and stop.
487        if show_enum_values and issubclass(cls, enum.Enum):
488            for member in cls:
489                value_node = TreeNode(
490                    f"{member.name} = {member.value!r}", extra="enum_value"
491                )
492                node.add_child(value_node)
493            return
494
495        sig = inspect.signature(cls.__init__)
496        properties = {
497            name: member
498            for name, member in inspect.getmembers(cls)
499            if isinstance(member, property)
500            and not name.startswith("_")
501            and name not in exclude
502        }
503
504        for name, param in sig.parameters.items():
505            if name == "self":
506                continue
507            if param.kind in (
508                inspect.Parameter.VAR_KEYWORD,
509                inspect.Parameter.VAR_POSITIONAL,
510            ):
511                continue
512            if name.startswith("_") or name in exclude:
513                continue
514            if name in properties:
515                # Prefer the property representation when both exist.
516                continue
517
518            ann = param.annotation
519            ref_types = list(_iter_referenced_types(ann))
520            if not ref_types:
521                ref_types = _infer_sequence_item_types(cls, name, ann)
522
523            ann_text = _type_to_str(ann)
524            origin = get_origin(ann)
525            args = get_args(ann)
526            is_untyped_sequence = (
527                ann is list or origin in (list, tuple, set, Sequence)
528            ) and not args
529            if is_untyped_sequence and len(ref_types) == 1:
530                seq_name = (
531                    "list"
532                    if ann is list or origin is list
533                    else (
534                        "tuple"
535                        if origin is tuple
536                        else ("set" if origin is set else "Sequence")
537                    )
538                )
539                ann_text = f"{seq_name}[{ref_types[0].__name__}]"
540
541            child = TreeNode(f"{name}: {ann_text}")
542            node.add_child(child)
543
544            for ref_type in ref_types:
545                if (
546                    ref_type.__module__.startswith("simetri")
547                    and ref_type is not cls
548                ):
549                    if ref_type.__name__ in exclude:
550                        if (
551                            issubclass(ref_type, enum.Enum)
552                            and child.extra is None
553                        ):
554                            child.extra = "enum_class"
555                        continue
556                    sub_extra = (
557                        "enum_class"
558                        if issubclass(ref_type, enum.Enum)
559                        else "class"
560                    )
561                    sub = TreeNode(ref_type.__name__, extra=sub_extra)
562                    child.add_child(sub)
563                    if ref_type.__name__ not in stubs:
564                        _add_class_tree(
565                            sub, ref_type, visited.copy(), depth + 1, max_depth
566                        )
567
568        for name, member in properties.items():
569            ret = inspect.signature(member.fget).return_annotation
570            if ret is inspect._empty and name in sig.parameters:
571                ret = sig.parameters[name].annotation
572            prop_node = TreeNode(
573                f"{name} -> {_type_to_str(ret)}", extra="property"
574            )
575            node.add_child(prop_node)
576
577            for ref_type in _iter_referenced_types(ret):
578                if (
579                    ref_type.__module__.startswith("simetri")
580                    and ref_type is not cls
581                ):
582                    if ref_type.__name__ in exclude:
583                        if (
584                            issubclass(ref_type, enum.Enum)
585                            and prop_node.extra == "property"
586                        ):
587                            prop_node.extra = "enum_class"
588                        continue
589                    sub_extra = (
590                        "enum_class"
591                        if issubclass(ref_type, enum.Enum)
592                        else "class"
593                    )
594                    sub = TreeNode(ref_type.__name__, extra=sub_extra)
595                    prop_node.add_child(sub)
596                    if ref_type.__name__ not in stubs:
597                        _add_class_tree(
598                            sub, ref_type, visited.copy(), depth + 1, max_depth
599                        )
600
601        methods = {
602            name: member
603            for name, member in inspect.getmembers(cls)
604            if inspect.isroutine(member)
605            and not name.startswith("_")
606            and name not in exclude
607        }
608
609        for name, member in methods.items():
610            try:
611                ret = inspect.signature(member).return_annotation
612                ret_str = _type_to_str(ret)
613            except (ValueError, TypeError):
614                ret_str = "Any"
615            method_node = TreeNode(f"{name} -> {ret_str}", extra="method")
616            node.add_child(method_node)
617
618    if exclude is None:
619        exclude = set()
620    exclude = set(exclude)
621    if stubs is None:
622        stubs = set()
623    stubs = set(stubs)
624
625    bases = [
626        base.__name__ for base in class_obj.__bases__ if base is not object
627    ]
628    title = (
629        f"{class_obj.__name__}({', '.join(bases)})"
630        if bases
631        else class_obj.__name__
632    )
633    root = TreeNode(title, extra="root", font_color=sg.orange)
634    _add_class_tree(root, class_obj, visited=set(), depth=0, max_depth=2)
635
636    if draw:
637        if canvas is None:
638            canvas = sg.Canvas()
639        make_tree(
640            root,
641            canvas=canvas,
642            file_path=file_path,
643            overwrite=overwrite,
644            **kwargs,
645        )
646
647    return root

650def print_tree(
651    root: str | Path | type | TreeNode,
652    prefix: str = "",
653    file_tree: bool = True,
654    canvas: Any | None = None,
655    file_path: str | None = None,
656    overwrite: bool = False,
657    exclude: Iterable[str] | None = None,
658    stubs: Iterable[str] | None = None,
659    **kwargs: object,
660) -> TreeNode | None:
661    """Print or draw a tree.
662
663    Args:
664        root: Root directory (for file trees), a class object, or a TreeNode.
665        prefix: Prefix used for console file tree formatting.
666        file_tree: If True, use filesystem tree printer (existing behavior).
667        canvas: Optional canvas for graphic tree rendering.
668        file_path: Optional output path for graphic tree rendering.
669        overwrite: Overwrite output file when saving.
670        exclude: Optional exclude list used when root is a class.
671        stubs: Optional class-name list to render as single non-expanded nodes.
672        **kwargs: Extra arguments forwarded to make_tree.
673
674    Examples:
675        >>> node = TreeNode('x')
676        >>> out = print_tree(node, file_tree=False, canvas=sg.Canvas())
677        >>> out.tag
678        'x'
679    """
680    if file_tree:
681        _print_file_tree(root, prefix)
682        return None
683
684    if canvas is None:
685        canvas = sg.Canvas()
686
687    if inspect.isclass(root):
688        node = tree_from_class(
689            canvas, root, exclude=exclude, stubs=stubs, draw=False
690        )
691    else:
692        node = root
693
694    make_tree(
695        node, canvas=canvas, file_path=file_path, overwrite=overwrite, **kwargs
696    )
697    return node

700def list_directories(path: str | Path) -> list[str]:
701    """Return subdirectory names under ``path`` (non-recursive).
702
703    Args:
704        path: Directory to list.
705
706    Returns:
707        Names of immediate child directories.
708
709    Examples:
710        >>> 'extensions' in list_directories('src/simetri')
711        True
712    """
713    return [e for e in os.listdir(path) if isdir(join(path, e))]

558                        "enum_class"
559                        if issubclass(ref_type, enum.Enum)
560                        else "class"
561                    )
562                    sub = TreeNode(ref_type.__name__, extra=sub_extra)
563                    child.add_child(sub)
564                    if ref_type.__name__ not in stubs:
565                        _add_class_tree(
566                            sub, ref_type, visited.copy(), depth + 1, max_depth
567                        )
568
569        for name, member in properties.items():
570            ret = inspect.signature(member.fget).return_annotation
571            if ret is inspect._empty and name in sig.parameters:
572                ret = sig.parameters[name].annotation
573            prop_node = TreeNode(
574                f"{name} -> {_type_to_str(ret)}", extra="property"
575            )
576            node.add_child(prop_node)
577
578            for ref_type in _iter_referenced_types(ret):
579                if (
580                    ref_type.__module__.startswith("simetri")
581                    and ref_type is not cls
582                ):
583                    if ref_type.__name__ in exclude:
584                        if (
585                            issubclass(ref_type, enum.Enum)
586                            and prop_node.extra == "property"
587                        ):
588                            prop_node.extra = "enum_class"
589                        continue
590                    sub_extra = (
591                        "enum_class"
592                        if issubclass(ref_type, enum.Enum)
593                        else "class"
594                    )
595                    sub = TreeNode(ref_type.__name__, extra=sub_extra)
596                    prop_node.add_child(sub)
597                    if ref_type.__name__ not in stubs:
598                        _add_class_tree(
599                            sub, ref_type, visited.copy(), depth + 1, max_depth
600                        )
601
602        methods = {
603            name: member
604            for name, member in inspect.getmembers(cls)
605            if inspect.isroutine(member)
606            and not name.startswith("_")
607            and name not in exclude
608        }
609
610        for name, member in methods.items():
611            try:
612                ret = inspect.signature(member).return_annotation
613                ret_str = _type_to_str(ret)
614            except (ValueError, TypeError):
615                ret_str = "Any"
616            method_node = TreeNode(f"{name} -> {ret_str}", extra="method")
617            node.add_child(method_node)
618
619    if exclude is None:
620        exclude = set()
621    exclude = set(exclude)
622    if stubs is None:
623        stubs = set()
624    stubs = set(stubs)
625
626    bases = [
627        base.__name__ for base in class_obj.__bases__ if base is not object
628    ]
629    title = (
630        f"{class_obj.__name__}({', '.join(bases)})"
631        if bases
632        else class_obj.__name__
633    )
634    root = TreeNode(title, extra="root", font_color=sg.orange)
635    _add_class_tree(root, class_obj, visited=set(), depth=0, max_depth=2)
636
637    if draw:
638        if canvas is None:
639            canvas = sg.Canvas()
640        make_tree(
641            root,
642            canvas=canvas,
643            file_path=file_path,
644            overwrite=overwrite,
645            **kwargs,
646        )
647
648    return root
