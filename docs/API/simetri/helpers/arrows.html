 62def _split_annotation_kwargs(
 63    draw_kwargs: dict[str, object],
 64) -> tuple[dict[str, object], dict[str, object]]:
 65    """Separate annotation part styles from leader shape kwargs."""
 66    style_names = set(_ANNOTATION_HEAD_KEYS)
 67    style_names.update(_ANNOTATION_SHAFT_KEYS)
 68    style_names.update(_ANNOTATION_TAG_KEYS)
 69    style_names.update(_LEADER_LINE_KEYS)
 70    style_kwargs: dict[str, object] = {}
 71    leader_kwargs: dict[str, object] = {}
 72    for key, value in draw_kwargs.items():
 73        if key in style_names or key in ("alpha", "color"):
 74            style_kwargs[key] = value
 75        else:
 76            leader_kwargs[key] = value
 77    return style_kwargs, leader_kwargs

239class AnnotationArrow(Group):
240    """A leader from a feature point to text or a circled number.
241
242    The leader is two segments: an angled shaft from ``tip`` to ``elbow``,
243    then a landing from ``elbow`` to ``landing``. The arrowhead sits at
244    ``tip``. If only ``landing`` or only ``elbow`` is given, the missing
245    point is inferred so the landing is horizontal.
246
247    Args:
248        tip (PointType): Feature point the arrowhead points at.
249        text (str | int): Label text, or a number for a balloon. Defaults
250            to "".
251        landing (PointType, optional): End of the landing, at the label.
252            Defaults to None.
253        elbow (PointType, optional): Break between the angled shaft and
254            the landing. Defaults to None.
255        landing_length (float, optional): Horizontal landing length used
256            when inferring ``elbow`` or ``landing``. Defaults to None
257            (``runtime_defaults["landing_length"]``).
258        circled (bool, optional): If True, the label is a balloon
259            (circled number or text). Defaults to False.
260        font_size (float, optional): Label font size. Defaults to None
261            (``runtime_defaults["font_size"]``).
262        **kwargs: ``shaft_line_alpha``, ``shaft_line_color``,
263            ``shaft_line_dash_array``, and ``shaft_line_width`` style
264            the angled shaft and the landing. ``head_fill_alpha``,
265            ``head_fill_color``, ``head_line_alpha``,
266            ``head_line_color``, and ``head_line_width`` style the
267            arrowhead. ``tag_bold``, ``tag_fill``, ``tag_fill_color``,
268            ``tag_font_alpha``, ``tag_font_color``, ``tag_font_family``,
269            ``tag_font_size``, ``tag_line_color``, ``tag_line_width``,
270            and ``tag_stroke`` style the label. ``color`` and ``alpha``
271            set every part; a prefixed name wins. ``tag_stroke`` is
272            left unchanged unless given. Other shape kwargs, such as
273            ``line_color``, go to the leader ``Arrow`` and the landing
274            ``Shape``.
275
276    Examples:
277        >>> import simetri.graphics as sg
278        >>> note = sg.AnnotationArrow((0, 0), "A", landing=(40, 20))
279        >>> note.tip
280        (0, 0)
281        >>> note.elbow
282        (20, 20)
283        >>> note.landing
284        (40, 20)
285        >>> balloon = sg.AnnotationArrow(
286        ...     (0, 0), 1, landing=(40, 20), circled=True
287        ... )
288        >>> balloon.text
289        '1'
290        >>> balloon.tag.frame_shape == sg.FrameShape.CIRCLE
291        True
292        >>> balloon.tag.stroke
293        True
294        >>> note.tag.stroke
295        False
296        >>> styled = sg.AnnotationArrow(
297        ...     (0, 0),
298        ...     "A",
299        ...     landing=(40, 20),
300        ...     head_fill_color=sg.red,
301        ...     shaft_line_color=sg.blue,
302        ...     tag_font_color=sg.green,
303        ... )
304        >>> styled.arrow.line.line_color == sg.blue
305        True
306        >>> styled.landing_line.line_color == sg.blue
307        True
308        >>> styled.arrow.heads[0].fill_color == sg.red
309        True
310        >>> styled.tag.font_color == sg.green
311        True
312        >>> sg.AnnotationArrow((0, 0), "A")  # doctest: +IGNORE_EXCEPTION_DETAIL
313        Traceback (most recent call last):
314        ValueError: AnnotationArrow requires landing or elbow.
315    """
316
317    def __init__(
318        self,
319        tip: PointType,
320        text: str | int = "",
321        landing: PointType | None = None,
322        elbow: PointType | None = None,
323        landing_length: float | None = None,
324        circled: bool = False,
325        font_size: float | None = None,
326        **kwargs: object,
327    ) -> None:
328        """Create a broken-leader annotation arrow.
329
330        See the class docstring for argument details.
331
332        Examples:
333            >>> import simetri.graphics as sg
334            >>> note = sg.AnnotationArrow((0, 0), "A", landing=(40, 20))
335            >>> note.tip
336            (0, 0)
337            >>> note.elbow
338            (20, 20)
339            >>> note.landing
340            (40, 20)
341        """
342        if elbow is None and landing is None:
343            raise ValueError("AnnotationArrow requires landing or elbow.")
344        if elbow is not None and landing is not None:
345            if landing_length is not None:
346                raise ValueError(
347                    "Do not pass landing_length when both elbow "
348                    "and landing are given."
349                )
350        else:
351            (landing_length,) = get_defaults(
352                ["landing_length"], [landing_length]
353            )
354
355        (font_size,) = get_defaults(["font_size"], [font_size])
356        tip_x, tip_y = tip[:2]
357        self.tip = (tip_x, tip_y)
358        self.circled = circled
359        self.font_size = font_size
360        self.text = str(text)
361
362        if elbow is None:
363            landing_x, landing_y = landing[:2]
364            if landing_x >= tip_x:
365                elbow_x = landing_x - landing_length
366            else:
367                elbow_x = landing_x + landing_length
368            elbow_y = landing_y
369            self.elbow = (elbow_x, elbow_y)
370            self.landing = (landing_x, landing_y)
371        elif landing is None:
372            elbow_x, elbow_y = elbow[:2]
373            if elbow_x >= tip_x:
374                landing_x = elbow_x + landing_length
375            else:
376                landing_x = elbow_x - landing_length
377            landing_y = elbow_y
378            self.elbow = (elbow_x, elbow_y)
379            self.landing = (landing_x, landing_y)
380        else:
381            elbow_x, elbow_y = elbow[:2]
382            landing_x, landing_y = landing[:2]
383            self.elbow = (elbow_x, elbow_y)
384            self.landing = (landing_x, landing_y)
385
386        style_kwargs, leader_kwargs = _split_annotation_kwargs(kwargs)
387        self.arrow = Arrow(self.elbow, self.tip, **leader_kwargs)
388        items = [self.arrow]
389        abs_tol = runtime_defaults["abs_tol"]
390        landing_span = distance(self.elbow, self.landing)
391        if landing_span > abs_tol:
392            self.landing_line = Shape(
393                [self.elbow, self.landing], fill=False, **leader_kwargs
394            )
395            items.append(self.landing_line)
396        else:
397            self.landing_line = None
398
399        land_dx = landing_x - elbow_x
400        land_dy = landing_y - elbow_y
401        land_len = hypot(land_dx, land_dy)
402        if land_len > abs_tol:
403            unit_x = land_dx / land_len
404            unit_y = land_dy / land_len
405        elif landing_x >= tip_x:
406            unit_x, unit_y = 1.0, 0.0
407        else:
408            unit_x, unit_y = -1.0, 0.0
409
410        text_gap = runtime_defaults["text_offset"]
411        if circled:
412            tag_x, tag_y = landing_x, landing_y
413            tag_align = Align.CENTER
414        else:
415            tag_x = landing_x + unit_x * text_gap
416            tag_y = landing_y + unit_y * text_gap
417            if landing_x >= tip_x:
418                tag_align = Align.LEFT
419            else:
420                tag_align = Align.RIGHT
421        self.tag = Tag(
422            self.text,
423            (tag_x, tag_y),
424            font_size=font_size,
425            align=tag_align,
426        )
427        if circled:
428            self.tag.frame_shape = FrameShape.CIRCLE
429            self.tag.stroke = True
430            self.tag.fill = True
431            self.tag.fill_color = colors.white
432        _style_annotation(self, style_kwargs)
433        items.append(self.tag)
434        super().__init__(items, subtype=Types.ANNOTATION)

437class ArrowHead(Shape):
438    """An ArrowHead object is a shape that represents the head of an arrow.
439
440    Args:
441        length (float, optional): The length of the arrow head. Defaults to None.
442        width_ (float, optional): The width of the arrow head. Defaults to None.

447        >>> import simetri.graphics as sg
448        >>> head = sg.ArrowHead(length=80, width_=3)
449        >>> head.vertices
450        ((0.0, 0.0), (0.0, -1.5), (80.0, 0.0), (0.0, 1.5))
451        >>> head.head_length
452        80
453        >>> head.head_width
454        3
455    """
456
457    def __init__(
458        self,
459        length: float | None = None,
460        width_: float | None = None,
461        points: list | None = None,
462        **kwargs: object,
463    ) -> None:
464        """Create an arrow head shape.
465
466        See the class docstring for argument details.
467        """
468        length, width_ = get_defaults(
469            ["arrow_head_length", "arrow_head_width"], [length, width_]
470        )
471        if points is None:
472            w2 = width_ / 2
473            points = [(0, 0), (0, -w2), (length, 0), (0, w2)]

499def arrow(
500    p1: PointType,
501    p2: PointType,
502    head_length: float = 10,
503    head_width: float = 4,
504    line_width: float = 1,
505    line_color: Color = colors.black,
506    fill_color: Color = colors.black,
507    centered: bool = False,
508) -> Group:
509    """Return an arrow from p1 to p2.
510
511    Args:
512        p1 (tuple): The starting point of the arrow.
513        p2 (tuple): The ending point of the arrow.
514        head_length (int, optional): The length of the arrow head. Defaults to 10.
515        head_width (int, optional): The width of the arrow head. Defaults to 4.
516        line_width (int, optional): The width of the arrow line. Defaults to 1.
517        line_color (Color, optional): The color of the arrow line. Defaults to colors.black.
518        fill_color (Color, optional): The fill color of the arrow head. Defaults to colors.black.
519        centered (bool, optional): Whether the arrow is centered. Defaults to False.
520
521    Returns:
522        Group: A Group object containing the arrow shapes.
523
524    Examples:
525        >>> import simetri.graphics as sg
526        >>> shaft = sg.arrow((0, 0), (40, 0))
527        >>> len(shaft)
528        2
529        >>> shaft[0].vertices
530        ((0.0, 0.0), (40.0, 0.0))
531        >>> shaft[1].vertices
532        ((30.0, 2.0), (40.0, 0.0), (30.0, -2.0))
533    """
534    x1, y1 = p1[:2]
535    x2, y2 = p2[:2]
536    dx = x2 - x1
537    dy = y2 - y1
538    angle = atan2(dy, dx)
539    body = Shape(
540        [(x1, y1), (x2, y2)],
541        closed=False,
542        line_color=line_color,
543        fill_color=fill_color,
544        line_width=line_width,
545    )
546    w2 = head_width / 2
547    head = Shape(
548        [(-head_length, w2), (0, 0), (-head_length, -w2)],
549        closed=True,
550        line_color=line_color,
551        fill_color=fill_color,
552        line_width=line_width,
553    )
554    head.rotate(angle)
555    if centered:
556        head.translate(*midpoint((x1, y1), (x2, y2)))
557    else:
558        head.translate(x2, y2)
559    return Group([body, head])

600    """An ArcArrow object is an arrow with an arc.
601
602    Args:
603        center: The center of the arc.
604        radius_x: Semi-axis along x.
605        radius_y: Semi-axis along y; defaults to ``radius_x``.
606        start_angle: The starting angle of the arc in radians. Defaults to 0.

609            ``span_angle`` or ``end_angle`` is required.
610        end_angle: Ending angle in radians. Mutually exclusive with
611            ``span_angle``.

820def vec_arrow(
821    vec: VecType,
822    *,
823    start: PointType | None = None,
824    end: PointType | None = None,
825    **kwargs: object,
826) -> Arrow:
827    """Return an ``Arrow`` from a displacement and a start or end point.
828
829    ``vec`` is a displacement. It is not stored on the ``Arrow``. Extra
830    keyword arguments are passed to ``Arrow``.
831
832    Args:
833        vec (VecType): Displacement from tail to tip.
834        start (PointType, optional): Tail position. Defaults to None.
835        end (PointType, optional): Tip position. Defaults to None.
836        **kwargs: Passed to ``Arrow``.
837
838    Returns:
839        Arrow: Arrow from ``start`` to ``start + vec``, or from
840        ``end - vec`` to ``end``.
841
842    Raises:
843        ValueError: If ``vec`` is zero, if neither ``start`` nor ``end``
844            is given, or if both are given and ``end - start`` does not
845            match ``vec``.
846
847    Examples:
848        >>> import simetri.graphics as sg
849        >>> arrow = sg.vec_arrow(sg.Vector(60, 80), start=(40, 20))
850        >>> arrow.p1
851        (40, 20)
852        >>> arrow.p2
853        (100, 100)
854        >>> arrow = sg.vec_arrow(sg.Vector(60, 80), end=(13, 24))
855        >>> arrow.p1
856        (-47, -56)
857        >>> arrow.p2
858        (13, 24)
859        >>> sg.vec_arrow(sg.Vector(0, 0), start=(40, 40))
860        Traceback (most recent call last):
861        ValueError: Cannot create an Arrow from a zero-length Vector.
862        >>> sg.vec_arrow(sg.Vector(60, 80), start=(0, 0), end=(40, 0))
863        Traceback (most recent call last):
864        ValueError: start and end are not consistent with the Vector displacement (60, 80).
865    """
866    vector_x, vector_y = vec[:2]
867    if vector_x == 0 and vector_y == 0:
868        raise ValueError("Cannot create an Arrow from a zero-length Vector.")
869    if start is None and end is None:
870        raise ValueError("vec_arrow requires start or end.")
871    if start is not None and end is not None:
872        start_x, start_y = start[:2]
873        end_x, end_y = end[:2]
874        displacement_x = end_x - start_x
875        displacement_y = end_y - start_y
876        offset = hypot(displacement_x - vector_x, displacement_y - vector_y)
877        if offset <= runtime_defaults["abs_tol"]:
878            issue_warning(
879                f"Duplicate position used for Vector({start}, {end}).",
880                warning_type=WarningType.vector.duplicate,
881            )
882            return Arrow((start_x, start_y), (end_x, end_y), **kwargs)
883        raise ValueError(
884            "start and end are not consistent with the Vector "
885            f"displacement {(vector_x, vector_y)}."
886        )
887    if start is not None:
888        start_x, start_y = start[:2]
889        return Arrow(
890            (start_x, start_y),
891            (start_x + vector_x, start_y + vector_y),
892            **kwargs,
893        )
894    end_x, end_y = end[:2]
895    return Arrow(
896        (end_x - vector_x, end_y - vector_y),
897        (end_x, end_y),
898        **kwargs,
899    )

Return an Arrow from a displacement and a start or end point.

vec is a displacement. It is not stored on the Arrow. Extra keyword arguments are passed to Arrow.

Arguments:
  • vec (VecType): Displacement from tail to tip.
  • start (PointType, optional): Tail position. Defaults to None.
  • end (PointType, optional): Tip position. Defaults to None.
  • **kwargs: Passed to Arrow.
Returns:

Arrow: Arrow from start to start + vec, or from end - vec to end.

Raises:
  • ValueError: If vec is zero, if neither start nor end is given, or if both are given and end - start does not match vec.
