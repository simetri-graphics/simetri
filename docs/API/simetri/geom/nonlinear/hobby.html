103class HobbyCurve:
104    """A class for calculating the control points required to draw a Hobby curve.
105
106    Attributes:
107        points (list[HobbyPoint]): The list of points defining the curve.
108        ctrl_pts (list[tuple]): The calculated control points.
109        is_cyclic (bool): Whether the curve is closed.
110        begin_curl (float): Curl value for the beginning of the curve.
111        end_curl (float): Curl value for the end of the curve.
112        n_points (int): Number of points in the curve.
113        debug_mode (bool): Whether to print debug information.
114
115    Examples:
116        >>> import simetri.graphics as sg
117        >>> curve = sg.HobbyCurve([(0, 0), (40, 40), (80, 0)])
118        >>> [[round(float(c), 6) for c in q[:2]] for q in curve.get_ctrl_points()]
119        [[0.0, 22.09139], [17.90861, 40.0], [62.09139, 40.0], [80.0, 22.09139]]
120    """
121
122    def __init__(
123        self,
124        points: list[tuple[float, float]],
125        tension: float = 1,
126        cyclic: bool = False,
127        begin_curl: float = 1,
128        end_curl: float = 1,
129        debug: bool = False,
130    ) -> None:
131        """Initialize a HobbyCurve with the given parameters.
132
133        Args:
134            points: List of (x, y) tuples representing the curve's points.
135            tension: Tension parameter controlling the "tightness" of the curve.
136            cyclic: Whether the curve should be closed.
137            begin_curl: Curl value for the beginning of the curve.
138            end_curl: Curl value for the end of the curve.
139            debug: Whether to print debug information.
140
141        Raises:
142            ValueError: If fewer than 2 points are provided.
143        """
144        if len(points) < 2:
145            raise ValueError("Algorithm needs more than 2 points")
146        self.points = [HobbyPoint(*point, tension) for point in points]
147        self.ctrl_pts = []
148        self.is_cyclic = cyclic
149        self.begin_curl = begin_curl
150        self.end_curl = end_curl
151        self.n_points = len(points)
152        self.debug_mode = debug
153
154    def get_ctrl_points(self) -> list[tuple]:
155        """Calculate and return all of the control points of the Hobby curve.
156
157        Executes the Hobby algorithm by calculating distance values, psi values,
158        theta values, and phi values, then uses these to compute control points.
159
160        Returns:
161            A list of (x, y) tuples representing the Bezier control points.
162
163        Examples:
164            >>> import simetri.graphics as sg
165            >>> curve = sg.HobbyCurve([(0, 0), (40, 40), (80, 0)])
166            >>> len(curve.get_ctrl_points())
167            4
168        """
169        self.calculate_d_vals()
170        self.calculate_psi_vals()
171        self.calculate_theta_vals()
172        self.calculate_phi_vals()
173        self.show_debug_msg()
174        self.ctrl_pts = self.calculate_ctrl_pts()
175        return self.ctrl_pts
176
177    def calculate_d_vals(self) -> None:
178        """Calculate the pairwise distances between consecutive points in the curve.
179
180        Examples:
181            >>> import simetri.graphics as sg
182            >>> curve = sg.HobbyCurve([(0, 0), (60, 0), (60, 80)])
183            >>> curve.calculate_d_vals()
184            >>> round(curve.points[0].d_val, 6)
185            60.0
186        """
187        # Skip last point if path is non-cyclic
188        point_inds = (
189            range(self.n_points) if self.is_cyclic else range(self.n_points - 1)
190        )
191        for i in point_inds:
192            z_i = self.points[i % self.n_points]
193            z_j = self.points[(i + 1) % self.n_points]
194            z_i.d_val = abs(z_i - z_j)
195
196    def calculate_psi_vals(self) -> None:
197        """Calculate the psi values by finding the angle of the polygonal turns.
198
199        Raises:
200            ZeroDivisionError: If consecutive points have the same coordinates.
201
202        Examples:
203            >>> import simetri.graphics as sg
204            >>> curve = sg.HobbyCurve([(0, 0), (40, 0), (40, 40)])
205            >>> curve.calculate_d_vals()
206            >>> curve.calculate_psi_vals()
207            >>> round(curve.points[1].psi, 6)
208            1.570796
209        """
210        # Skip first and last point if path is non-cyclic
211        point_inds = (
212            range(self.n_points)
213            if self.is_cyclic
214            else range(1, self.n_points - 1)
215        )
216        for i in point_inds:
217            z_h = self.points[i - 1]
218            z_i = self.points[i]
219            z_j = self.points[(i + 1) % self.n_points]
220            try:
221                polygonal_turn = (z_j - z_i) / (z_i - z_h)
222                # print(z_j - z_i, z_i - z_h)
223            except ZeroDivisionError:
224                raise ZeroDivisionError(
225                    f"Consecutive points {(z_h.x, z_h.y)} and {(z_i.x, z_i.y)} cause zero division."
226                )
227            z_i.psi = np.arctan2(polygonal_turn.imag, polygonal_turn.real)
228
229    def calculate_theta_vals(self) -> None:
230        """Calculate the theta values by solving a linear system of equations.
231
232        This is the core of Hobby's algorithm, creating and solving a system of equations
233        to find the optimal angles for smooth splines.
234
235        Examples:
236            >>> import simetri.graphics as sg
237            >>> curve = sg.HobbyCurve([(0, 0), (40, 40), (80, 0)])
238            >>> _ = curve.get_ctrl_points()
239            >>> round(float(curve.points[0].theta), 6)
240            0.785398
241        """
242        A = np.zeros(
243            self.n_points
244        )  # Inappropriate names, but they mirror Knuth's notation.
245        B = np.zeros(self.n_points)
246        C = np.zeros(self.n_points)
247        D = np.zeros(self.n_points)
248        R = np.zeros(self.n_points)
249
250        # Calculate the entries of the five vectors.
251        # Skip first and last point if path is non-cyclic.
252        point_ind = (
253            range(self.n_points)
254            if self.is_cyclic
255            else range(1, self.n_points - 1)
256        )
257        for i in point_ind:
258            z_h = self.points[i - 1]
259            z_i = self.points[i]
260            z_j = self.points[(i + 1) % self.n_points]
261
262            A[i] = z_h.alpha / (z_i.beta**2 * z_h.d_val)
263            B[i] = (3 - z_h.alpha) / (z_i.beta**2 * z_h.d_val)
264            C[i] = (3 - z_j.beta) / (z_i.alpha**2 * z_i.d_val)
265            D[i] = z_j.beta / (z_i.alpha**2 * z_i.d_val)
266            R[i] = -B[i] * z_i.psi - D[i] * z_j.psi
267
268        # Set up matrix M such that the soln. Mx = R are the theta values.
269        M = np.zeros((self.n_points, self.n_points))
270        for i in range(self.n_points):
271            # Fill i-th row of M
272            M[i][i - 1] = A[i]
273            M[i][i] = B[i] + C[i]
274            M[i][(i + 1) % self.n_points] = D[i]
275
276        # Special formulas for first and last rows of M with non-cyclic paths.
277        if not self.is_cyclic:
278            # First row of M
279            alpha_0 = self.points[0].alpha
280            beta_1 = self.points[1].beta
281            xi_0 = (alpha_0**2 * self.begin_curl) / beta_1**2
282            M[0][0] = alpha_0 * xi_0 + 3 - beta_1
283            M[0][1] = (3 - alpha_0) * xi_0 + beta_1
284            R[0] = -((3 - alpha_0) * xi_0 + beta_1) * self.points[1].psi
285            # Last row of M
286            alpha_n_1 = self.points[-2].alpha
287            beta_n = self.points[-1].beta
288            xi_n = (beta_n**2 * self.end_curl) / alpha_n_1**2
289            M[-1][-2] = (3 - beta_n) * xi_n + alpha_n_1
290            M[-1][-1] = beta_n * xi_n + 3 - alpha_n_1
291            R[-1] = 0
292
293        # Solve for theta values.
294        thetas = np.linalg.solve(M, R)
295        for i, point in enumerate(self.points):
296            point.theta = thetas[i]
297
298    def calculate_phi_vals(self) -> None:
299        """Calculate the phi values using the relationship theta + phi + psi = 0.
300
301        Examples:
302            >>> import simetri.graphics as sg
303            >>> curve = sg.HobbyCurve([(0, 0), (40, 40), (80, 0)])
304            >>> _ = curve.get_ctrl_points()
305            >>> round(float(curve.points[1].phi + curve.points[1].theta + curve.points[1].psi), 10)
306            0.0
307        """
308        for point in self.points:
309            point.phi = -(point.psi + point.theta)
310
311    def calculate_ctrl_pts(self) -> list[tuple]:
312        """Calculate the Bezier control points between consecutive points.
313
314        Returns:
315            A list of (x, y) tuples representing the control points, with two control points
316            for each curve segment between consecutive points.
317
318        Examples:
319            >>> import simetri.graphics as sg
320            >>> curve = sg.HobbyCurve([(0, 0), (40, 40), (80, 0)])
321            >>> _ = curve.get_ctrl_points()
322            >>> len(curve.calculate_ctrl_pts())
323            4
324        """
325        ctrl_pts = []
326        # Skip last point if path is non-cyclic
327        point_inds = (
328            range(self.n_points) if self.is_cyclic else range(self.n_points - 1)
329        )
330        for i in point_inds:
331            z_i = self.points[i]
332            z_j = self.points[(i + 1) % self.n_points]
333            rho_coefficient = z_i.alpha * velocity(z_i.theta, z_j.phi)
334            sigma_coefficient = z_j.beta * velocity(z_j.phi, z_i.theta)
335            ctrl_pt_a = z_i + (1 / 3) * rho_coefficient * cmath.exp(
336                complex(0, z_i.theta)
337            ) * (z_j - z_i)
338            ctrl_pt_b = z_j - (1 / 3) * sigma_coefficient * cmath.exp(
339                complex(0, -z_j.phi)
340            ) * (z_j - z_i)
341            ctrl_pts.append((ctrl_pt_a.real, ctrl_pt_a.imag))
342            ctrl_pts.append((ctrl_pt_b.real, ctrl_pt_b.imag))
343        return ctrl_pts
344

366def hobby_ctrl_points(
367    points: Sequence[PointType],
368    tension: float = 1,
369    cyclic: bool = False,
370    begin_curl: float = 1,
371    end_curl: float = 1,
372    debug: bool = False,
373) -> list[tuple]:
374    """Calculate cubic Bezier control points using John Hobby's algorithm.
375
376    Args:
377        points: List of (x, y) tuples representing the curve's points.
378        tension: Controls the "tightness" of the curve (lower is tighter).
379        cyclic: Whether the curve should be closed.
380        begin_curl: Curl value for the beginning of the curve.
381        end_curl: Curl value for the end of the curve.
382        debug: Whether to print debug information.
383
384    Returns:
385        A list of (x, y) tuples representing the Bezier control points.
386
387    Examples:
388
389        >>> import simetri.graphics as sg
390        >>> ctrl = sg.hobby_ctrl_points([(0, 0), (40, 40), (80, 0)], tension=1)
391        >>> [[round(float(c), 6) for c in q[:2]] for q in ctrl]
392        [[0.0, 22.09139], [17.90861, 40.0], [62.09139, 40.0], [80.0, 22.09139]]
393    """
394    curve = HobbyCurve(
395        points,
396        tension=tension,
397        cyclic=cyclic,
398        begin_curl=begin_curl,
399        end_curl=end_curl,
400        debug=debug,
401    )
402    ctrl_points = curve.get_ctrl_points()
403
404    # Calculate whitespace padding for pretty print.
405    max_pad = 0
406    for ctrl_point in ctrl_points:
407        x, y = ctrl_point[:2]
408        # Calculate number of digits in x, y before decimal, and take the
409        # max for nice padding.
410        padding = max(
411            1 if abs(x) <= 0.1 else int(np.ceil(np.log10(abs(x)))) + 1,
412            1 if abs(y) <= 0.1 else int(np.ceil(np.log10(abs(y)))) + 1,
413        )
414        max_pad = max(max_pad, padding)
415
416    if debug:
417        # Pretty print control points.
418        precision = 10
419        space = precision + max_pad + 1  # +1 for negative sign
420        i = 0
421        while i < len(ctrl_points) - 1:
422            x_1, y_1 = ctrl_points[i]
423            x_2, y_2 = ctrl_points[i + 1]
424            print(
425                f"({x_1:<{space}.{precision}f}, {y_1:<{space}.{precision}f}) "
426                f"and "
427                f"({x_2:<{space}.{precision}f}, {y_2:<{space}.{precision}f})"
428            )
429            i += 2
430
431    return ctrl_points

434def velocity(theta: float, phi: float) -> float:
435    """Calculate the "velocity" function used in Metafont's curve algorithm.
436
437    This function implements the specific velocity formula from Knuth's Metafont.
438
439    Args:
440        theta: The theta angle value.
441        phi: The phi angle value.
442
443    Returns:
444        The computed velocity value used in control point calculations.
445
446    Examples:
447        >>> import simetri.graphics as sg
448        >>> round(sg.velocity(0.0, 0.0), 6)
449        1.0
450    """
451    numerator = 2 + np.sqrt(2) * (np.sin(theta) - (1 / 16) * np.sin(phi)) * (
452        np.sin(phi) - (1 / 16) * np.sin(theta)
453    ) * (np.cos(theta) - np.cos(phi))
454    denominator = (
455        1
456        + (1 / 2) * (np.sqrt(5) - 1) * np.cos(theta)
457        + (1 / 2) * (3 - np.sqrt(5)) * np.cos(phi)
458    )
459    return numerator / denominator

462def hobby_shape(
463    points: Sequence[PointType],
464    cyclic: bool = False,
465    tension: float = 1,
466    begin_curl: float = 1,
467    end_curl: float = 1,
468    n_points: int | None = None,
469) -> Shape:
470    """Create a ``Shape`` along a Hobby spline through the given points.
471
472    Computes cubic Bezier control points with Hobby's algorithm, then samples
473    points along each segment.
474
475    Args:
476        points: List of ``(x, y)`` tuples defining the spline path.
477        cyclic: If True, close the curve by connecting the last point to the first.
478        tension: Controls curve tightness (lower is tighter). Defaults to 1.
479        begin_curl: Curl at the start of an open curve. Defaults to 1.
480        end_curl: Curl at the end of an open curve. Defaults to 1.
481        n_points: Samples per Bezier segment. Defaults to
482            ``runtime_defaults["n_hobby_points"]``.
483
484    Returns:
485        Shape: Polyline vertices along the smooth Hobby curve.
486
487    Examples:
488
489        >>> import simetri.graphics as sg
490        >>> shape = sg.hobby_shape([(0, 0), (40, 20), (20, 0)], n_points=5)
491        >>> [[round(float(c), 6) for c in q[:2]] for q in shape.vertices]
492        [[0.0, 0.0], [-3.912294, 19.985868], [8.297592, 33.404816], [26.358682, 35.121356], [40.0, 20.0], [40.0, 20.0], [40.252568, 11.133414], [36.079436, 3.920564], [28.866586, -0.252568], [20.0, 0.0]]
493    """
494    if n_points is None:
495        n_points = runtime_defaults["n_hobby_points"]
496    controls = hobby_ctrl_points(
497        points,
498        tension=tension,
499        cyclic=cyclic,
500        begin_curl=begin_curl,
501        end_curl=end_curl,
502    )
503    n = len(points)
504    res = []
505    if cyclic:
506        for i in range(n):
507            ind = i * 2
508            p0 = points[i]
509            p1 = controls[ind]
510            p2 = controls[ind + 1]
511            p3 = points[(i + 1) % n]
512            bez_pnts = bezier_points(p0, p1, p2, p3, n_points)
513            res.extend(bez_pnts)
514    else:
515        for i in range(n - 1):
516            ind = i * 2
517            p0 = points[i]
518            p1 = controls[ind]
519            p2 = controls[ind + 1]
520            p3 = points[(i + 1)]
521            bez_pnts = bezier_points(p0, p1, p2, p3, n_points)
522            res.extend(bez_pnts)
523    return Shape(res)

230    def calculate_theta_vals(self) -> None:
231        """Calculate the theta values by solving a linear system of equations.
232
233        This is the core of Hobby's algorithm, creating and solving a system of equations
234        to find the optimal angles for smooth splines.
235
236        Examples:
237            >>> import simetri.graphics as sg
238            >>> curve = sg.HobbyCurve([(0, 0), (40, 40), (80, 0)])
239            >>> _ = curve.get_ctrl_points()
240            >>> round(float(curve.points[0].theta), 6)
241            0.785398
242        """
243        A = np.zeros(
244            self.n_points
245        )  # Inappropriate names, but they mirror Knuth's notation.
246        B = np.zeros(self.n_points)
247        C = np.zeros(self.n_points)
248        D = np.zeros(self.n_points)
249        R = np.zeros(self.n_points)
250
251        # Calculate the entries of the five vectors.
252        # Skip first and last point if path is non-cyclic.
253        point_ind = (
254            range(self.n_points)
255            if self.is_cyclic
256            else range(1, self.n_points - 1)
257        )
258        for i in point_ind:
259            z_h = self.points[i - 1]
260            z_i = self.points[i]
261            z_j = self.points[(i + 1) % self.n_points]
262
263            A[i] = z_h.alpha / (z_i.beta**2 * z_h.d_val)
264            B[i] = (3 - z_h.alpha) / (z_i.beta**2 * z_h.d_val)
265            C[i] = (3 - z_j.beta) / (z_i.alpha**2 * z_i.d_val)
266            D[i] = z_j.beta / (z_i.alpha**2 * z_i.d_val)
267            R[i] = -B[i] * z_i.psi - D[i] * z_j.psi
268
269        # Set up matrix M such that the soln. Mx = R are the theta values.
270        M = np.zeros((self.n_points, self.n_points))
271        for i in range(self.n_points):
272            # Fill i-th row of M
273            M[i][i - 1] = A[i]
274            M[i][i] = B[i] + C[i]
275            M[i][(i + 1) % self.n_points] = D[i]
276
277        # Special formulas for first and last rows of M with non-cyclic paths.
278        if not self.is_cyclic:
279            # First row of M
280            alpha_0 = self.points[0].alpha
281            beta_1 = self.points[1].beta
282            xi_0 = (alpha_0**2 * self.begin_curl) / beta_1**2
283            M[0][0] = alpha_0 * xi_0 + 3 - beta_1
284            M[0][1] = (3 - alpha_0) * xi_0 + beta_1
285            R[0] = -((3 - alpha_0) * xi_0 + beta_1) * self.points[1].psi
286            # Last row of M
287            alpha_n_1 = self.points[-2].alpha
288            beta_n = self.points[-1].beta
289            xi_n = (beta_n**2 * self.end_curl) / alpha_n_1**2
290            M[-1][-2] = (3 - beta_n) * xi_n + alpha_n_1
291            M[-1][-1] = beta_n * xi_n + 3 - alpha_n_1
292            R[-1] = 0
293
294        # Solve for theta values.
295        thetas = np.linalg.solve(M, R)
296        for i, point in enumerate(self.points):
297            point.theta = thetas[i]

300        """Calculate the phi values using the relationship theta + phi + psi = 0.
301
302        Examples:

312    def calculate_ctrl_pts(self) -> list[tuple]:
313        """Calculate the Bezier control points between consecutive points.
314
315        Returns:
316            A list of (x, y) tuples representing the control points, with two control points
317            for each curve segment between consecutive points.
318
319        Examples:
320            >>> import simetri.graphics as sg
321            >>> curve = sg.HobbyCurve([(0, 0), (40, 40), (80, 0)])
322            >>> _ = curve.get_ctrl_points()
323            >>> len(curve.calculate_ctrl_pts())
324            4
325        """
326        ctrl_pts = []
327        # Skip last point if path is non-cyclic
328        point_inds = (
329            range(self.n_points) if self.is_cyclic else range(self.n_points - 1)
330        )
331        for i in point_inds:
332            z_i = self.points[i]
333            z_j = self.points[(i + 1) % self.n_points]
334            rho_coefficient = z_i.alpha * velocity(z_i.theta, z_j.phi)
335            sigma_coefficient = z_j.beta * velocity(z_j.phi, z_i.theta)
336            ctrl_pt_a = z_i + (1 / 3) * rho_coefficient * cmath.exp(
337                complex(0, z_i.theta)
338            ) * (z_j - z_i)
339            ctrl_pt_b = z_j - (1 / 3) * sigma_coefficient * cmath.exp(
340                complex(0, -z_j.phi)
341            ) * (z_j - z_i)
342            ctrl_pts.append((ctrl_pt_a.real, ctrl_pt_a.imag))
343            ctrl_pts.append((ctrl_pt_b.real, ctrl_pt_b.imag))
344        return ctrl_pts

291            M[-1][-1] = beta_n * xi_n + 3 - alpha_n_1
292            R[-1] = 0
293
294        # Solve for theta values.
295        thetas = np.linalg.solve(M, R)
296        for i, point in enumerate(self.points):
297            point.theta = thetas[i]

def velocity(theta: float, phi: float) -> float: 
435def velocity(theta: float, phi: float) -> float:
436    """Calculate the "velocity" function used in Metafont's curve algorithm.
437
438    This function implements the specific velocity formula from Knuth's Metafont.
439
440    Args:
441        theta: The theta angle value.
442        phi: The phi angle value.
443
444    Returns:
445        The computed velocity value used in control point calculations.
446
447    Examples:
448        >>> import simetri.graphics as sg
449        >>> round(sg.velocity(0.0, 0.0), 6)
450        1.0
451    """
452    numerator = 2 + np.sqrt(2) * (np.sin(theta) - (1 / 16) * np.sin(phi)) * (
453        np.sin(phi) - (1 / 16) * np.sin(theta)
454    ) * (np.cos(theta) - np.cos(phi))
455    denominator = (
456        1
457        + (1 / 2) * (np.sqrt(5) - 1) * np.cos(theta)
458        + (1 / 2) * (3 - np.sqrt(5)) * np.cos(phi)
459    )
460    return numerator / denominator
